  private func image(_ name: String, _ view: UIView) throws {
    try require(view.window != nil && !view.bounds.isEmpty, "Cannot capture mounted \(name)")

    let rendered = UIGraphicsImageRenderer(bounds: view.bounds).image { _ in
      drawn = view.drawHierarchy(in: view.bounds, afterScreenUpdates: true)
    }
    try require(drawn, "Actual hierarchy rendering unavailable: \(name)")

    a.name = "\(caseID)-\(name)"; a.lifetime = .keepAlways; add(a)
  }
  private func views(_ root: UIView) -> [UIView] { [root] + root.subviews.flatMap { views($0) } }
  private func visible(_ view: UIView) -> Bool {
    guard let w = view.window, !view.bounds.isEmpty else { return false }
    var child: UIView? = view
    while let v = child {
      if v.isHidden || v.alpha == 0 { return false }
      if v.clipsToBounds && !v.bounds.intersects(view.convert(view.bounds, to: v)) { return false }
      child = v.superview
    }
    return w.bounds.intersects(view.convert(view.bounds, to: w))
  }
  private func control(_ id: String) throws -> UIView {
    let deadline = Date().addingTimeInterval(10) // resource/UI availability, no latency oracle
    repeat {
      if let v = views(keyboard.rootView).first(where: { $0.accessibilityIdentifier == id && visible($0) }) { return v }
      pump()
    } while Date() < deadline
    try require(false, "Visible production control unavailable: \(id)")
    throw BindingError.unavailable
  }
  private func activate(_ view: UIView) throws {
    try require(visible(view), "Activation target is not mounted/visible")
    if let button = view as? UIControl {
      try require(button.isEnabled, "Production control disabled")
      button.sendActions(for: .touchUpInside)
    } else {
      try require(view.accessibilityActivate(), "Production accessibility activation failed")
    }
    pump()
  }

  private func hostState(_ before: String, _ after: String = "") throws {
    try require(equal(textView.text ?? "", before + after), "Literal host differs: \((textView.text ?? "").debugDescription)")

    guard let range = textView.selectedTextRange else { try require(false, "Host has no caret"); return }
    let rect = textView.caretRect(for: range.start)
    try require(textView.window != nil && range.isEmpty && !rect.isEmpty && rect.intersects(textView.bounds), "Host caret is not in the visible mounted editor")
  }

  }

  /// One contact, actual KeysView callbacks and its unchanged installed delegate.
  /// One-point increments are mechanics; the actual host caret alone stops it.
  private func cursor(_ action: Int, direction: Int, before: String, after: String = "") throws {
    let keys = keyboard.rootView.keys
    let space = try control("key-space")
    let start = space.convert(CGPoint(x: space.bounds.midX, y: space.bounds.midY), to: keys)
    let original = textView.text ?? ""
    let touch = R53NativeContact(in: keys, point: start)
    keys.touchesBegan([touch], with: nil)
    pump(0.65) // hold, then slide; never a fixed-distance correctness oracle
    let target = before.utf16.count
    while textView.selectedRange != NSRange(location: target, length: 0) {
      let next = CGPoint(x: touch.point.x + CGFloat(direction), y: start.y)
      guard keys.bounds.contains(next) else {
        keys.touchesCancelled([touch], with: nil)
        try require(false, "Caret target not reached within visible continuous contact surface; no retry/repair")
        return
      }
      touch.point = next; touch.contactPhase = .moved
      keys.touchesMoved([touch], with: nil); pump(1.0 / 60)
    }
    touch.contactPhase = .ended; keys.touchesEnded([touch], with: nil); pump()
    try require(equal(textView.text ?? "", original), "Cursor contact mutated host")
    try check(action, before, after)
  }

  private struct EmojiChoice {
    let value: String
    let choose: () throws -> Void
  }
  private func glyphVisible(_ image: UIImageView) -> Bool {
    guard visible(image), image.image != nil else { return false }
    // Require the actual glyph image, rather than the entire cell, to be unclipped.
    var ancestor = image.superview
    while let v = ancestor {
      if v.clipsToBounds && !v.bounds.contains(image.convert(image.bounds, to: v)) { return false }
      ancestor = v.superview
    }
    return true
  }
  private func observeEmoji(_ stage: String, search: Bool = false, matching: String? = nil, recent: Bool = false) throws -> EmojiChoice {
    let container = try control(search ? "emoji-search-results" : "emoji-panel")
    let choice: EmojiChoice
    if search {
      // Read the live control's render operand beside its actual UIImageView.
      // No search engine/result list, ID parsing, predetermined rank or post-insertion inference.
      let candidates = views(container).compactMap { $0 as? EmojiResultControl }.filter { glyphVisible($0.imageView) }
      guard let cell = candidates.first(where: { matching == nil || equal($0.result.value, matching!) }) else {
        try require(false, "No visible selectable full search emoji"); throw BindingError.unavailable
      }
      let value = cell.result.value
      choice = EmojiChoice(value: value) { [weak self, weak cell] in
        guard let self, let cell else { throw BindingError.unavailable }
        try self.require(self.glyphVisible(cell.imageView) && self.equal(cell.result.value, value), "Observed search presentation changed")
        try self.activate(cell)
      }
    } else {
      guard let collection = views(container).compactMap({ $0 as? UICollectionView }).first else {
        try require(false, "Mounted production emoji collection missing"); throw BindingError.unavailable
      }
      func shown() -> [(EmojiCell, IndexPath)] {
        collection.visibleCells.compactMap { cell in
          guard let cell = cell as? EmojiCell, let path = collection.indexPath(for: cell),
                (!recent || path.section == recentSection), glyphVisible(cell.imageView),
                let value = cell.emoji, matching == nil || equal(value, matching!) else { return nil }
          return (cell, path)
        }
      }
      // Browse only the actual recents section if a matching entry is offscreen.
      // Section comes from the actual activated Frequently Used button, never a
      // presumed category position, emoji identity, history count or ranking.
      while shown().isEmpty, recent {
        let next = min(collection.contentOffset.x + collection.bounds.width / 2,
                       max(0, collection.contentSize.width - collection.bounds.width))
        guard next > collection.contentOffset.x else { break }
        collection.setContentOffset(CGPoint(x: next, y: collection.contentOffset.y), animated: false); pump()
        if !collection.indexPathsForVisibleItems.contains(where: { $0.section == recentSection }) { break }
      }
      guard let (cell, path) = shown().first, let value = cell.emoji else {
        try require(false, "Recorded emoji not visibly available in requested collection section"); throw BindingError.unavailable
      }
      choice = EmojiChoice(value: value) { [weak self, weak cell, weak collection] in
        guard let self, let cell, let collection else { throw BindingError.unavailable }
        try self.require(collection.cellForItem(at: path) === cell && self.glyphVisible(cell.imageView) && self.equal(cell.emoji ?? "", value), "Observed collection cell changed")
        try self.require(collection.delegate != nil, "Production collection delegate missing")
        collection.selectItem(at: path, animated: false, scrollPosition: [])
        collection.delegate?.collectionView?(collection, didSelectItemAt: path)
        self.pump()
      }
    }
    try require(!choice.value.isEmpty, "Displayed emoji is empty")
    note(stage + "-full-emoji-before-selection", choice.value + "\n" + choice.value.unicodeScalars.map { String(format: "U+%04X", $0.value) }.joined(separator: " "))
    try image(stage + "-displayed-emoji-before-selection", container)
    return choice
  }

  /// Actual public SwiftUI accessibility hierarchy, including virtual elements.
  private func accessibilityObjects(_ root: NSObject) -> [NSObject] {
    var seen = Set<ObjectIdentifier>()
    func walk(_ object: NSObject) -> [NSObject] {
      guard seen.insert(ObjectIdentifier(object)).inserted else { return [] }
      var children = (object as? UIView)?.subviews.map { $0 as NSObject } ?? []
      if let elements = object.accessibilityElements { children += elements.compactMap { $0 as? NSObject } }
      let count = object.accessibilityElementCount()
      if count > 0 && count != NSNotFound {
        for index in 0..<count { if let e = object.accessibilityElement(at: index) as? NSObject { children.append(e) } }
      }
      return [object] + children.flatMap(walk)
    }
    return walk(root)
  }
  private func setting(_ title: String) throws -> NSObject {
    let root = settingsPage.view!
    func found() -> NSObject? {
      accessibilityObjects(root).first { object in
        let label = object.accessibilityLabel ?? ""
        let named = label == title || label.hasPrefix(title + ",") || label.hasPrefix(title + "\n")
        let frame = object.accessibilityFrame
        return named && !frame.isEmpty && root.convert(root.bounds, to: nil).intersects(frame)
          && (object is UISwitch || object.accessibilityTraits.contains(.button) || object.accessibilityValue != nil)
      }
    }
    let scrolls = views(root).compactMap { $0 as? UIScrollView }.filter { $0.contentSize.height > $0.bounds.height }
    for scroll in scrolls { scroll.setContentOffset(CGPoint(x: scroll.contentOffset.x, y: -scroll.adjustedContentInset.top), animated: false) }
    pump()
    if let result = found() { return result }
    for scroll in scrolls {
      while scroll.contentOffset.y < scroll.contentSize.height - scroll.bounds.height + scroll.adjustedContentInset.bottom {
        let next = min(scroll.contentOffset.y + scroll.bounds.height / 2,
                       scroll.contentSize.height - scroll.bounds.height + scroll.adjustedContentInset.bottom)
        scroll.setContentOffset(CGPoint(x: scroll.contentOffset.x, y: next), animated: false); pump()
        if let result = found() { return result }
      }
    }
    try require(false, "Real production settings control unavailable: \(title); no model/store fallback")
    throw BindingError.unavailable
  }
  private func displayedValue(_ object: NSObject) throws -> Bool {
    if let toggle = object as? UISwitch { return toggle.isOn }
    switch object.accessibilityValue?.lowercased() {
    case "1", "on": return true
    case "0", "off": return false
    default: try require(false, "Real settings control has no readable displayed Boolean"); throw BindingError.unavailable
    }
  }
  private func visit(_ action: Int, _ changes: [(String, Bool)], _ before: String) throws {
    let originalRange = textView.selectedRange
    try tap("keyboard-settings") // production button -> PreviewHost openURL -> mounted HomeView
    try require(editor.presentedViewController === settingsPage, "Real app root settings did not open")
    pump(0.1)
    for (title, desired) in changes {
      let toggle = try setting(title)
      if try displayedValue(toggle) != desired {
        try require(!toggle.accessibilityTraits.contains(.notEnabled), "Named real toggle is disabled")
        if let uiSwitch = toggle as? UISwitch {
          try require(uiSwitch.isEnabled, "Named UISwitch disabled")
          uiSwitch.setOn(desired, animated: false); uiSwitch.sendActions(for: .valueChanged)
        } else {
          try require(toggle.accessibilityActivate(), "Real SwiftUI toggle activation unavailable; no preference write substitute")
        }
        pump()
      }
      expectedSettings[title] = desired
    }
    for title in settingNames {
      let toggle = try setting(title)
      try require(try displayedValue(toggle) == expectedSettings[title]!, "Displayed independent setting differs: \(title)")
      note("a\(action)-setting-\(title)", "displayed=\(try displayedValue(toggle))")
      try image("a\(action)-settings-\(title)", settingsPage.view)
    }
    settingsPage.dismiss(animated: false); pump()
    try require(textView.becomeFirstResponder(), "Cannot return to mounted keyboard")
    pump()
    try require(textView.selectedRange == originalRange, "Settings visit changed host caret")
    try check(action, before)
  }
  private func glide(_ action: Int, _ path: String, trail: Bool) throws -> String {
    let keys = keyboard.rootView.keys
    let points = try path.map { c -> CGPoint in
      let cap = try control("key-\(c)")
      return cap.convert(CGPoint(x: cap.bounds.midX, y: cap.bounds.midY), to: keys)
    }
    try require(points.count >= 2, "Authored visible key path unavailable")
    let original = textView.text ?? ""
    let touch = R53NativeContact(in: keys, point: points[0])
    keys.touchesBegan([touch], with: nil)
    for (start, end) in zip(points, points.dropFirst()) {
      for step in 1...12 {
        let fraction = CGFloat(step) / 12
        touch.point = CGPoint(x: start.x + (end.x - start.x) * fraction, y: start.y + (end.y - start.y) * fraction)
        touch.contactPhase = .moved; keys.touchesMoved([touch], with: nil); pump(1.0 / 60)
      }
    }
    // Capture while the SAME contact remains down, before touchesEnded/fade.
    try image("a\(action)-DURING-CONTACT-trail", keyboard.rootView)
    note("a\(action)-trail-manual-check-PENDING", "Fixed frame: \(caseID)-a\(action)-DURING-CONTACT-trail. Authored trail visible=\(trail). Main must adjudicate the actual rendered image; no layer/settings oracle, later replacement frame, retry or rescue.")
    touch.contactPhase = .ended; keys.touchesEnded([touch], with: nil); pump()
    let actual = textView.text ?? ""
    // Only nonempty changed committed text, no exact word/rank/spacing oracle.
    let shared = zip(original, actual).prefix(while: { $0.0 == $0.1 }).count
    let committed = String(actual.dropFirst(shared))
    try require(!committed.isEmpty && !equal(actual, original), "No nonempty slide-typed span committed")
    note("a\(action)-actual-slide-host", "before=\(original.debugDescription)\nactual=\(actual.debugDescription)\nnewSpan=\(committed.debugDescription)")
    try check(action, actual)
    return actual
  }
  private func physical(_ action: Int) {
    note("a\(action)-physical-feedback", "Physical system-click audibility: unavailable. Tactile haptics: unavailable. Assert displayed independent settings and editing continuity only.")
  }

/// Test event carrier only: no engine/controller calls, KVC or private HID API.
/// One object retains contact identity across began/moved/ended callbacks.
@MainActor
private final class R53NativeContact: UITouch {
  weak var target: UIView?
  var point: CGPoint
  var contactPhase: UITouch.Phase = .began
  init(in view: UIView, point: CGPoint) { target = view; self.point = point; super.init() }
  override var view: UIView? { target }
  override var window: UIWindow? { target?.window }
  override var timestamp: TimeInterval { ProcessInfo.processInfo.systemUptime }
  override var phase: UITouch.Phase { contactPhase }
  override func location(in view: UIView?) -> CGPoint { target?.convert(point, to: view) ?? point }
  override func previousLocation(in view: UIView?) -> CGPoint { location(in: view) }
}
