import Testing
@testable import KeyboardCore

/// Analytic new controlled examples. No word oracle or recorded-human claim.
struct GlideContactReferenceR56Tests {
  @Test func stationaryArrivalAndReleaseAreBothNecessary() throws {
    let a = KeyPoint(x: 0, y: 0), b = KeyPoint(x: 2, y: 0)
    let stopped = try #require(GlideDecoder.liftSpeed(points: [a, b, b], times: [0, 0.2, 0.35], window: 0.05))
    let firstOnly = try #require(GlideDecoder.liftSpeed(points: [a, b], times: [0, 0.2], window: 0.05))
    let latestOnly = try #require(GlideDecoder.liftSpeed(points: [a, b], times: [0, 0.35], window: 0.05))
    #expect(stopped == 0)
    #expect(firstOnly == 1)
    #expect(latestOnly == 1)
  }

  @Test func firstLatestInteriorCompressionChangesTheDiscreteWindow() throws {
    let a = KeyPoint(x: 0, y: 0), b = KeyPoint(x: 2, y: 0), c = KeyPoint(x: 3, y: 0)
    let raw = try #require(GlideDecoder.liftSpeed(points: [a, b, b, b, c], times: [0, 0.2, 0.24, 0.27, 0.3], window: 0.05))
    let compressed = try #require(GlideDecoder.liftSpeed(points: [a, b, b, c], times: [0, 0.2, 0.27, 0.3], window: 0.05))
    #expect(abs(raw - 5.0 / 3) < 1e-12)
    #expect(abs(compressed - 1) < 1e-12)
  }
}
