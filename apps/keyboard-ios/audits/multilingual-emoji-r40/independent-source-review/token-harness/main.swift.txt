import Foundation
var checked=0,failures:[[String:Any]]=[],intersection=0,marks=0
func check(_ s:String) {
 checked += 1
 let expected=referenceTokens(s),actual=EmojiSuggester.auditTokens(s)
 if actual != expected { failures.append(["query":s,"expected":expected,"actual":actual]) }
}
for i in 0..<128 {for j in 0..<128 { check(String(UnicodeScalar(i)!)+String(UnicodeScalar(j)!)); check("Ab"+String(UnicodeScalar(i)!)+String(UnicodeScalar(j)!)+"CD") }}
for i in 0...0x10ffff {
 guard let scalar=UnicodeScalar(i) else {continue}
 let category=scalar.properties.generalCategory
 let base=[Unicode.GeneralCategory.uppercaseLetter,.lowercaseLetter,.titlecaseLetter,.modifierLetter,.otherLetter].contains(category)
 let mark=[Unicode.GeneralCategory.nonspacingMark,.spacingMark,.enclosingMark].contains(category)
 if CharacterSet.letters.contains(scalar) != base { intersection += 1;check(String(scalar));check("k"+String(scalar)+"a");check("a"+String(scalar)+"k") }
 if mark { marks += 1;for pre in ["","k","a","a\u{0301}","k\u{0345}"] {check(pre+String(scalar)+"b")} }
}
for s in ["\u{0345}K","\u{093e}p","a'\u{093e}k","k\u{ff9e}a","\u{0600} a","İ I ı i","Ñ ñ ñ","Årbol Árbol ARBOL","a\u{20dd} k\u{20dd}","\u{1d15e}","👩‍❤️‍👨 abc",String(repeating:"a",count:81)] {check(s)}
let result:[String:Any]=["checks":checked,"foundationCategoryDisagreementScalars":intersection,"markScalars":marks,"failures":failures]
let output=try! JSONSerialization.data(withJSONObject:result,options:[.sortedKeys,.prettyPrinted]);try! output.write(to:URL(fileURLWithPath:"/tmp/pulpo-r40-source-review/token-controls.json"));print("token checks \(checked), failures \(failures.count)")
