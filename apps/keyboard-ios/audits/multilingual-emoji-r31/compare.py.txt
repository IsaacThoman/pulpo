"""Frozen before reveal: retain EVERY changed raw row, including fail-to-fail."""
import json,sys
from pathlib import Path
base=json.loads(Path(sys.argv[1]).read_text())['rows']; cand=json.loads(Path(sys.argv[2]).read_text())['rows']
assert [r['id'] for r in base]==[r['id'] for r in cand]
changes=[];summary={}
for label,rows in [('baseline',base),('candidate',cand)]:
 useful=[r for r in rows if r['role']=='useful']; controls=[r for r in rows if r['role']=='control']
 summary[label]={'rows':len(rows),'usefulRows':len(useful),'usefulPassed':sum(r['passed'] for r in useful),
 'usefulTop3':sum(r['rank'] is not None and r['rank']<=3 for r in useful),
 'controlsPassed':sum(r['passed'] for r in controls),'controls':len(controls),
 'unavailableRows':sum(bool(r['unavailable']) for r in rows),'unavailableTargets':sum(len(r['unavailable']) for r in rows),
 'acceptedAlternativesTop8':sum(len(r['acceptedTop8']) for r in useful),
 'uniqueAcceptedEmoji':sorted(set(e for r in useful for e in r['acceptedTop8'])),
 'families':sorted(set(r['family'] for r in useful)),
 'familiesAnyUseful':sorted(set(r['family'] for r in useful if r['passed'])),
 'byStratum':{s:{'rows':sum(r['stratum']==s for r in useful),'passed':sum(r['passed'] for r in useful if r['stratum']==s)} for s in ['en','es','mixed']}}
for b,c in zip(base,cand):
 if b!=c:changes.append({'id':b['id'],'baseline':b,'candidate':c,'lostEnglishOrOtherValues':[e for e in b['values'] if e not in c['values']],
 'lostAcceptedTop8':[e for e in b['acceptedTop8'] if e not in c['acceptedTop8']],
 'status':('pass' if b['passed'] else 'fail')+'->'+('pass' if c['passed'] else 'fail')})
summary['changedRows']=len(changes)
summary['newFailures']=[c['id'] for b,c in zip(base,cand) if b['passed'] and not c['passed']]
summary['failedCandidateRows']=[r['id'] for r in cand if not r['passed']]
Path(sys.argv[3]).write_text(json.dumps({'summary':summary,'changes':changes},ensure_ascii=False,indent=2)+'\n')
print(json.dumps(summary,ensure_ascii=False,indent=2))
