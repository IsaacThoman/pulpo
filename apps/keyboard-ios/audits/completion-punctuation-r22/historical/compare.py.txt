#!/usr/bin/env python3
"""Frozen paired scorer: retain every stack/task/mode, including missing choices."""
import json,sys,collections,statistics
b,c=[json.load(open(p)) for p in sys.argv[1:3]]
key=lambda r:(r['id'],r['stack'],r['mode'])
assert len(b)==len(c) and len({key(r) for r in b})==len(b)
assert [key(r) for r in b]==[key(r) for r in c]
flags=['displayed','accepted','fullPass','suffixPreserved','literalDisplayed','literalExact','literalUndo','deletionUndo','autoCaseExact','staleSelectionSafe','staleCaretSafe']
def counts(rows):
 return dict(n=len(rows),**{f:sum(r[f] and (r['displayed'] if f.startswith('stale') else True) for r in rows) for f in flags},unavailable=sum(not r['displayed'] for r in rows),privateSuggestions=sum(r['privateSlots'] for r in rows),privateLearning=sum(r['privatePersonalCount'] for r in rows),lexiconPresent=sum(r['targetInLexicon'] for r in rows))
paired=[];regressions=[]
for x,y in zip(b,c):
 losses=[f for f in flags if x[f] and not y[f]]
 for f in ['suggestions','autocorrection','autoCommit','autoBackspace','privateDocument']:
  if x.get(f)!=y.get(f):losses.append(f+' changed')
 if y['privateSlots'] or y['privatePersonalCount']:losses.append('private policy')
 if losses:regressions.append(dict(key=key(x),losses=losses))
 paired.append(dict(key=key(x),gains=[f for f in flags if not x[f] and y[f]],losses=losses,baseline=x,candidate=y))
def times(rows,f):
 v=sorted(r[f] for r in rows if r.get(f) is not None)
 return dict(n=len(v),median=statistics.median(v) if v else None,p95=v[min(len(v)-1,int(len(v)*.95))] if v else None,max=max(v) if v else None)
summary=dict(baseline=counts(b),candidate=counts(c),regressions=regressions,
 gainedFullPass=[key(x) for x,y in zip(b,c) if not x['fullPass'] and y['fullPass']],
 perStack={s:dict(baseline=counts([r for r in b if r['stack']==s]),candidate=counts([r for r in c if r['stack']==s])) for s in sorted({r['stack'] for r in b})},
 timing={label:{f:times(rows,f) for f in ['firstMicroseconds','selectMicroseconds']} for label,rows in [('baseline',b),('candidate',c)]})
json.dump(dict(summary=summary,cases=paired),open(sys.argv[3],'w'),ensure_ascii=False,sort_keys=True,separators=(',',':'))
print(json.dumps({k:v for k,v in summary.items() if k not in ['gainedFullPass','perStack']},indent=2))
print('Gained full passes:',len(summary['gainedFullPass']))
