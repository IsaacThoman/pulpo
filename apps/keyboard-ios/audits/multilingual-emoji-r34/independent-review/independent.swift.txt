import Foundation

@main struct IndependentReview {
  static func catalog(_ entries: [[String: Any]]) throws -> EmojiCatalog {
    try JSONDecoder().decode(EmojiCatalog.self, from: JSONSerialization.data(withJSONObject:
      ["version": "fixture", "categories": [["id": "fixture", "name": "Fixture", "emoji": entries]]]))
  }
  static func main() throws {
    let args = CommandLine.arguments
    let output = URL(fileURLWithPath: args[3])
    let queries = try JSONDecoder().decode([String].self, from: Data(contentsOf: URL(fileURLWithPath: args[2])))
    let index = EmojiSuggester(catalog: try EmojiCatalog.load(url: URL(fileURLWithPath: args[1])))
    var compatibility: [[String: Any]] = []
    for query in queries {
      compatibility.append(["query": query, "search": index.search(query).map(\.value),
        "suggestions": index.suggestions(for: query).map(\.value),
        "exactSuggestions": index.suggestions(for: query, limit: 8, prefixes: false).map(\.value)])
    }
    try JSONSerialization.data(withJSONObject: compatibility, options: [.sortedKeys]).write(to: output.appendingPathComponent("compat.json"))
    var controls: [[String: Any]] = []
    for name in ["copper lantern", "copper"] {
      let fixture = try catalog([["e": "r34-fixture-copper", "n": name, "k": name == "copper" ? ["beaco"] : ["beacon"], "s": "ñavoru", "v": 1]])
      let candidate = EmojiSuggester(catalog: fixture)
      for query in ["ñ", "Ñ", "n\u{303}", "ñavoru", "navoru"] {
        for cap in [1, 48, 60] {
          controls.append(["fixture": name, "query": query, "cap": cap,
            "actual": candidate.search(query, limit: cap).map(\.value)])
        }
      }
    }
    let variants: [(String, String, [String])] = [
      ("apostrophe", "l'or", ["l'or", "or l", "l or or"]),
      ("curly", "l’or", ["l’or", "or l", "or or l"]),
      ("vowelMc", "a\u{093E}b", ["a\u{093E}b", "ab", "AB AB"]),
      ("consonantMc", "c\u{093E}b", ["c\u{093E}b", "C\u{093E}B", "c\u{093E}b c\u{093E}b"]),
      ("vowelMe", "a\u{20DD}b", ["a\u{20DD}b", "ab", "ab ab"]),
      ("consonantMn", "c\u{034F}b", ["c\u{034F}b", "C\u{034F}B", "c\u{034F}b c\u{034F}b"]),
      ("newLetter", "a\u{105C0}b", ["a\u{105C0}b", "a b", "b a a"]),
    ]
    for (family, english, variants) in variants {
      for reservationCount in [7, 60] {
        var entries: [[String: Any]] = (0..<reservationCount).map { i in
          ["e": "reserved-\(i)", "n": english, "v": 1]
        }
        entries += (0..<80).map { i in
          ["e": "local-\(i)", "n": "zzopaque", "s": variants[0], "v": 1]
        }
        let fixture = try catalog(entries)
        let candidate = EmojiSuggester(catalog: fixture)
        for query in variants {
          for cap in [1, 48, 60] {
            for preferences in [[:], ["reserved-0": 2, "local-79": 2]] {
              controls.append(["fixture": family, "reservationCount": reservationCount,
                "query": query, "cap": cap, "preferred": preferences,
                "actual": candidate.search(query, limit: cap, preferring: preferences).map(\.value)])
            }
          }
        }
      }
    }
    try JSONSerialization.data(withJSONObject: controls, options: [.prettyPrinted, .sortedKeys]).write(to: output.appendingPathComponent("controls.json"))
    print("completed \(compatibility.count) compatibility rows, \(controls.count) fixture rows")
  }
}
