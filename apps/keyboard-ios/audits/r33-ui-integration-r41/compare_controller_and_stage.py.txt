#!/usr/bin/env python3
"""Exposed matched controller comparison, then explicitly version R33 dependency."""
import hashlib, json, pathlib, subprocess
ROOT=pathlib.Path(__file__).resolve().parents[4]
RAW=pathlib.Path('/tmp/pulpo-r41-validation')
DD=RAW/'DerivedData'
UI=['apps/keyboard-ios/App/PulpoKeyboardApp.swift','apps/keyboard-ios/PulpoKeyboardKit/Sources/KeyboardUI/KeyboardController.swift','apps/keyboard-ios/PulpoKeyboardKit/Sources/KeyboardUI/SuggestionBarView.swift']
DEVICE='F3191384-9F35-4AC4-A739-37E6C7C8D584'

def freeze(label):
    files=UI+['apps/keyboard-ios/UITests/LearningForgetControllerTests.swift','apps/keyboard-ios/PulpoKeyboardKit/Sources/KeyboardCore/Dictionary/PersonalDictionary.swift','apps/keyboard-ios/PulpoKeyboardKit/Sources/KeyboardCore/Shared/LearnedDataPersistence.swift']
    value={'files':{p:hashlib.sha256((ROOT/p).read_bytes()).hexdigest() for p in files},'binaries':{str(p):hashlib.sha256(p.read_bytes()).hexdigest() for p in (DD/'Build/Products').rglob('*') if p.is_file() and p.name in ['PulpoKeyboard','PulpoKeyboardUITests','PulpoKeyboardAppTests']}}
    (RAW/(label+'-source-binary-hashes.json')).write_text(json.dumps(value,indent=2)+'\n')

def native(label, build=False):
    cmd=['xcodebuild','build-for-testing' if build else 'test-without-building','-project','apps/keyboard-ios/PulpoKeyboard.xcodeproj','-scheme','PulpoKeyboard','-destination','platform=iOS Simulator,id='+DEVICE,'-derivedDataPath',str(DD),'-parallel-testing-enabled','NO','-collect-test-diagnostics','never','-resultBundlePath',str(RAW/(label+'.xcresult'))]
    if not build:cmd+=['-only-testing:PulpoKeyboardUITests/LearningForgetControllerTests']
    with open(RAW/(label+'.log'),'w') as log:code=subprocess.call(cmd,cwd=ROOT,stdout=log,stderr=subprocess.STDOUT)
    print(label+'_exit='+str(code),flush=True)
    if build:
        if code:raise RuntimeError('Build failed; inspect indicated error')
        return
    s=subprocess.check_output(['xcrun','xcresulttool','get','test-results','summary','--path',str(RAW/(label+'.xcresult')),'--compact'],text=True)
    (RAW/(label+'-summary.json')).write_text(s+'\n');j=json.loads(s)
    print(json.dumps({k:j.get(k) for k in ['totalTestCount','passedTests','failedTests','skippedTests','testFailures']}),flush=True)
    return j

for label,commit,expected in [('baseline-controller-v6','403a87ba',1),('candidate-controller-v3','c156797b',0)]:
    for p in UI:(ROOT/p).write_bytes(subprocess.check_output(['git','show',commit+':'+p],cwd=ROOT))
    native(label+'-build',build=True);freeze(label)
    result=native(label)
    if result['totalTestCount']!=1 or result['failedTests']!=expected:raise RuntimeError('Unexpected matched controller outcome; stop before dependency change')

old='2357e19156e79edc9842a3b9626d9ab18b3f6f36'
new='4fa9b1e7b39cb1e5db835eeb3c4b111f6540c566'
files=subprocess.check_output(['git','diff','--name-only',old+'..'+new,'--','apps/keyboard-ios/App','apps/keyboard-ios/PulpoKeyboardKit'],cwd=ROOT,text=True).splitlines()
assert set(files)=={'apps/keyboard-ios/PulpoKeyboardKit/Sources/KeyboardCore/Dictionary/PersonalDictionary.swift','apps/keyboard-ios/PulpoKeyboardKit/Sources/KeyboardCore/Shared/LearnedDataPersistence.swift','apps/keyboard-ios/PulpoKeyboardKit/Tests/KeyboardCoreTests/LearningPersistenceTests.swift'}
for p in files:assert (ROOT/p).read_bytes()==subprocess.check_output(['git','show',old+':'+p],cwd=ROOT),p
patch=subprocess.check_output(['git','diff','--binary',old+'..'+new,'--',*files],cwd=ROOT)
(RAW/'r33-dependency-v2.patch').write_bytes(patch)
subprocess.run(['git','apply','--check'],cwd=ROOT,input=patch,check=True)
subprocess.run(['git','apply'],cwd=ROOT,input=patch,check=True)
hashes={}
for p in files:
    b=subprocess.check_output(['git','show',new+':'+p],cwd=ROOT)
    assert (ROOT/p).read_bytes()==b,p
    hashes[p]=hashlib.sha256(b).hexdigest()
ui={p:hashlib.sha256((ROOT/p).read_bytes()).hexdigest() for p in UI}
receipt={'oldExactDependency':old,'newExactDependency':new,'classification':'INTEGRATION DEPENDENCY ONLY, UNADMITTED R33','files':hashes,'unchangedUI':ui}
(ROOT/'apps/keyboard-ios/audits/r33-ui-integration-r41/dependency-v2-hashes.json').write_text(json.dumps(receipt,indent=2)+'\n')
subprocess.run(['git','add','--',*files],cwd=ROOT,check=True)
subprocess.run(['git','commit','-m','chore(keyboard): refresh unadmitted R33 integration dependency'],cwd=ROOT,check=True)
print('latest_dependency_staged='+subprocess.check_output(['git','rev-parse','HEAD'],cwd=ROOT,text=True).strip(),flush=True)
