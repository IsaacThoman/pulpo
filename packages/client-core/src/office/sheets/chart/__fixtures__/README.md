# Chart fixtures

Small chart parts (`xl/charts/chartN.xml`) for the parser and renderer tests.

- `openpyxl/` — generated by [openpyxl](https://openpyxl.readthedocs.io/) 3.1 with the script below.
  openpyxl writes the default namespace without prefixes and no cached values, so tests render them
  with `fixtureResolver` (the same cell data, in `../test-data.ts`).
- `libreoffice/` — the same workbook opened and re-saved as .xlsx by LibreOffice (headless), which
  adds `c:` prefixes, number/string caches, and explicit text formatting.
- `excel/` — hand-written to match what Excel 2016+ writes (c14 `mc:AlternateContent` style,
  scheme colors with lumMod/lumOff, `c:txPr`, `c:dPt`, `c:dLbls`, `rot="-60000000"`).

To look at the rendered output, run
`CHART_RENDER_OUT=/tmp/charts npx vitest run packages/client-core/src/office/sheets/chart/render-fixtures`
and convert the SVGs, e.g. `qlmanage -t -s 800 -o /tmp/charts-png /tmp/charts/*.svg` on macOS.

## Generator

```python
import sys
from openpyxl import Workbook
from openpyxl.chart import BarChart, LineChart, AreaChart, PieChart, DoughnutChart, ScatterChart, BubbleChart, RadarChart, Reference, Series, BarChart3D, PieChart3D
from openpyxl.chart.label import DataLabelList
from openpyxl.chart.marker import Marker
from openpyxl.chart.series import SeriesLabel
out = sys.argv[1]
wb = Workbook(); ws = wb.active; ws.title = 'Sheet1'
rows = [('Month','Apples','Pears','Plums'),('Jan',40,30,-5),('Feb',55,25,10),('Mar',70,45,22),('Apr',60,50,18),('May',85,48,30),('Jun',95,62,26)]
for r in rows: ws.append(r)
data = Reference(ws, min_col=2, min_row=1, max_col=4, max_row=7)
cats = Reference(ws, min_col=1, min_row=2, max_row=7)
def place(ch, cell, w=15, h=8): ch.width=w; ch.height=h; ws.add_chart(ch, cell)
# 1 clustered column
c = BarChart(); c.type='col'; c.title='Fruit sales'; c.y_axis.title='Units'; c.x_axis.title='Month'
c.add_data(data, titles_from_data=True); c.set_categories(cats); place(c,'F1')
# 2 stacked bar horizontal
c = BarChart(); c.type='bar'; c.grouping='stacked'; c.overlap=100; c.title='Stacked bars'
c.add_data(data, titles_from_data=True); c.set_categories(cats); place(c,'F18')
# 3 percent stacked column
c = BarChart(); c.type='col'; c.grouping='percentStacked'; c.overlap=100; c.title='Percent stacked'
c.add_data(data, titles_from_data=True); c.set_categories(cats); place(c,'F35')
# 4 line with markers
c = LineChart(); c.title='Lines'; c.add_data(data, titles_from_data=True); c.set_categories(cats)
c.series[0].marker = Marker(symbol='circle', size=7); c.series[1].marker = Marker(symbol='triangle', size=7); c.series[1].smooth = True
c.series[2].marker = Marker(symbol='x', size=7)
c.dataLabels = None; place(c,'F52')
# 5 area stacked
c = AreaChart(); c.grouping='stacked'; c.title='Stacked area'
d2 = Reference(ws, min_col=2, min_row=1, max_col=3, max_row=7)
c.add_data(d2, titles_from_data=True); c.set_categories(cats); place(c,'F69')
# 6 pie with labels
c = PieChart(); c.title='Apples by month'
c.add_data(Reference(ws, min_col=2, min_row=1, max_row=7), titles_from_data=True); c.set_categories(cats)
c.dataLabels = DataLabelList(); c.dataLabels.showPercent = True; place(c,'F86')
# 7 doughnut
c = DoughnutChart(); c.title='Doughnut'; c.holeSize=50
c.add_data(Reference(ws, min_col=2, min_row=1, max_row=7), titles_from_data=True); c.set_categories(cats); place(c,'F103')
# 8 scatter
ws2 = wb.create_sheet('Data')
ws2.append(('x','y1','y2','size'))
import math
for i in range(1,21): ws2.append((i*0.5, math.sin(i/3)*10+12, i*1.1, (i%5)+1))
c = ScatterChart(); c.title='Scatter'; c.style=13
x = Reference(ws2, min_col=1, min_row=2, max_row=21)
for col in (2,3):
  s = Series(Reference(ws2, min_col=col, min_row=1, max_row=21), x, title_from_data=True)
  s.marker = Marker('circle'); 
  if col==2: s.graphicalProperties.line.noFill=True
  c.series.append(s)
c.x_axis.title='x'; c.y_axis.title='y'; place(c,'F120')
# 9 bubble
c = BubbleChart(); c.title='Bubbles'
s = Series(values=Reference(ws2, min_col=2, min_row=2, max_row=11), xvalues=Reference(ws2, min_col=1, min_row=2, max_row=11), zvalues=Reference(ws2, min_col=4, min_row=2, max_row=11), title='Bubbles A')
c.series.append(s); place(c,'F137')
# 10 radar
c = RadarChart(); c.type='marker'; c.title='Radar'
c.add_data(d2, titles_from_data=True); c.set_categories(cats); place(c,'F154')
# 11 combo bar + line secondary
b = BarChart(); b.add_data(Reference(ws, min_col=2, min_row=1, max_row=7), titles_from_data=True); b.set_categories(cats); b.title='Combo'
l = LineChart(); l.add_data(Reference(ws, min_col=4, min_row=1, max_row=7), titles_from_data=True)
l.y_axis.axId=200; l.y_axis.title='Plums'; l.y_axis.crosses='max'
b += l; place(b,'F171')
# 12 bar3d
c = BarChart3D(); c.title='3D bars'; c.add_data(data, titles_from_data=True); c.set_categories(cats); place(c,'F188')
# 13 single series, no title (auto title), legend bottom, data labels
c = BarChart(); c.add_data(Reference(ws, min_col=3, min_row=1, max_row=7), titles_from_data=True); c.set_categories(cats)
c.legend.position='b'; c.dataLabels = DataLabelList(); c.dataLabels.showVal=True; place(c,'F205')
wb.save(out)
```

File names map to the generated charts in order: bar-clustered, bar-stacked-horizontal,
bar-percent-stacked, line-markers, area-stacked, pie-labels, doughnut, scatter, bubble, radar,
combo-secondary, bar-3d, single-series-labels.
