import XCTest

@MainActor
final class R34NativeInteractionTests: XCTestCase {
  func testLocalizedPicksAndCancelAcrossThreeLanguageSettings() {
    continueAfterFailure = false
    let app = XCUIApplication()
    app.launchArguments = ["-PKPreview", "-PKResetSettings", "-PKResetLearning"]
    app.launch()
    XCTAssertTrue(app.key("key-q").waitForExistence(timeout: 12))
    for language in ["English", "Español", "English + Español"] {
      app.key("keyboard-language").tap()
      app.buttons[language].firstMatch.tap()
      XCTAssertTrue(app.key("keyboard-language").label.contains(language))
      for (query, expected) in [("vicuna", "🦙"), ("teclado", "⌨️"), ("keyboard teclado", "⌨️")] {
        let original = app.previewText
        app.key("key-emoji").tap()
        XCTAssertTrue(app.key("emoji-search-open").waitForExistence(timeout: 8))
        app.key("emoji-search-open").tap()
        for letter in query { app.key(letter == " " ? "key-space" : "key-\(letter)").tap() }
        XCTAssertEqual(app.key("emoji-search-field").value as? String, query)
        XCTAssertEqual(app.previewText, original)
        let target = app.key("emoji-result-\(expected)")
        XCTAssertTrue(target.waitForExistence(timeout: 4))
        XCTAssertTrue(target.isHittable)
        target.tap()
        XCTAssertEqual(app.previewText, original + expected)
        app.key("emoji-search-back").tap()
        XCTAssertTrue(app.key("emoji-panel").exists)
        app.buttons["emoji-abc"].tap()
        XCTAssertFalse(app.key("emoji-search-field").exists)
      }
      let beforeCancel = app.previewText
      app.key("key-emoji").tap()
      app.key("emoji-search-open").tap()
      for c in "teclado" { app.key("key-\(c)").tap() }
      app.key("key-letters").tap()
      XCTAssertFalse(app.key("emoji-search-field").exists)
      XCTAssertEqual(app.previewText, beforeCancel)
      app.key("key-x").tap()
      XCTAssertEqual(app.previewText, beforeCancel + "x")
      snapshot("r34-\(language)-letters-after-cancel")
    }
  }
}
