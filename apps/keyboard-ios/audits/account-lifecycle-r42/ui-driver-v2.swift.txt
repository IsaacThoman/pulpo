import XCTest

@MainActor
final class AccountLifecycleR42UITests: XCTestCase {
  private let app = XCUIApplication()
  private var server: String!

  override func setUp() async throws {
    continueAfterFailure = false
    guard let value = ProcessInfo.processInfo.environment["R35_NATIVE_SERVER"], value.hasPrefix("http://127.0.0.1:") else {
      throw XCTSkip("Opt-in account UI proof requires loopback R35_NATIVE_SERVER")
    }
    server = value
    app.launchArguments = ["-PKSignOut"]
    app.launch()
  }

  private func control(_ action: String) async throws -> [String: Any] {
    let (data, _) = try await URLSession.shared.data(from: URL(string: server + "/r35-control/" + action)!)
    return try JSONSerialization.jsonObject(with: data) as! [String: Any]
  }
  private func issued() async throws {
    for _ in 0..<100 {
      if let requests = try await control("state")["requests"] as? [[String: Any]], !requests.isEmpty { return }
      try await Task.sleep(for: .milliseconds(20))
    }
    XCTFail("Native login never reached the loopback server")
  }
  private func open() {
    app.open(URL(string: "pulpokeyboard://account")!)
    if !app.textFields["server"].waitForExistence(timeout: 3) {
      try? record("sheet-open-failure")
      XCTFail(app.debugDescription)
    }
  }
  private func replace(_ field: XCUIElement, with text: String, below: Bool = false) {
    for _ in 0..<5 where !field.isHittable { if below { app.swipeUp() } else { app.swipeDown() } }
    field.tap()
    let previous = field.value as? String ?? ""
    if !previous.isEmpty && previous != field.placeholderValue {
      field.coordinate(withNormalizedOffset: CGVector(dx: 0.98, dy: 0.5)).tap()
      field.typeText(String(repeating: XCUIKeyboardKey.delete.rawValue, count: previous.count))
    }
    field.typeText(text)
    if field.elementType != .secureTextField { XCTAssertEqual(field.value as? String, text) }
  }
  private func fill(email: String) {
    replace(app.textFields["server"], with: server)
    replace(app.textFields["email"], with: email, below: true)
    replace(app.secureTextFields["password"], with: "fake-password", below: true)
  }
  private func submit() {
    let button = app.buttons["submit-sign-in"]
    for _ in 0..<6 where !button.isHittable { app.swipeUp() }
    XCTAssertTrue(button.isHittable && button.isEnabled)
    button.tap()
    try? record("after-submit")
  }
  private func record(_ name: String) throws {
    let attachment = XCTAttachment(screenshot: XCUIScreen.main.screenshot())
    attachment.name = "r35-" + name
    attachment.lifetime = .keepAlways
    add(attachment)
    if let output = ProcessInfo.processInfo.environment["R35_NATIVE_EVIDENCE"] {
      try XCUIScreen.main.screenshot().pngRepresentation.write(to: URL(fileURLWithPath: output + "/" + self.name.replacingOccurrences(of: "/", with: "-") + "-" + name + ".png"))
    }
  }
  private func accountNamed(_ name: String) -> XCUIElement {
    app.descendants(matching: .any).matching(NSPredicate(format: "value == %@ OR label CONTAINS %@", name, name)).firstMatch
  }
  private func dismissPasswordPrompt() {
    let notNow = app.buttons["Not Now"]
    if notNow.waitForExistence(timeout: 1) { notNow.tap(); return }
    let systemNotNow = XCUIApplication(bundleIdentifier: "com.apple.springboard").buttons["Not Now"]
    if systemNotNow.exists { systemNotNow.tap() }
  }
  private func assertSignedIn() throws {
    XCTAssertTrue(app.navigationBars["Sign In to Pulpo"].waitForNonExistence(timeout: 5))
    dismissPasswordPrompt()
    for _ in 0..<4 where !accountNamed("R35 Retry").exists { app.swipeUp() }
    if !accountNamed("R35 Retry").waitForExistence(timeout: 3) {
      try record("retry-failure")
      XCTFail(app.debugDescription)
    }
  }
  private func revealAccount() {
    XCTAssertTrue(app.navigationBars["Keyboard"].waitForExistence(timeout: 5))
    XCTAssertFalse(app.navigationBars["Sign In to Pulpo"].exists)
    func visibleAccount() -> Bool {
      let signIn = app.buttons["sign-in"], signOut = app.buttons["Sign Out"]
      return (signIn.exists && signIn.isHittable) || (signOut.exists && signOut.isHittable)
    }
    // Short drags avoid flinging past this early List row. Search both directions
    // because sheet dismissal can preserve the root's previous scroll offset.
    for upwards in [true, false] {
      for _ in 0..<24 where !visibleAccount() {
        let start = app.coordinate(withNormalizedOffset: CGVector(dx: 0.9, dy: upwards ? 0.78 : 0.48))
        let end = app.coordinate(withNormalizedOffset: CGVector(dx: 0.9, dy: upwards ? 0.48 : 0.78))
        start.press(forDuration: 0.05, thenDragTo: end)
      }
    }
    if !visibleAccount() {
      try? record("account-root-not-visible")
      XCTFail(app.debugDescription)
    }
  }
  private func signedOutAfterRelease() async throws {
    dismissPasswordPrompt()
    _ = try await control("release")
    revealAccount()
    // UI wait is bounded by an observable outcome; no claim about recalling remote work.
    XCTAssertFalse(accountNamed("R35 Held").waitForExistence(timeout: 1))
    XCTAssertTrue(app.buttons["sign-in"].isHittable)
    try record("signed-out-after-release")
    app.terminate()
    app.launchArguments = ["-PKAccountR35"]
    app.launch()
    revealAccount()
    try record("signed-out-after-relaunch")
    XCTAssertTrue(app.buttons["sign-in"].isHittable, "Cancelled login must remain signed out after relaunch")
  }

  func testCancelPendingLoginThenRetryAndSettings() async throws {
    _ = try await control("reset")
    open(); fill(email: "held@fake.invalid"); submit()
    try await issued()
    XCTAssertTrue(app.buttons["Cancel"].isHittable)
    try record("pending-cancel")
    app.buttons["Cancel"].tap()
    try await signedOutAfterRelease()
    open(); fill(email: "retry@fake.invalid"); submit()
    try assertSignedIn()
    try record("retry-success")
    let find = app.buttons["find-setting"]
    XCTAssertTrue(find.isHittable)
    find.tap(); app.buttons["Language"].tap()
    XCTAssertTrue(app.buttons["typing-language"].isHittable)
    find.tap(); app.buttons["One-Handed"].tap()
    XCTAssertTrue(app.buttons["one-handed-setting"].isHittable)
    try record("settings-after-retry")
  }

  func testDismissPendingSheetThenRetry() async throws {
    _ = try await control("reset")
    open(); fill(email: "held@fake.invalid"); submit()
    try await issued()
    let title = app.navigationBars["Sign In to Pulpo"]
    title.coordinate(withNormalizedOffset: CGVector(dx: 0.5, dy: 0.5)).press(forDuration: 0.1,
      thenDragTo: app.coordinate(withNormalizedOffset: CGVector(dx: 0.5, dy: 0.9)))
    XCTAssertTrue(app.navigationBars["Sign In to Pulpo"].waitForNonExistence(timeout: 3))
    try await signedOutAfterRelease()
    open(); fill(email: "retry@fake.invalid"); submit()
    try assertSignedIn()
    try record("dismiss-retry-success")
  }

  func testTwoFactorFocusBackAndSuccessfulRetry() async throws {
    _ = try await control("reset")
    open(); fill(email: "twofactor@fake.invalid"); submit()
    let code = app.textFields["code"]
    XCTAssertTrue(code.waitForExistence(timeout: 5))
    app.typeText("654321")
    XCTAssertEqual(code.value as? String, "654321", "Challenge focuses the code field")
    try record("two-factor-focused")
    let back = app.buttons["sign-in-back"]
    for _ in 0..<5 where !back.isHittable { app.swipeUp() }
    XCTAssertTrue(back.isHittable)
    back.tap()
    XCTAssertFalse(code.exists)
    submit()
    XCTAssertTrue(code.waitForExistence(timeout: 5))
    app.typeText("123456")
    XCTAssertEqual(code.value as? String, "123456")
    submit()
    try assertSignedIn()
    try record("two-factor-success")
  }
}
