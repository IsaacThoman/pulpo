import CryptoKit
import Foundation
import Testing
@testable import KeyboardCore

/// Exposed R5 oracles remain unchanged. Every starting model is actually executed.
@Suite(.serialized)
struct PhraseInsertionR26LegacyR5Tests {
  struct Action: Decodable { let kind: String; let text: String }
  struct Task: Decodable {
    let id: String; let mode: String; let initialLeft: String; let initialRight: String
    let actions: [Action]; let expected: String; let control: String
  }
  struct Model: Encodable { let mode: String; let filename: String; let sha256: String }
  struct Result: Encodable {
    let id: String; let language: String; let originalMode: String; let finalMode: String
    let expected: String; let actual: String; let caret: Int; let passed: Bool
    let completed: Bool; let bounded: Bool; let conserved: Bool
    let dictionaries: [Model]; let notes: [String]
    let primitives: [PhraseInsertionR26Tests.Primitive]
  }
  @Test func originalTasksAllStartingModes() throws {
    let env = ProcessInfo.processInfo.environment
    guard let path = env["PULPO_R26_R5_SUITE"], let output = env["PULPO_R26_RESULTS"] else { return }
    let tasks = try JSONDecoder().decode([Task].self, from: Data(contentsOf: URL(fileURLWithPath: path)))
    var results: [Result] = []
    for initialLanguage in KeyboardLanguage.allCases {
      for task in tasks {
        let modelURL = Fixtures.languageData.appendingPathComponent(initialLanguage.rawValue + ".pkdict")
        var stack = try LanguageStack(dictionaryURL: modelURL, personalURL: nil)
        var models: [Model] = []
        func record(_ language: KeyboardLanguage) throws {
          let filename = language.rawValue + ".pkdict"
          let hash = SHA256.hash(data: try Data(contentsOf: Fixtures.languageData.appendingPathComponent(filename))).map { String(format: "%02x", $0) }.joined()
          models.append(Model(mode: language.rawValue, filename: filename, sha256: hash))
        }
        try record(initialLanguage)
        let fixture: [String: Any] = ["id":task.id,"contentLabel":"control","category":"legacy-r5","prefix":task.initialLeft,"selected":"","suffix":task.initialRight,"actions":[],"expected":task.expected,"caret":0]
        let c = try JSONDecoder().decode(PhraseInsertionR26Tests.Case.self, from: JSONSerialization.data(withJSONObject: fixture))
        let host = PhraseInsertionR26Tests.Host(c)
        var settings = KeyboardSettings(); settings.language = initialLanguage
        let engine = KeyboardEngine(document: host, suggester: stack.suggester, glide: stack.glide, settings: settings)
        engine.documentDidChange()
        var completed = true
        var notes: [String] = []
        for action in task.actions where completed {
          switch action.kind {
          case "snippet": engine.insertTranscript(action.text)
          case "emoji": engine.insertText(action.text)
          case "type": engine.type(action.text)
          case "mode":
            let language: KeyboardLanguage
            switch action.text {
            case "en": language = .english
            case "es": language = .spanish
            case "bilingual": language = .bilingual
            default: completed = false; notes.append("Unknown original mode action"); continue
            }
            stack = try LanguageStack(dictionaryURL: Fixtures.languageData.appendingPathComponent(language.rawValue + ".pkdict"), sharing: stack)
            engine.settings.language = language
            engine.useLanguageStack(stack)
            try record(language)
          case "suggestion":
            let choices = engine.suggestions.slots
            if let choice = choices.first(where: { $0.text == action.text }) { engine.select(choice) }
            else {
              completed = false
              notes.append("Requested original visible suggestion unavailable; actual offered: " + choices.map(\.text).joined(separator: " | "))
            }
          case "swipe":
            var generator = SeededGenerator(state: 782_106)
            let choices = engine.insertGlide(SyntheticSwipe.points(for: action.text, noise: 0, generator: &generator))
            notes.append("Actual swipe winner: " + (choices.first?.text ?? "<none>"))
            if choices.isEmpty { completed = false }
          default: completed = false; notes.append("Unsupported original action " + action.kind)
          }
          #expect(engine.suggester === stack.suggester && engine.suggester?.model === stack.model)
        }
        results.append(Result(id: task.id, language: initialLanguage.rawValue, originalMode: task.mode, finalMode: engine.settings.language.rawValue,
                              expected: task.expected, actual: host.text, caret: host.cursor, passed: completed && host.text == task.expected,
                              completed: completed, bounded: host.bounded, conserved: host.conserved,
                              dictionaries: models, notes: notes, primitives: host.primitives))
      }
    }
    #expect(results.count == tasks.count * 3)
    for t in tasks { #expect(Set(results.filter { $0.id == t.id }.map(\.language)) == Set(["en_US", "es", "en_es"])) }
    #expect(results.allSatisfy { $0.bounded })
    let encoder = JSONEncoder(); encoder.outputFormatting = [.sortedKeys, .withoutEscapingSlashes]
    try encoder.encode(results).write(to: URL(fileURLWithPath: output))
    print("R26 R5 exact \(results.filter(\.passed).count)/\(results.count); unavailable \(results.filter { !$0.completed }.count)")
  }
}
