import Foundation
import Testing
@testable import KeyboardCore

@Suite struct LocalizedEmojiSearchTests {
  private func catalog(_ entries: [[String: Any]]) throws -> EmojiCatalog {
    try JSONDecoder().decode(EmojiCatalog.self, from: JSONSerialization.data(withJSONObject:
      ["version": "fixture", "categories": [["id": "fixture", "name": "Fixture", "emoji": entries]]]))
  }

  @Test func spanishAndMixedSearchKeepContextualSuggestionsEnglish() throws {
    let entries: [[String: Any]] = [
      ["e": "⌨️", "n": "keyboard", "v": 1, "k": ["computer"], "s": "teclado"],
      ["e": "🚲", "n": "bicycle", "v": 1, "s": "bicicleta"],
    ]
    let index = EmojiSuggester(catalog: try catalog(entries))
    #expect(index.search("teclado").map(\.value) == ["⌨️"])
    #expect(index.search("computer tecl").map(\.value) == ["⌨️"])
    #expect(index.search("teclado bicycle").isEmpty)
    #expect(index.search(":TECLÁDO:").map(\.value) == ["⌨️"])
    #expect(index.suggestions(for: "teclado").isEmpty)
    #expect(index.suggestions(for: "keyboard").map(\.value) == ["⌨️"])
    #expect(index.search("keyboard").map(\.value) == ["⌨️"])
  }

  @Test func consonantMarksRemainDistinctWithoutChangingEnglishAliases() throws {
    let index = EmojiSuggester(catalog: try catalog([
      ["e": "🧶", "n": "vicuña", "v": 1, "k": ["são", "piña"], "s": "lanera"],
      ["e": "🪡", "n": "needle", "v": 1, "s": "lañera"],
    ]))
    for query in ["lanera", "LANÉRA", "lane\u{301}ra"] { #expect(index.search(query).map(\.value) == ["🧶"]) }
    for query in ["lañera", "LAÑERA", "lan\u{303}era"] { #expect(index.search(query).map(\.value) == ["🪡"]) }
    for query in ["vicuna", "VICUÑA", "vicun\u{303}a", "sao", "SÃO", "pina", "PIN\u{303}A"] {
      #expect(index.search(query, limit: 8).map(\.value).contains("🧶"))
    }
  }

  @Test func overfullSearchKeepsTheSameLocalizedPicksAfterReorderingAndRepeating() throws {
    let glyphs = ["😀", "😃", "😄", "😁", "😆", "😅", "🤣", "😂", "🙂", "🙃", "🫠", "🫫", "😉", "😊", "😇", "🥰", "😍", "🤩", "😘", "😗", "☺️", "😚", "😙", "🥲", "😋", "😛", "😜", "🤪", "😝", "🤑", "🤗", "🤭", "🫢", "🫣", "🤫", "🤔", "🫡", "🤐", "🤨", "😐", "😑", "😶", "🫥", "😶‍🌫️", "😏", "😒", "🙄", "😬", "😮‍💨"]
    let index = EmojiSuggester(catalog: try catalog(glyphs.enumerated().map { i, value in
      ["e": value, "n": "fixture \(i)", "v": 1,
       "s": i == 47 ? "nube violeta" : (i == 48 ? "violeta nube" : "nube violeta extra\(i)")]
    }))
    for cap in [1, 48, 60] {
      let original = index.search("nube violeta", limit: cap).map(\.value)
      #expect(original.count == min(cap, glyphs.count))
      for query in ["violeta nube", "nube violeta violeta", ":NUBE: violeta", "violeta nube nube"] {
        #expect(index.search(query, limit: cap).map(\.value) == original)
      }
    }
  }

  @Test func legacyCatalogsLimitsAndShortPrefixesStayCompatible() throws {
    let entries: [[String: Any]] = [
      ["e": "👑", "n": "queen", "v": 1, "s": "quimera"],
      ["e": "⚡️", "n": "zap", "v": 1, "s": "queso"],
    ]
    let current = EmojiSuggester(catalog: try catalog(entries))
    let old = EmojiSuggester(catalog: try catalog(entries.map { $0.filter { $0.key != "s" } }))
    for query in ["q", "z", ":q:", "q q", "", "::", "-", String(repeating: "q", count: 81)] {
      #expect(current.search(query).map(\.value) == old.search(query).map(\.value))
    }
    for limit in [-1, 0] { #expect(current.search("quimera", limit: limit).isEmpty) }
    #expect(current.search("queso").map(\.value) == ["⚡️"])
  }
}
