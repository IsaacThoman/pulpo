"""Exposed, independent protocol smoke; no semantic utility claims.

smoke-runner.py BASELINE_RUNNER CATALOG OUTPUTDIR [--candidate-runner RUNNER]
  [--suite ARBITRARY_EXPOSED_SUITE] [--summary LOW_VOLUME_JSON]

Generated exposed inputs and every execution artifact live in OUTPUTDIR, which
must be outside the repository. --suite also executes any strictly valid exposed
suite in full; its oracle failures are reported, never silently dropped.
Candidate mode checks EVERY required row and exact localized fixture suffix;
all failures are aggregated with raw outputs retained. Candidate oracle failures
(including arbitrary --suite oracles) produce a nonzero exit. Baseline-only mode
reports future localized failures as expected data, not baseline failure. It
it allows the inherited English portion to depend on query order/repetition.
"""

def write(path, value):
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")


def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def add_r40_controls(add, groups):
    """Independent exposed fixtures; constants are oracles, never candidate output."""
    import unicodedata

    def entries(prefix, count, name):
        return [{"emoji": f"r40-{prefix}-{i:03d}",
                 "name": name + f" item{chr(97+i//26)}{chr(97+i%26)}", "keywords": []}
                for i in range(count)]

    # Five unrelated admitted non-ASCII letters: their projections occur ONLY
    # inside English words, with no indexed-term prefix witness. Reservation=0.
    for label, letter, projection in (("cedilla", "ç", "c"), ("acute-g", "ǵ", "g"),
                                      ("stroke-l", "ł", "ł"), ("greek", "ζ", "ζ"),
                                      ("cyrillic", "й", "и")):
        english = entries("internal-" + label, 73, "vex" + projection + "ara talumber")
        local_entries = entries("local-" + label, 86, "velvet lantern")
        aliases = [{"emoji": e["emoji"], "name": letter + "avela umber"} for e in local_entries]
        for limit in (1, 48, 60):
            for kind, queries in (("single", [letter, letter.upper(), unicodedata.normalize("NFD", letter),
                                             f"({letter})", f"{letter} {letter}"]),
                                  ("pair", [f"{letter} umber", f"umber {letter}",
                                            f"umber {letter} umber", f"{letter.upper()}9UMBER"])):
                ids = []
                for index, query in enumerate(queries):
                    case_id = f"r40-internal-{label}-{kind}-{limit}-{index}"; ids.append(case_id)
                    wanted = [e["emoji"] for e in local_entries[:limit]]
                    add(case_id, query, action="fixture", values=[], local=wanted,
                        entries=english + local_entries, spanishAliases=aliases, limit=limit,
                        preferred={english[-1]["emoji"]: 2, local_entries[-1]["emoji"]: 2},
                        family="exposed-capacity")
                groups.append({"ids": ids, "kind": "localized-suffix-only"})

    # Marks on vowels fold; the same three mark categories on consonants remain.
    for label, mark in (("Mn", "\u0301"), ("Mc", "\u093e"), ("Me", "\u20dd")):
        fixture = entries("marks-" + label, 3, "velvet lantern")
        aliases = [{"emoji": fixture[0]["emoji"], "name": "z" + mark + "avela r" + mark + "omulo"},
                   {"emoji": fixture[1]["emoji"], "name": "zavela romulo"},
                   {"emoji": fixture[2]["emoji"], "name": "za" + mark + "vela ro" + mark + "mulo"}]
        for kind, queries, wanted in (
            ("consonant", ["z" + mark + "avela r" + mark + "omulo",
                            "R" + mark + "OMULO Z" + mark + "AVELA",
                            "r" + mark + "omulo z" + mark + "avela z" + mark + "avela"], [fixture[0]["emoji"]]),
            ("vowel", ["zavela romulo", "za" + mark + "vela ro" + mark + "mulo",
                       "ROMULO ZAVELA ROMULO"], [fixture[1]["emoji"], fixture[2]["emoji"]])):
            ids = []
            for index, query in enumerate(queries):
                case_id = f"r40-marks-{label}-{kind}-{index}"; ids.append(case_id)
                add(case_id, query, action="fixture", values=[], local=wanted,
                    entries=fixture, spanishAliases=aliases,
                    forbidden=[e["emoji"] for e in fixture if e["emoji"] not in wanted],
                    family="exposed-orthography")
            groups.append({"ids": ids, "kind": "localized-suffix-only"})

    # Ignored tokens still witness indexed phrase prefixes, even when R is empty.
    local_entries = entries("stop-local", 86, "velvet lantern")
    for kind, english_name, alias, queries, english_indexes in (
        ("only", "the rivet", "the", ["the", "THE", "the the", "(the)", "the/the"], {0, 1, 3}),
        ("phrase", "the arrivet", "the arrivel", ["the arri", "arri the", "arri the arri", "the'arri", "the/arri"], {0, 2, 3})):
        reserved = entries("stop-" + kind, 7, english_name)
        aliases = [{"emoji": e["emoji"], "name": alias} for e in local_entries]
        for limit in (1, 48, 60):
            ids = []
            for index, query in enumerate(queries):
                case_id = f"r40-stopword-{kind}-{limit}-{index}"; ids.append(case_id)
                legacy = [e["emoji"] for e in reserved[:limit]] if index in english_indexes else []
                wanted = [e["emoji"] for e in local_entries[:max(0, limit-7)]]
                add(case_id, query, action="fixture", values=legacy, local=wanted,
                    acceptable=wanted or legacy, expectEmpty=not (wanted or legacy),
                    entries=reserved + local_entries, spanishAliases=aliases, limit=limit,
                    family="exposed-capacity")
            groups.append({"ids": ids, "kind": "localized-suffix-only"})

    # More English envelope members than every cap: localized tranche must be 0.
    reserved = entries("saturated", 86, "anchor beacon")
    local_entries = entries("saturated-local", 86, "velvet lantern")
    aliases = [{"emoji": e["emoji"], "name": "anchor beacon"} for e in local_entries]
    for limit in (1, 48, 60):
        ids = []
        for index, query in enumerate(("anchor beacon", "beacon anchor", "anchor beacon anchor", "anchor'beacon")):
            case_id = f"r40-saturated-{limit}-{index}"; ids.append(case_id)
            add(case_id, query, action="fixture", local=[], acceptable=[reserved[-1]["emoji"]] if index in (0, 3) else [reserved[0]["emoji"]],
                entries=reserved + local_entries, spanishAliases=aliases, limit=limit,
                preferred={reserved[-1]["emoji"]: 2, local_entries[-1]["emoji"]: 2}, family="exposed-capacity")
        groups.append({"ids": ids, "kind": "localized-suffix-only"})

    # Full-name token SET equality, completion count, distinct words, catalog tie.
    ranked = entries("rank", 5, "velvet lantern")
    aliases = [{"emoji": e["emoji"], "name": name} for e, name in zip(ranked,
        ["mavela romulo extra", "mavelar romulo", "mavela romulo romulo", "romulo mavela", "mavelar romular"])]
    wanted = [ranked[i]["emoji"] for i in (2, 3, 0, 1, 4)]
    ids = []
    for index, query in enumerate(("mavela romulo", "romulo mavela", "ROMULO MAVELA ROMULO")):
        case_id = f"r40-ranking-{index}"; ids.append(case_id)
        add(case_id, query, action="fixture", values=[], local=wanted, entries=ranked,
            spanishAliases=aliases, preferred={ranked[-1]["emoji"]: 2}, family="exposed-capacity")
    groups.append({"ids": ids, "kind": "localized-suffix-only"})

def coverage(suite):
    schema = load(SCHEMA)
    result = {}
    for action in schema["actions"]:
        rows = [r for r in suite["cases"] if r["action"] == action]
        seen = set().union(*(set(r) for r in rows))
        required = set(schema["caseRequired"] + schema["optionalCommon"] + schema[action + "Parameters"])
        require(rows and required <= seen, f"unexercised {action} fields: {sorted(required-seen)}")
        result[action] = {"executedRows": len(rows), "requiredFields": schema["caseRequired"],
                          "commonOptionalFields": schema["optionalCommon"], "parameters": schema[action + "Parameters"],
                          "allCovered": True}
    entry_seen = set().union(*(set(e) for r in suite["cases"] for e in r.get("entries", [])))
    alias_seen = set().union(*(set(a) for r in suite["cases"] for a in r.get("spanishAliases", [])))
    require(set(schema["fixtureEntryRequired"] + schema["fixtureEntryOptional"]) <= entry_seen, "fixture entry fields unexercised")
    require(set(schema["fixtureAliasRequired"]) <= alias_seen, "alias fields unexercised")
    return {"actions": result, "topLevel": sorted(suite), "fixtureEntry": sorted(entry_seen), "fixtureAlias": sorted(alias_seen), "allCovered": True}


def negatives(suite):
    small = {"version": 2, "families": ["f"], "cases": [{"id": "a", "family": "f", "action": "search", "query": "heart", "acceptable": ["❤️"], "maxRank": 1, "role": "control"}]}
    fixture = copy.deepcopy(next(r for r in suite["cases"] if r["id"] == "fixture-english"))
    fixture["family"] = "f"
    fixture["id"] = "a"
    wrapper = copy.deepcopy(next(r for r in suite["cases"] if r["id"] == "wrapper-toned"))
    wrapper["family"] = "f"; wrapper["id"] = "a"
    tests = []

    def mutation(name, fn, kind="search"):
        x = copy.deepcopy(small)
        if kind != "search":
            x["cases"] = [copy.deepcopy(fixture if kind == "fixture" else wrapper)]
        fn(x)
        tests.append((name, x))

    for field in ("version", "families", "cases"):
        mutation("missing-top-" + field, lambda x, field=field: x.pop(field))
    mutation("unknown-top", lambda x: x.update(extra=True))
    for field in load(SCHEMA)["caseRequired"]:
        mutation("missing-case-" + field, lambda x, field=field: x["cases"][0].pop(field))
    bad_fields = {"id": ["", 1, None], "family": ["", "undeclared", [], True], "action": ["invalid", 4],
                  "query": [1, None, False], "acceptable": [[], "❤️", [""], [False]],
                  "maxRank": [0, 61, True, 1.0, "1"], "role": ["invalid", False],
                  "stratum": ["invalid", True], "forbidden": ["❤️", [""], [3]],
                  "expectEmpty": [0, "true", None], "equivalentTo": ["a", "missing", "", True],
                  "limit": [-1_000_001, 1_000_001, True, False, 1.0, -1.0, "1", None],
                  "preferred": [[], {"": 1}, {"❤️": True}, {"❤️": 1.0}, {"❤️": -1_000_001}, {"❤️": 1_000_001}]}
    for field, values in bad_fields.items():
        for i, value in enumerate(values):
            mutation(f"bad-{field}-{i}", lambda x, field=field, value=value: x["cases"][0].update({field: value}))
    for i, value in enumerate(([], [""], ["f", "f"], "f", [False])):
        mutation(f"families-{i}", lambda x, value=value: x.update(families=value))
    for i, value in enumerate((True, 2.0, 3, "2")):
        mutation(f"version-{i}", lambda x, value=value: x.update(version=value))

def run(runner, catalog, suite_path, output_path, success=True):
    # Stale output can never masquerade as execution of this input.
    output_path.unlink(missing_ok=True)
    result = subprocess.run([str(runner), str(catalog), str(suite_path), str(output_path)], text=True, capture_output=True, timeout=90)
    require(not result.stdout, "runner unexpectedly printed payload/stdout")
    if success:
        require(result.returncode == 0, f"runner execution failed: {result.stderr.strip()}")
        require(output_path.is_file(), "runner succeeded without output")
    else:
        require(result.returncode != 0 and not output_path.exists(), "invalid/missing resource unexpectedly executed")
    return result


def comparator_selfcheck(directory):
    # Output failures, fail-to-fail raw changes, explicit rank, negatives,
    # references and unique family utility are checked without any emoji semantics.
    base = {"version": 2, "families": ["explicit-family", "controls"], "cases": [
        {"id": "useful", "family": "explicit-family", "action": "search", "query": "x", "acceptable": ["A", "A", "B"], "maxRank": 1, "role": "useful", "stratum": "en", "forbidden": ["Z"]},
        {"id": "fail-fail", "family": "controls", "action": "search", "query": "y", "acceptable": ["Q"], "maxRank": 1, "role": "control"},
        {"id": "empty", "family": "controls", "action": "search", "query": "", "acceptable": [], "maxRank": 1, "role": "control", "expectEmpty": True},
        {"id": "equiv", "family": "controls", "action": "search", "query": "z", "acceptable": ["A"], "maxRank": 1, "role": "control", "equivalentTo": "useful"}]}
    validate(base)
    def row(case_id, values, query):
        return {"id": case_id, "values": values, "baseValues": values, "query": query}
    baseline = [row("useful", ["A"], "x"), row("fail-fail", ["X"], "y"), row("empty", [], ""), row("equiv", ["A"], "z")]
    candidate = [row("useful", ["B", "A", "Z"], "x"), row("fail-fail", ["Y"], "y"), row("empty", ["X"], ""), row("equiv", ["A"], "z")]
    b, c = directory / "comparator-selfcheck-baseline.json", directory / "comparator-selfcheck-candidate.json"
    write(directory / "comparator-selfcheck-suite.json", base); write(b, baseline); write(c, candidate)
    result = compare(base, b, c)
    require(result["rows"][0]["candidate"]["targetRanks"] == [{"target": "A", "rank": 2}, {"target": "B", "rank": 1}], "explicit rank/unique targets")
    require(result["rows"][0]["candidate"]["forbiddenHits"] == ["Z"] and not result["rows"][0]["candidate"]["pass"], "forbidden check")
    require(not result["rows"][2]["candidate"]["pass"] and not result["rows"][3]["candidate"]["pass"], "empty/equivalence check")
    require(any(r["id"] == "fail-fail" and r["transition"] == "fail-to-fail" for r in result["changedRawArrays"]), "fail-to-fail raw changes missing")
    require(result["summary"]["baseline"]["useful"]["uniqueAcceptedAlternatives"] == ["A"], "unique alternatives")
    require(result["summary"]["candidate"]["useful"]["uniqueAcceptedAlternatives"] == [], "failing forbidden row counted as utility")
    rank_only = copy.deepcopy(candidate); rank_only[0] = row("useful", ["X", "A"], "x")
    write(c, rank_only)
    require(not compare(base, b, c)["rows"][0]["candidate"]["pass"], "maxRank not enforced")
    single_alternative = copy.deepcopy(candidate); single_alternative[0] = row("useful", ["B"], "x")
    write(c, single_alternative)
    alternative_report = compare(base, b, c)
    require(alternative_report["summary"]["candidate"]["useful"]["uniqueAcceptedAlternatives"] == ["B"], "unique passing alternative count")
    wrapper_suite = copy.deepcopy(base)
    for case in wrapper_suite["cases"]:
        case["action"] = "wrapper"
    write(directory / "comparator-selfcheck-wrapper-suite.json", wrapper_suite)
    toned = copy.deepcopy(baseline)
    toned[0]["baseValues"] = ["base-one"]; toned[3]["baseValues"] = ["base-two"]
    write(c, toned)
    eq = compare(wrapper_suite, b, c)["rows"][3]["candidate"]["equivalence"]
    require(eq["equalValues"] is True and eq["equalBaseValues"] is False, "baseValues-only equivalence mismatch missed")
    write(c, candidate[:-1])
    missing = compare(base, b, c)
    require(not missing["summary"]["candidate"]["executionValid"] and len(missing["rows"]) == 4 and not missing["rows"][-1]["candidate"]["available"], "missing row silently dropped")
    write(c, candidate + [candidate[0]])
    require(not compare(base, b, c)["summary"]["candidate"]["executionValid"], "duplicate output id accepted")
    write(c, list(reversed(candidate)))
    require(not compare(base, b, c)["summary"]["candidate"]["executionValid"], "reordered output accepted")
    malformed = copy.deepcopy(candidate); malformed[0]["extra"] = True
    write(c, malformed)
    require(not compare(base, b, c)["summary"]["candidate"]["executionValid"], "unknown output field accepted")
    malformed = copy.deepcopy(candidate); malformed[0]["values"] = ["\ud800"]
    c.write_text(json.dumps(malformed), encoding="utf-8")
    require(not compare(base, b, c)["summary"]["candidate"]["executionValid"], "surrogate output accepted")
    malformed = copy.deepcopy(candidate); malformed[0]["query"] = "wrong"
    write(c, malformed)
    require(not compare(base, b, c)["summary"]["candidate"]["executionValid"], "direct query echo mismatch accepted")
    canonical_suite = copy.deepcopy(base)
    canonical_suite["cases"][0]["id"] = "é"
    canonical_suite["cases"][-1]["equivalentTo"] = "e\u0301"
    canonical_rows = copy.deepcopy(baseline); canonical_rows[0]["id"] = "é"
    canonical_path = directory / "comparator-canonical-output.json"
    write(directory / "comparator-canonical-suite.json", canonical_suite)
    write(canonical_path, canonical_rows)
    validate(canonical_suite)
    canonical_result = compare(canonical_suite, canonical_path, canonical_path)
    require(canonical_result["rows"][-1]["candidate"]["pass"], "canonical equivalence reference unresolved")
    c.write_text("not json", encoding="utf-8")
    require(compare(base, b, c)["summary"]["candidate"]["useful"]["unavailable"] == 1, "unreadable output denominator")
    c.unlink()
    require(compare(base, b, c)["summary"]["candidate"]["controls"]["unavailable"] == 3, "missing output denominator")
    return {"explicitTargetRanks": True, "canonicalIdentifierReference": True, "surrogateOutputRejected": True, "queryEchoChecked": True, "uniqueAlternatives": True, "familyUtilitySeparateFromControls": True,
            "forbiddenEmptyEquivalence": True, "baseValuesOnlyEquivalence": True, "maxRankBoundary": True, "failFailRawArrays": True, "missingMalformedDuplicateOrderUnknownOutput": True}


def candidate_checks(suite, baseline, comparison, localized, groups):
    """Inspect EVERY required row, retain raw failures, then make one decision."""
    rows, suffixes = [], {}
    candidate = {r["id"]: r["candidate"] for r in comparison["rows"]}
    for case in suite["cases"]:
        case_id = case["id"]; b, c = baseline[case_id], candidate[case_id]
        errors = list(c["errors"])
        checks = ["schema-oracles", "legacy-values-prefix", "legacy-baseValues-prefix"]
        if c["available"]:
            for field in ("values", "baseValues"):
                if c[field][:len(b[field])] != b[field]:
                    errors.append("legacy ordered prefix lost: " + field)
            suffixes[case_id] = c["values"][len(b["values"]):]
            if case_id in localized:
                checks.append("exact-localized-suffix")
                if suffixes[case_id] != localized[case_id]:
                    errors.append("independent localized fixture oracle failed")
                if c["baseValues"][len(b["baseValues"]):] != localized[case_id]:
                    errors.append("independent localized baseValues suffix oracle failed")
            else:
                checks.append("unchanged-legacy-array")
                if c["values"] != b["values"] or c["baseValues"] != b["baseValues"]:
                    errors.append("English-only/exact legacy control changed")
        rows.append({"id": case_id, "required": True, "checked": True,
                     "available": c["available"], "checks": checks, "pass": not errors,
                     "errors": errors, "baseline": b, "candidate": c,
                     "expectedLocalizedSuffix": localized.get(case_id),
                     "observedLocalizedSuffix": suffixes.get(case_id)})
    invariants = []
    for group in groups:
        ids = group["ids"]
        complete = all(i in suffixes for i in ids)
        equal = complete and all(suffixes[i] == suffixes[ids[0]] for i in ids)
        invariants.append({"ids": ids, "checked": True, "available": complete, "pass": equal,
                           "suffixes": {i: suffixes.get(i) for i in ids}})
    return {"passed": comparison["summary"]["candidate"]["allRowsPass"] and
                       all(r["pass"] for r in rows) and all(g["pass"] for g in invariants),
            "executedRows": sum(r["available"] for r in rows), "requiredRows": len(rows),
            "checkedRows": len(rows), "localizedFixtureAssertions": len(localized),
            "localizedInvariantGroups": len(groups), "failedRows": [r["id"] for r in rows if not r["pass"]],
            "failedInvariantGroups": [g["ids"] for g in invariants if not g["pass"]],
            "rows": rows, "invariants": invariants}


def required_assertion_selfcheck(directory, suite, baseline_path, localized, groups):
    """Evaluator-only replay/mutation, NOT an implementation or admission run."""
    baseline_rows = load(baseline_path)
    baseline = {r["id"]: r for r in baseline_rows}
    replay = copy.deepcopy(baseline_rows)
    for row in replay:
        row["values"] += localized.get(row["id"], [])
        row["baseValues"] += localized.get(row["id"], [])
    replay_path = directory / "evaluator-only-expected-replay.json"
    write(replay_path, replay)
    checks = candidate_checks(suite, baseline, compare(suite, baseline_path, replay_path), localized, groups)
    require(checks["passed"], "candidate evaluator rejects its declared combined oracles: " + repr(checks["failedRows"]))
    mutated = copy.deepcopy(replay)
    for row in mutated:
        row["values"] = ["r40-evaluator-only-deliberate-defect"]
        row["baseValues"] = row["values"][:]
    mutation_path = directory / "evaluator-only-all-rows-mutated.json"
    write(mutation_path, mutated)
    failures = candidate_checks(suite, baseline, compare(suite, baseline_path, mutation_path), localized, groups)
    require(len(failures["failedRows"]) == len(suite["cases"]), "candidate evaluator silently omits required rows")
    write(directory / "evaluator-only-mutated-checks.json", failures)
    return {"classification": "evaluator-only synthetic replay/mutation; no candidate implementation or admission claim",
            "declaredCombinedOraclesAccepted": checks["passed"], "requiredRows": len(suite["cases"]),
            "deliberatelyCorruptedRowsRejected": len(failures["failedRows"]),
            "expectedReplay": str(replay_path), "mutatedOutput": str(mutation_path),
            "mutatedChecks": str(directory / "evaluator-only-mutated-checks.json")}

                other = directory / "arbitrary-candidate-output.json"
                try:
                    run(args.candidate_runner.resolve(), args.catalog, args.suite.resolve(), other)
                except (OSError, ValueError, subprocess.SubprocessError) as error:
                    arbitrary_error = str(error)
            report = compare(extra_suite, extra_output, other)
            write(directory / "arbitrary-oracle-report.json", report)

            prefix_errors = []
            if args.candidate_runner:
                for row in report["rows"]:
                    b, c = row["baseline"], row["candidate"]
                    if b["available"] and c["available"]:
                        for key in ("values", "baseValues"):
                            if c[key][:len(b[key])] != b[key]:
                                prefix_errors.append({"id": row["id"], "field": key})
            result["arbitraryExposedSuite"] = {"path": str(args.suite.resolve()), "sha256": digest(args.suite), "rows": len(extra_suite["cases"]),
                                               "actions": sorted({row["action"] for row in extra_suite["cases"]}),
                                               "oracleReport": str(directory / "arbitrary-oracle-report.json"), "baselineOutput": str(extra_output),
                                               "candidateOutput": str(other) if args.candidate_runner else None, "summary": report["summary"],
                                               "legacyPrefixFailures": prefix_errors, "requiredRowsChecked": len(report["rows"]),
                                               "runnerExecutionError": arbitrary_error}
        result["protocolSmokePassed"] = result.get("candidate", {}).get("passed", True)
        if args.suite:
            result["protocolSmokePassed"] = result["protocolSmokePassed"] and all(result["arbitraryExposedSuite"]["summary"][mode]["executionValid"] for mode in ("baseline", "candidate"))
        if args.candidate_runner and args.suite:
            result["protocolSmokePassed"] = result["protocolSmokePassed"] and (result["arbitraryExposedSuite"]["summary"]["candidate"]["allRowsPass"] and not result["arbitraryExposedSuite"]["legacyPrefixFailures"] and result["arbitraryExposedSuite"]["runnerExecutionError"] is None)
        write(summary_path, result)
        if not result["protocolSmokePassed"]:
            parser.exit(1, f"smoke failed: candidate required assertions failed; all raw rows retained in {directory}; summary {summary_path}\n")
    except (OSError, ValueError, TypeError, subprocess.SubprocessError) as error:
        result["protocolSmokePassed"] = False
        result["executionError"] = str(error)
        if summary_path is not None:
            write(summary_path, result)
        parser.exit(1, f"smoke failed: {error}\n")
    print(json.dumps({"passed": True, "executedBaselineRows": result["executedBaselineRows"], "rejectedInvalidInputs": len(rejected),
                      "suite": str(suite_path), "summary": str(summary_path)}))

if __name__ == "__main__":
    main()
