/// Frozen synthetic paired driver. A missing native route is retained, never simulated.
@Suite(.serialized)
struct PhraseInsertionR26Tests {
  struct Action: Decodable {
    let kind: String
    let text: String?
    let offset: Int?
    let selection: Int?
    let callback: Bool?
    let newIdentity: Bool?
    let expected: String?
    let caret: Int?
    let phantom: Bool?
  }
  struct Case: Decodable {
    let id: String
    let contentLabel: String
    let category: String
    let prefix: String
    let selected: String
    let suffix: String
    let actions: [Action]
    let expected: String
    let caret: Int
    let context: String?
    let field: String?
    let capitalization: String?
    let smartPunctuation: Bool?
    let smartQuotes: Bool?
    let route: String?
    let oracle: String?
    let rationale: String?
  }
  struct SuiteFile: Decodable { let schema: Int; let cases: [Case] }
  struct Primitive: Codable {
    let kind: String
    let value: String
    let before: String
    let after: String
    let caretBefore: Int
    let caretAfter: Int
    let selectionBefore: Int
    let selectionAfter: Int
    let removed: String
    let bounded: Bool
    let conserved: Bool
  }
  struct Snapshot: Encodable {
    let action: String
    let text: String
    let caret: Int
    let utf16Caret: Int
    let selection: Int
    let logicalBefore: String
    let phantom: Bool
    let lastEdit: String
    let deleteUndoesLastEdit: Bool
    let smartSpacing: Bool
    let autoSpaces: Bool
    let checkpointsPass: Bool
  }
  struct Result: Encodable {
    let id: String
    let contentLabel: String
    let category: String
    let language: String
    let dictionary: String
    let dictionarySHA256: String
    let modelMode: String
    let settingsLanguage: String
    let modelIdentityVerified: Bool
    let expected: String
    let actual: String
    let expectedCaret: Int
    let actualCaret: Int
    let utf16Caret: Int
    let expectedSelection: Int
    let actualSelection: Int
    let exact: Bool
    let oracleDefined: Bool
    let completed: Bool
    let conserved: Bool
    let bounded: Bool
    let nativeDelivery: String
    let nativeUndo: String
    let smartPunctuation: Bool
    let smartQuotes: Bool?
    let field: String
    let context: String
    let primitives: [Primitive]
    let snapshots: [Snapshot]
    let notes: [String]
  }

  final class Host: TextDocument {
    var documentIdentifier: UUID? = UUID()
    var text: String
    var cursor: Int
    var selectionLength: Int
    var labels: [Bool] // true = protected original host character
    let context: String
    let initialBefore: String
    let initialAfter: String
    let initialSelection: String?
    var pending: [(String, String)] = []
    var primitives: [Primitive] = []
    var callback: (() -> Void)?
    var afterApply: (() -> Void)?
    var conserved = true
    var bounded = true
    init(_ c: Case) {
      text = c.prefix + c.selected + c.suffix
      cursor = c.prefix.count
      selectionLength = c.selected.count
      labels = Array(repeating: true, count: c.prefix.count) + Array(repeating: false, count: c.selected.count) + Array(repeating: true, count: c.suffix.count)
      context = c.context ?? "full"
      initialBefore = c.prefix
      initialAfter = c.suffix
      initialSelection = c.selected.isEmpty ? nil : c.selected
    }
    var chars: [Character] { Array(text) }
    var documentContextBeforeInput: String? {
      if ["hidden-before", "hidden-both"].contains(context) { return nil }
      return context == "own-lag" ? initialBefore : String(chars.prefix(cursor))
    }
    var documentContextAfterInput: String? {
      if ["hidden-after", "hidden-both"].contains(context) { return nil }
      return context == "own-lag" ? initialAfter : String(chars.dropFirst(cursor + selectionLength))
    }
    var selectedText: String? {
      if context == "own-lag" { return initialSelection }
      return selectionLength == 0 ? nil : String(chars.dropFirst(cursor).prefix(selectionLength))
    }
    var hasText: Bool { !text.isEmpty }
    func insertText(_ text: String) { request("insert", text) }
    func deleteBackward() { request("delete", "") }
    func request(_ kind: String, _ value: String) {
      if context == "queued" { pending.append((kind, value)) }
      else { apply(kind, value) }
    }
    func apply(_ kind: String, _ value: String) {
      callback?()
      let valid = cursor >= 0 && selectionLength >= 0 && cursor + selectionLength <= chars.count
      bounded = bounded && valid
      guard valid else { return }
      let original = text
      let oldCursor = cursor
      let oldSelection = selectionLength
      var removed = String(chars.dropFirst(cursor).prefix(selectionLength))
      var safe = true
      if kind == "insert" {
        safe = !labels[cursor..<(cursor + selectionLength)].contains(true)
        text = String(chars.prefix(cursor)) + value + String(chars.dropFirst(cursor + selectionLength))
        labels.replaceSubrange(cursor..<(cursor + selectionLength), with: Array(repeating: false, count: value.count))
        cursor += value.count
      } else if selectionLength > 0 {
        safe = !labels[cursor..<(cursor + selectionLength)].contains(true)
        text = String(chars.prefix(cursor)) + String(chars.dropFirst(cursor + selectionLength))
        labels.removeSubrange(cursor..<(cursor + selectionLength))
      } else if cursor > 0 {
        safe = !labels[cursor - 1]
        removed = String(chars[cursor - 1])
        text = String(chars.prefix(cursor - 1)) + String(chars.dropFirst(cursor))
        labels.remove(at: cursor - 1)
        cursor -= 1
      }
      selectionLength = 0
      conserved = conserved && safe
      // A scalar inserted at a grapheme boundary can merge Characters. This
      // Character-indexed host cannot model an interior UIKit UTF-16 caret.
      // Fail bounds explicitly rather than reporting fabricated caret evidence.
      let postValid = cursor >= 0 && cursor <= text.count && labels.count == text.count
      bounded = bounded && postValid
      primitives.append(Primitive(kind: kind, value: value, before: original, after: text, caretBefore: oldCursor, caretAfter: cursor, selectionBefore: oldSelection, selectionAfter: 0, removed: removed, bounded: valid && postValid, conserved: safe))
      afterApply?()
      callback?()
    }
    func flush() {
      let commands = pending
      pending.removeAll()
      for (kind, value) in commands { apply(kind, value) }
    }
    func move(_ offset: Int, selection: Int) {
      flush()
      bounded = bounded && offset >= 0 && selection >= 0 && offset + selection <= chars.count
      guard bounded else { return }
      let oldCursor = cursor
      let oldSelection = selectionLength
      cursor = offset
      selectionLength = selection
      for i in cursor..<(cursor + selection) { labels[i] = false }
      primitives.append(Primitive(kind: "hostMove", value: "", before: text, after: text, caretBefore: oldCursor, caretAfter: cursor, selectionBefore: oldSelection, selectionAfter: selection, removed: "", bounded: true, conserved: true))
    }
    func replace(_ value: String, offset: Int, selection: Int) {
      flush()
      let original = text
      let oldCursor = cursor
      let oldSelection = selectionLength
      bounded = bounded && offset >= 0 && selection >= 0 && offset + selection <= value.count
      guard bounded else { return }
      text = value
      labels = Array(repeating: true, count: value.count)
      cursor = offset
      selectionLength = selection
      for i in cursor..<(cursor + selection) { labels[i] = false }
      primitives.append(Primitive(kind: "hostReplace", value: value, before: original, after: text, caretBefore: oldCursor, caretAfter: cursor, selectionBefore: oldSelection, selectionAfter: selection, removed: original, bounded: true, conserved: true))
    }
  }

  @Test func frozenDocuments() throws {
    let env = ProcessInfo.processInfo.environment
    guard let suite = env["PULPO_R26_SUITE"], let output = env["PULPO_R26_RESULTS"] else { return }
    let file = try JSONDecoder().decode(SuiteFile.self, from: Data(contentsOf: URL(fileURLWithPath: suite)))
    #expect(file.schema == 1)
    #expect(Set(file.cases.map(\.id)).count == file.cases.count)
    var results: [Result] = []
    for c in file.cases {
      #expect(c.caret <= c.expected.count)
      for a in c.actions {
        if ["transcript", "spaced", "raw", "keys", "hostReplace"].contains(a.kind) { _ = try #require(a.text) }
        if ["hostMove", "hostReplace"].contains(a.kind) { _ = try #require(a.offset) }
        if let expected = a.expected, let caret = a.caret { #expect(caret <= expected.count) }
        #expect(["transcript", "spaced", "raw", "keys", "backspace", "space", "return", "hostMove", "hostReplace", "private", "nativeUndo"].contains(a.kind))
      }
    }
    for language in KeyboardLanguage.allCases {
      let dictionary = language.rawValue + ".pkdict"
      let url = Fixtures.languageData.appendingPathComponent(dictionary)
      let hash = SHA256.hash(data: try Data(contentsOf: url)).map { String(format: "%02x", $0) }.joined()
      for c in file.cases {
        let host = Host(c)
        let stack = try LanguageStack(dictionaryURL: url, personalURL: nil)
        var settings = KeyboardSettings()
        settings.language = language
        if let smart = c.smartPunctuation { settings.smartPunctuation = smart }
        let engine = KeyboardEngine(document: host, suggester: stack.suggester, glide: stack.glide, settings: settings)
        var traits = InputTraits()
        traits.capitalization = c.capitalization == "sentences" ? .sentences : .none
        traits.smartQuotes = c.smartQuotes
        if c.field == "private" { traits.isSecure = true }
        if c.field == "url" { traits.keyboard = .url }
        if c.field == "numeric" { traits.keyboard = .numberPad }
        engine.traits = traits
        var time: TimeInterval = 100
        engine.clock = { time }
        if c.context != "fresh" { engine.documentDidChange() }
        if ["queued", "reentrant"].contains(host.context) { host.callback = { engine.documentDidChange() } }
        let identity = engine.suggester === stack.suggester && engine.glide === stack.glide && engine.suggester?.model === stack.model && engine.suggester?.model.lexicon === stack.lexicon && engine.settings.language == language
        #expect(identity)
        var completed = c.route != "controller"
        var notes: [String] = []
        var snapshots: [Snapshot] = []
        func snapshot(_ action: String, expected: String? = nil, caret: Int? = nil, phantom: Bool? = nil) {
          snapshots.append(Snapshot(action: action, text: host.text, caret: host.cursor,
                                    utf16Caret: String(host.chars.prefix(host.cursor)).utf16.count,
                                    selection: host.selectionLength, logicalBefore: engine.contextBeforeCursor,
                                    phantom: engine.phantomSpace, lastEdit: String(describing: engine.lastEdit),
                                    deleteUndoesLastEdit: engine.deleteUndoesLastEdit,
                                    smartSpacing: engine.policy.smartSpacing, autoSpaces: engine.policy.autoSpaces,
                                    checkpointsPass: (expected == nil || Array(host.text.utf8) == Array(expected!.utf8)) && (caret == nil || host.cursor == caret) && (phantom == nil || engine.phantomSpace == phantom)))
        }
        snapshot("initial")
        if !completed { notes.append("UIKit controller delivery unavailable; no engine substitute") }
        for a in c.actions where completed {
          time += 0.01
          switch a.kind {
          case "transcript": engine.insertTranscript(a.text ?? "")
          case "spaced": engine.insertText(a.text ?? "", spacing: true)
          case "raw": engine.insertText(a.text ?? "", spacing: false)
          case "keys": engine.type(a.text ?? "")
          case "space": engine.insertSpace()
          case "return": engine.insertReturn()
          case "backspace": engine.deleteBackward()
          case "hostMove":
            host.move(a.offset ?? 0, selection: a.selection ?? 0)
            if a.newIdentity == true { host.documentIdentifier = UUID() }
            if a.callback == true { engine.documentDidChange() }
          case "hostReplace":
            host.replace(a.text ?? "", offset: a.offset ?? 0, selection: a.selection ?? 0)
            if a.newIdentity == true { host.documentIdentifier = UUID() }
            if a.callback == true { engine.documentDidChange() }
          case "private":
            traits.isSecure = true
            engine.traits = traits
            if a.newIdentity == true { host.documentIdentifier = UUID() }
            if a.callback == true { engine.documentDidChange() }
          case "nativeUndo":
            completed = false
            notes.append("Native host Undo unavailable; backspace is not a substitute")
          default:
            completed = false
            notes.append("Unsupported action: " + a.kind)
          }
          snapshot(a.kind, expected: a.expected, caret: a.caret, phantom: a.phantom)
        }
        host.flush()
        snapshot("flushed")
        if !host.bounded {
          completed = false
          notes.append("Character-bound host cannot represent this range/grapheme transition; retained unavailable bounds failure")
        }
        let oracle = c.oracle != "ambiguous"
        if let rationale = c.rationale { notes.append(rationale) }
        results.append(Result(id: c.id, contentLabel: c.contentLabel, category: c.category,
                              language: language.rawValue, dictionary: dictionary, dictionarySHA256: hash,
                              modelMode: language.rawValue, settingsLanguage: engine.settings.language.rawValue,
                              modelIdentityVerified: identity, expected: c.expected, actual: host.text,
                              expectedCaret: c.caret, actualCaret: host.cursor,
                              utf16Caret: String(host.chars.prefix(host.cursor)).utf16.count,
                              expectedSelection: 0, actualSelection: host.selectionLength,
                              exact: oracle && completed && Array(host.text.utf8) == Array(c.expected.utf8) && host.cursor == c.caret && host.selectionLength == 0 && snapshots.allSatisfy(\.checkpointsPass),
                              oracleDefined: oracle, completed: completed, conserved: host.conserved,
                              bounded: host.bounded, nativeDelivery: "unverified: no UIKit/device execution",
                              nativeUndo: "unavailable: no host Undo API", smartPunctuation: settings.smartPunctuation,
                              smartQuotes: traits.smartQuotes, field: c.field ?? "normal", context: c.context ?? "full",
                              primitives: host.primitives, snapshots: snapshots, notes: notes))
      }
    }
    #expect(results.count == file.cases.count * KeyboardLanguage.allCases.count)
    for c in file.cases {
      #expect(Set(results.filter { $0.id == c.id }.map(\.language)) == Set(["en_US", "es", "en_es"]))
    }
    let encoder = JSONEncoder()
    encoder.outputFormatting = [.prettyPrinted, .sortedKeys, .withoutEscapingSlashes]
    try encoder.encode(results).write(to: URL(fileURLWithPath: output))
    print("R26 exact \(results.filter(\.exact).count)/\(results.filter(\.oracleDefined).count); ambiguous \(results.filter { !$0.oracleDefined }.count); unavailable \(results.filter { !$0.completed }.count); conservation failures \(results.filter { !$0.conserved }.count); bounds failures \(results.filter { !$0.bounded }.count)")
    #expect(results.allSatisfy { $0.conserved && $0.bounded })
  }

  /// Source-derived controls are exposed development evidence, never blind gains.
  @Test(arguments: KeyboardLanguage.allCases)
  func sourceOwnershipControls(language: KeyboardLanguage) throws {
    for smart in [false, true] {
      for (key, expected) in [("?!", "Bring fresh ?!supplies"), (",", "Bring fresh, supplies"), ("numbers-apostrophe", smart ? "Bring fresh’, supplies" : "Bring fresh', supplies"), ("explicit-page", "Bring fresh ,supplies"), ("shift", "Bring fresh ,supplies")] {
        let c = try JSONDecoder().decode(Case.self, from: Data(#"{"id":"source","contentLabel":"control","category":"source","prefix":"Bring ","selected":"","suffix":"supplies","actions":[],"expected":"","caret":0}"#.utf8))
        let host = Host(c)
        let stack = try LanguageStack(dictionaryURL: Fixtures.languageData.appendingPathComponent(language.rawValue + ".pkdict"), personalURL: nil)
        var settings = KeyboardSettings()
        settings.language = language
        settings.smartPunctuation = smart
        let engine = KeyboardEngine(document: host, suggester: stack.suggester, glide: stack.glide, settings: settings)
        var traits = InputTraits(); traits.capitalization = .none
        engine.traits = traits
        engine.documentDidChange()
        if key == "numbers-apostrophe" { engine.page = .numbers }
        engine.insertTranscript("fresh")
        switch key {
        case "numbers-apostrophe": engine.insertCharacter("'"); engine.insertCharacter(",")
        case "explicit-page": engine.page = .numbers; engine.insertCharacter(",")
        case "shift": engine.shiftDown(); engine.shiftUp(); engine.insertCharacter(",")
        default: engine.insertCharacter(key) // one key, not split into Characters
        }
        #expect(host.text == expected)
        #expect(host.conserved && host.bounded)
        #expect(!engine.phantomSpace)
      }
    }
  }

  @Test(arguments: KeyboardLanguage.allCases)
  func sourceScriptConservationControls(language: KeyboardLanguage) throws {
    for (payload, suffix, separates) in [("café", "supplies", true), ("cafe\u{301}", "supplies", true), ("niño", "supplies", true), ("fresh", "ñandú", true), ("世", "界", false), ("fresh", "界", false), ("λογος", "supplies", false), ("текст", "supplies", false), ("café 世", "supplies", false)] {
      let fixture: [String: Any] = ["id":"script","contentLabel":"control","category":"source","prefix":"Bring ","selected":"","suffix":suffix,"actions":[],"expected":"","caret":0]
      let c = try JSONDecoder().decode(Case.self, from: JSONSerialization.data(withJSONObject: fixture))
      let host = Host(c)
      let stack = try LanguageStack(dictionaryURL: Fixtures.languageData.appendingPathComponent(language.rawValue + ".pkdict"), personalURL: nil)
      var settings = KeyboardSettings(); settings.language = language
      let engine = KeyboardEngine(document: host, suggester: stack.suggester, glide: stack.glide, settings: settings)
      var traits = InputTraits(); traits.capitalization = .none; engine.traits = traits
      engine.documentDidChange()
      engine.insertTranscript(payload)
      let expected = "Bring " + payload + (separates ? " " : "") + suffix
      #expect(Array(host.text.utf8) == Array(expected.utf8))
      #expect(host.bounded && host.conserved)
    }
  }

  @Test(arguments: KeyboardLanguage.allCases)
  func sourceReentrantDestinationControls(language: KeyboardLanguage) throws {
    for mutation in ["identity", "move", "private", "replace", "suffix", "selection"] {
      let fixture: [String: Any] = ["id":"reentrant","contentLabel":"control","category":"source","prefix":"Bring ","selected":"","suffix":"supplies","actions":[],"expected":"","caret":0]
      let c = try JSONDecoder().decode(Case.self, from: JSONSerialization.data(withJSONObject: fixture))
      let host = Host(c)
      let stack = try LanguageStack(dictionaryURL: Fixtures.languageData.appendingPathComponent(language.rawValue + ".pkdict"), personalURL: nil)
      var settings = KeyboardSettings(); settings.language = language
      let engine = KeyboardEngine(document: host, suggester: stack.suggester, glide: stack.glide, settings: settings)
      var traits = InputTraits(); traits.capitalization = .none; engine.traits = traits
      engine.documentDidChange()
      host.afterApply = {
        host.afterApply = nil
        switch mutation {
        case "identity": host.documentIdentifier = UUID()
        case "move": host.move(0, selection: 0)
        case "private": traits.isSecure = true; engine.traits = traits
        case "suffix": host.replace("Bring freshforeign", offset: 11, selection: 0)
        case "selection": host.move(11, selection: 1)
        default: host.replace("foreign", offset: 7, selection: 0)
        }
        engine.documentDidChange()
      }
      engine.insertTranscript("fresh")
      let expected = mutation == "replace" ? "foreign" : mutation == "suffix" ? "Bring freshforeign" : "Bring freshsupplies"
      #expect(Array(host.text.utf8) == Array(expected.utf8))
      #expect(host.primitives.filter { $0.kind == "insert" }.count == 1)
      #expect(host.bounded && host.conserved)
      #expect(!String(describing: engine.lastEdit).hasPrefix("phraseSpace"))
    }
  }

  @Test(arguments: KeyboardLanguage.allCases)
  func sourceReentrantContinuationControls(language: KeyboardLanguage) throws {
    for stage in ["delete", "punctuation", "marker"] {
      for mutation in ["identity", "move", "private", "replace", "suffix", "selection"] {
        let fixture: [String: Any] = ["id":"interrupted","contentLabel":"control","category":"source","prefix":"Bring ","selected":"","suffix":"supplies","actions":[],"expected":"","caret":0]
        let c = try JSONDecoder().decode(Case.self, from: JSONSerialization.data(withJSONObject: fixture))
        let host = Host(c)
        let stack = try LanguageStack(dictionaryURL: Fixtures.languageData.appendingPathComponent(language.rawValue + ".pkdict"), personalURL: nil)
        var settings = KeyboardSettings(); settings.language = language
        let engine = KeyboardEngine(document: host, suggester: stack.suggester, glide: stack.glide, settings: settings)
        var traits = InputTraits(); traits.capitalization = .none; engine.traits = traits
        engine.documentDidChange()
        engine.insertTranscript("fresh")
        let initialPrimitives = host.primitives.count
        host.afterApply = {
          guard let last = host.primitives.last,
                (stage == "delete" && last.kind == "delete") || (stage == "punctuation" && last.kind == "insert" && last.value == ",") || (stage == "marker" && last.kind == "insert" && last.value == " ") else { return }
          host.afterApply = nil
          switch mutation {
          case "identity": host.documentIdentifier = UUID()
          case "move": host.move(0, selection: 0)
          case "private": traits.isSecure = true; engine.traits = traits
          case "suffix": host.replace("Bring freshforeign", offset: 11, selection: 0)
          case "selection": host.move(0, selection: 1)
          default: host.replace("foreign", offset: 7, selection: 0)
          }
          engine.documentDidChange()
        }
        engine.insertCharacter(",")
        let unchanged = stage == "delete" ? "Bring freshsupplies" : stage == "punctuation" ? "Bring fresh,supplies" : "Bring fresh, supplies"
        let expected = mutation == "replace" ? "foreign" : mutation == "suffix" ? "Bring freshforeign" : unchanged
        #expect(Array(host.text.utf8) == Array(expected.utf8))
        #expect(host.primitives.dropFirst(initialPrimitives).filter { $0.kind == "insert" }.count == (stage == "delete" ? 0 : stage == "punctuation" ? 1 : 2))
        #expect(host.bounded && host.conserved)
        #expect(!String(describing: engine.lastEdit).hasPrefix("phraseSpace"))
      }
    }
  }
