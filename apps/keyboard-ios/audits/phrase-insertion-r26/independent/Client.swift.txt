import Foundation
import KeyboardCore
struct Input: Codable { let id, stage: String; let callback: Bool }
struct Event: Codable { let kind, value, before, after: String; let caret: Int }
struct Output: Codable { let id, language, document: String; let caret: Int; let events: [Event] }
final class Host: TextDocument {
 var text = "Bring supplies"; var cursor = 6; var selection = 0
 let documentIdentifier: UUID? = UUID()
 var hook: ((String,String)->Void)?
 var notify: (() -> Void)?
 var events: [Event] = []
 var chars: [Character] { Array(text) }
 var documentContextBeforeInput: String? { String(chars.prefix(cursor)) }
 var documentContextAfterInput: String? { String(chars.dropFirst(cursor + selection)) }
 var selectedText: String? { selection == 0 ? nil : String(chars.dropFirst(cursor).prefix(selection)) }
 var hasText: Bool { !text.isEmpty }
 func insertText(_ value: String) {
  let before = text
  text = String(chars.prefix(cursor)) + value + String(chars.dropFirst(cursor + selection)); cursor += value.count; selection = 0
  events.append(Event(kind:"insert",value:value,before:before,after:text,caret:cursor)); hook?("insert",value)
 }
 func deleteBackward() {
  let before = text
  if selection > 0 { text = String(chars.prefix(cursor)) + String(chars.dropFirst(cursor+selection)); selection=0 }
  else if cursor > 0 { text = String(chars.prefix(cursor-1)) + String(chars.dropFirst(cursor)); cursor -= 1 }
  events.append(Event(kind:"delete",value:"",before:before,after:text,caret:cursor)); hook?("delete","")
 }
 func foreign() { let before=text; text="FOREIGNsupplies"; cursor=7; selection=0; events.append(Event(kind:"hostReplace",value:"",before:before,after:text,caret:cursor)) }
}
let args=CommandLine.arguments
let inputs=try JSONDecoder().decode([Input].self,from:Data(contentsOf:URL(fileURLWithPath:args[1])))
var rows:[Output]=[]
for language in KeyboardLanguage.allCases {
 for input in inputs {
  let stack=try LanguageStack(dictionaryURL:URL(fileURLWithPath:args[2]).appendingPathComponent(language.rawValue+".pkdict"),personalURL:nil)
  let host=Host(); var settings=KeyboardSettings(); settings.language=language; settings.autoCapitalization=false
  let engine=KeyboardEngine(document:host,suggester:stack.suggester,glide:stack.glide,settings:settings)
  engine.documentDidChange(); engine.insertTranscript("fresh")
  host.hook = { kind,value in
   let match = input.stage == "delete" ? kind == "delete" : input.stage == "punctuation" ? kind == "insert" && value == "," : input.stage == "marker" ? kind == "insert" && value == " " : false
   if match { host.hook=nil; host.foreign(); if input.callback { engine.documentDidChange() } }
  }
  engine.insertCharacter(",")
  rows.append(Output(id:input.id,language:language.rawValue,document:host.text,caret:host.cursor,events:host.events))
 }
}
let encoder=JSONEncoder(); encoder.outputFormatting=[.prettyPrinted,.sortedKeys,.withoutEscapingSlashes]
try encoder.encode(rows).write(to:URL(fileURLWithPath:args[3]))
