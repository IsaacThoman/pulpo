import UIKit
import XCTest
@testable import KeyboardUI

/// Native controller regression coverage. Running these requires a parent-owned
/// simulator; core benchmark results do not establish these tests passed.
@MainActor
final class TechnicalTokenControllerTests: XCTestCase {
  func testLiteralPathThenOrdinaryCorrectionWithImmediateAndBatchedHosts() throws {
    for language in KeyboardLanguage.allCases {
      for batching in [false, true] {
        let h = try harness("Save ", language: language)
        defer { h.disappear() }
        try type("/var/log/audit.txt ", on: h, batching: batching)
        let prose = language == .spanish ? "es inportante. " : "and teh report. "
        try type(prose, on: h, batching: batching)
        h.settle()
        let expected = language == .spanish ? "Save /var/log/audit.txt es importante. " : "Save /var/log/audit.txt and the report. "
        XCTAssertEqual(h.proxy.host.string, expected, "\(language), batched=\(batching)")
      }
    }
  }

  func testReplacementOfSelectedFilenameKeepsTheEntireFollowingDocument() throws {
    let h = try harness("File: old.txtis ready.", language: .bilingual)
    defer { h.disappear() }
    h.proxy.hostChanges { state in
      state.cursor = "File: ".count
      state.selection = "old.txt".count
    }
    h.deliver(callbacks: [.selectionWill, .selectionDid])
    try type("draft.txt ", on: h, batching: false)
    XCTAssertEqual(h.proxy.host.string, "File: draft.txt is ready.")
    XCTAssertEqual(h.proxy.host.after, "is ready.")
  }

  func testPrivateTraitsStillPreventCorrectionAndLearning() throws {
    for field in ["secure", "email", "url"] {
      let h = try harness("value ", language: .bilingual)
      defer { h.disappear() }
      if field == "secure" { h.proxy.secureEntryValue = true }
      if field == "email" { h.proxy.keyboardTypeValue = .emailAddress }
      if field == "url" { h.proxy.keyboardTypeValue = .URL }
      h.fire([.textDid])
      try type("teh manana /var/log/audit.txt ", on: h, batching: false)
      h.engine.save()
      XCTAssertEqual(h.proxy.host.string, "value teh manana /var/log/audit.txt ")
      XCTAssertEqual(h.engine.suggester?.model.personal.totalCount, 0)
    }
  }

  private func harness(_ text: String, language: KeyboardLanguage) throws -> ControllerHarness {
    let h = ControllerHarness(text: text, waitsForEmoji: false)
    let stack = try LanguageStack(dictionaryURL: ControllerHarness.languageData.appendingPathComponent(language.rawValue + ".pkdict"), personalURL: nil)
    h.engine.useLanguageStack(stack)
    h.engine.settings.language = language
    h.engine.documentDidChange()
    return h
  }

  /// Symbols use actual page-switch keys and letters actual key activation.
  /// No direct typing calls on the engine, so controller/proxy dispatch remains
  /// under test.
  private func type(_ text: String, on h: ControllerHarness, batching: Bool) throws {
    for character in text {
      h.now += 0.25
      let action: KeyAction = character == " " ? .space : .character(String(character))
      if h.key(action) == nil {
        let target: KeyboardPage = character.isLetter ? .letters : .numbers
        let switchKey = try XCTUnwrap(h.key(.page(target)))
        h.controller.keysView(h.controller.rootView.keys, touchDown: switchKey)
        h.controller.rootView.layoutIfNeeded()
      }
      let key = try XCTUnwrap(h.key(action))
      let keys = h.controller.rootView.keys
      h.controller.keysView(keys, touchDown: key)
      let point = keys.letterCenters[character] ?? CGPoint(x: keys.bounds.midX, y: keys.bounds.maxY - 20)
      h.controller.keysView(keys, activate: key, at: point)
      if !batching { h.settle() }
      h.pump()
    }
  }
}
