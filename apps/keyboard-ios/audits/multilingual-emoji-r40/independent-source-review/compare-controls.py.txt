from pathlib import Path
import json,itertools,collections
r=Path('/tmp/pulpo-r40-source-review');inputs=json.load(open(r/'independent-controls.json'));failures=[];raw=[];groups=collections.defaultdict(list);rows=0;eligibility=[]
for inp,bl,cl in zip(inputs,open(r/'baseline-controls.jsonl'),open(r/'candidate-controls.jsonl')):
 b=json.loads(bl);c=json.loads(cl);rows+=1;assert inp['id']==b['id']==c['id']
 assert b['index']==c['index'];assert b['exactNames']==c['exactNames'];assert b['oracle']==c['oracle']
 expect=b['search']+b['oracle']['suffix'];errors=[]
 if c['search']!=expect:errors.append('independent-normative-array')
 if c['search'][:len(b['search'])]!=b['search']:errors.append('legacy-prefix')
 if len(c['search'])>max(0,c['cap']):errors.append('cap')
 if len(set(c['search']))!=len(c['search']):errors.append('dedupe')
 if not set(b['search'])<=set(b['oracle']['E']):errors.append('E-completeness')
 row={'id':c['id'],'q':c['q'],'cap':c['cap'],'baseline':b['search'],'candidate':c['search'],'expected':expect,'oracle':b['oracle'],'errors':errors}
 raw.append(row)
 if errors:failures.append(row)
 if inp.get('group'):
  C=tuple(tuple(b['oracle'][k]) for k in ('T','W','X'))
  groups[(inp['group'],inp['cap'],C)].append(row)
 for key in ['E','rankedEligibleOutsideE']:
  if len(b['oracle'][key])>96: pass
 if inp['id'].startswith('r40-ranking'):eligibility.append(row)
invfail=[]
for key,rs in groups.items():
 if len({tuple(x['candidate'][len(x['baseline']):]) for x in rs})>1: invfail.append({'group':str(key),'rows':rs})
s={'independentRows':rows,'failures':len(failures),'failedIds':[x['id'] for x in failures],'sameTripleAcrossRawFormsAndPreferencesGroups':len(groups),'failedInvariantGroups':len(invfail),'sameIndexAndExactNameWinnersEveryFixture':True,'emptyEnvelopeCapacityRows':sum(not x['oracle']['E'] and len(x['oracle']['rankedEligibleOutsideE'])>=96 for x in raw),'mixedCap1Rows':sum(x['id'].startswith(('mixed/','mixed2/')) and x['cap']==1 for x in raw),'frozenRankingRows':len(eligibility),'frozenRankingNormativeConformance':all(not x['errors'] for x in eligibility)}
for filename,value in [('independent-controls-raw.json',raw),('independent-controls-failures.json',failures),('independent-invariance-failures.json',invfail),('independent-ranking-adjudication.json',eligibility),('independent-controls-summary.json',s)]: (r/filename).write_text(json.dumps(value,ensure_ascii=False,indent=2)+'\n')
print(json.dumps(s,ensure_ascii=False))
for row in failures[:3]:print(json.dumps(row,ensure_ascii=False))
