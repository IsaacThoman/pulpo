#!/usr/bin/env python3
"""Frozen, literal evaluation of independently authored synthetic intentions."""
import argparse
import gzip
import hashlib
import json
from pathlib import Path


def read(path):
    data = path.read_bytes()
    return json.loads(gzip.decompress(data) if path.suffix == ".gz" else data)


def utf16(text):
    return len(text.encode("utf-16-le")) // 2


def evaluate(packet, baseline, candidate):
    sha = hashlib.sha256(packet.read_bytes()).hexdigest()
    cases = read(packet)["cases"]
    assert baseline["packetSHA256"] == candidate["packetSHA256"] == sha
    ids = [case["id"] for case in cases]
    assert len(ids) == len(set(ids))
    assert [row["id"] for row in baseline["rows"]] == ids
    assert [row["id"] for row in candidate["rows"]] == ids
    result = {"schema": 1, "packetSHA256": sha, "rows": []}
    for case, old, new in zip(cases, baseline["rows"], candidate["rows"], strict=True):
        assert old["input"] == new["input"] and old["callbacks"] == new["callbacks"], case["id"]
        record = {"id": case["id"], "language": case["language"], "class": case["oracle"]["valueClass"]}
        for arm, row in [("baseline", old), ("candidate", new)]:
            oracle, after = case["oracle"], row["after"]
            accepted = any(after["text"] == item["text"] and after["caret"] == item["caret"] and after["selectionLength"] == 0 for item in oracle["accepted"])
            glides = row["events"].count("glide")
            taps = sum(event.startswith("tap:") for event in row["events"])
            action = glides == oracle["glideCallbacks"] and taps == oracle["tapCallbacks"]
            offered = after["visibleChoices"] == after["engineChoices"]
            available = row["choiceAvailableCount"] > 0
            fired = row["choiceFired"] is not None
            binding = available == fired
            if fired:
                choice = row["choiceFired"]
                state = row["afterChoice"]
                binding = binding and any(item["text"] == choice and item["kind"] == "alternative" for item in after["visibleChoices"])
                binding = binding and state["text"] == case["before"] + choice + case["after"]
                binding = binding and state["caret"] == utf16(case["before"] + choice) and state["selectionLength"] == 0
                binding = binding and state["text"] != after["text"]
            model_lists = row["preDispatch"]
            fidelity = glides == 0 or (len(model_lists) == 1 and model_lists[0] == row["terminalReference"])
            record[arm] = {"accepted": accepted, "action": action, "bar": offered, "choiceBinding": binding,
                           "choiceAvailable": available, "choiceFired": fired, "referenceFidelity": fidelity,
                           "text": after["text"], "caret": after["caret"], "glides": glides, "taps": taps}
        record["gain"] = not record["baseline"]["accepted"] and record["candidate"]["accepted"]
        record["regression"] = record["baseline"]["accepted"] and not record["candidate"]["accepted"]
        result["rows"].append(record)
    rows = result["rows"]
    result["counts"] = {"raw": len(rows), "gains": sum(row["gain"] for row in rows),
                        "regressions": sum(row["regression"] for row in rows),
                        "ties": sum(not row["gain"] and not row["regression"] for row in rows),
                        "failureToFailure": sum(not row["baseline"]["accepted"] and not row["candidate"]["accepted"] for row in rows)}
    for arm in ["baseline", "candidate"]:
        result["counts"][arm] = {key: sum(row[arm][key] for row in rows) for key in ["accepted", "action", "bar", "choiceBinding", "choiceAvailable", "choiceFired", "referenceFidelity"]}
        result["counts"][arm]["choiceUnfired"] = sum(row[arm]["choiceAvailable"] and not row[arm]["choiceFired"] for row in rows)
        result["counts"][arm]["choiceUnavailable"] = sum(not row[arm]["choiceAvailable"] for row in rows)
    # This is a limited synthetic promotion gate, not a population accuracy claim.
    result["admit"] = result["counts"]["gains"] > 0 and result["counts"]["regressions"] == 0
    result["admit"] &= all(row[arm][key] for row in rows for arm in ["baseline", "candidate"] for key in ["action", "bar", "choiceBinding"])
    result["admit"] &= all(row["candidate"]["referenceFidelity"] for row in rows)
    result["admit"] &= all(row[arm]["accepted"] for row in rows if row["class"] in {"cancel", "literal"} for arm in ["baseline", "candidate"])
    result["admit"] &= any(row["baseline"]["accepted"] and row["candidate"]["accepted"] for row in rows if row["class"] in {"positive", "fast_control"})
    result["admit"] &= result["counts"]["failureToFailure"] > 0
    return result


if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("packet", type=Path)
    parser.add_argument("baseline", type=Path)
    parser.add_argument("candidate", type=Path)
    parser.add_argument("output", type=Path)
    args = parser.parse_args()
    result = evaluate(args.packet, read(args.baseline), read(args.candidate))
    args.output.write_text(json.dumps(result, sort_keys=True, indent=2) + "\n")
    print(json.dumps({"admit": result["admit"], "counts": result["counts"]}, sort_keys=True))
