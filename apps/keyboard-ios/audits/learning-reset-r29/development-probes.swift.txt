import Foundation
import Testing
@testable import KeyboardCore

struct PersonalDictionaryResetR29Tests {
  private func withURL(_ body: (URL) throws -> Void) throws {
    let directory = FileManager.default.temporaryDirectory.appendingPathComponent("pk-reset-r29-\(UUID().uuidString)")
    try FileManager.default.createDirectory(at: directory, withIntermediateDirectories: true)
    defer { try? FileManager.default.removeItem(at: directory) }
    try body(directory.appendingPathComponent("PersonalDictionary.json"))
  }

  @Test func retainedDirtyWriterMustNotRestorePreClearContents() throws {
    try withURL { url in
      let writer = PersonalDictionary(url: url)
      writer.learn("oldword", after: "the", pinned: true)
      writer.save()
      writer.learn("unsavedold", after: "the", pinned: true)
      PersonalDictionary(url: url).clear()
      writer.saveInBackground()
      let disk = PersonalDictionary(url: url)
      #expect(disk.totalCount == 0)
      #expect(disk.historyCount("the") == 0)
    }
  }

  @Test func newLearningAfterClearMustNotCarryRetainedOldContents() throws {
    try withURL { url in
      let writer = PersonalDictionary(url: url)
      writer.pin("oldword")
      writer.save()
      PersonalDictionary(url: url).clear()
      writer.learn("newword", after: "next", pinned: true)
      writer.save()
      let disk = PersonalDictionary(url: url)
      #expect(!disk.isKnown("oldword"))
      #expect(disk.count(of: "newword") == 1)
      #expect(disk.pairCount("next", "newword") == 1)
    }
  }

  @Test func reloadMustPreserveUnsavedLearningWhenExternalChangeIsOrdinary() throws {
    try withURL { url in
      let writer = PersonalDictionary(url: url), other = PersonalDictionary(url: url)
      writer.pin("localword")
      other.pin("externalword")
      other.save()
      #expect(writer.reloadIfChangedOnDisk())
      writer.save()
      let disk = PersonalDictionary(url: url)
      #expect(disk.isKnown("localword"))
      #expect(disk.isKnown("externalword"))
    }
  }

  @Test func staleSaveMustPreserveExternalLearningAndForget() throws {
    try withURL { url in
      let writer = PersonalDictionary(url: url)
      writer.pin("café")
      writer.save()
      let other = PersonalDictionary(url: url)
      other.forget("CAFE")
      other.pin("novedad")
      other.save()
      writer.pin("localword")
      writer.save()
      let disk = PersonalDictionary(url: url)
      #expect(disk.isBlocked("café"))
      #expect(!disk.isKnown("café"))
      #expect(disk.isKnown("novedad"))
      #expect(disk.isKnown("localword"))
    }
  }

  @Test func appearanceReloadThenLearningRespectsClear() throws {
    try withURL { url in
      let writer = PersonalDictionary(url: url)
      writer.pin("oldword")
      writer.saveInBackground()
      PersonalDictionary(url: url).clear()
      #expect(writer.reloadIfChangedOnDisk())
      writer.pin("nuevapalabra")
      writer.saveInBackground()
      let disk = PersonalDictionary(url: url)
      #expect(!disk.isKnown("oldword"))
      #expect(disk.isKnown("nuevapalabra"))
    }
  }

  @Test func noURLRemainsLocalAndResettable() {
    let writer = PersonalDictionary(url: nil)
    writer.pin("café")
    writer.saveInBackground()
    #expect(writer.isKnown("café"))
    #expect(!writer.reloadIfChangedOnDisk())
    writer.clear()
    #expect(writer.totalCount == 0)
    #expect(!writer.isKnown("café"))
    writer.pin("nuevo")
    writer.save()
    #expect(writer.isKnown("nuevo"))
  }
  @Test func failedWriteRetriesWithoutAnotherLearningAction() throws {
    try withURL { original in
      let blocker = original.deletingLastPathComponent().appendingPathComponent("blocked")
      try Data("not a directory".utf8).write(to: blocker)
      let url = blocker.appendingPathComponent("PersonalDictionary.json")
      let writer = PersonalDictionary(url: url)
      writer.pin("retryword")
      writer.saveInBackground()
      #expect(writer.hasUnsavedChanges)
      #expect(writer.isKnown("retryword"))
      try FileManager.default.removeItem(at: blocker)
      writer.save()
      #expect(!writer.hasUnsavedChanges)
      #expect(PersonalDictionary(url: url).isKnown("retryword"))
    }
  }

  @Test func ownCompletedWriteReloadDoesNotLoseLaterUnsavedLearning() throws {
    try withURL { url in
      let writer = PersonalDictionary(url: url)
      writer.pin("savedword")
      writer.saveInBackground()
      writer.pin("laterword")
      #expect(!writer.reloadIfChangedOnDisk())
      #expect(writer.isKnown("laterword"))
      #expect(writer.hasUnsavedChanges)
      writer.save()
      #expect(PersonalDictionary(url: url).isKnown("laterword"))
    }
  }

}
