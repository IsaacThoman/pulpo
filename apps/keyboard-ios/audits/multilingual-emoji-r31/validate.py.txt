"""Opaque pre-seal validation; emits only row IDs/counts, never hidden queries."""
import json, sys, unicodedata, hashlib
from pathlib import Path

def validate(path, exclusions):
 x=json.loads(Path(path).read_text()); ex=json.loads(Path(exclusions).read_text())
 blocked={unicodedata.normalize('NFC',v).casefold() for v in ex['strings']}
 assert x['version']==1 and isinstance(x['families'],list) and isinstance(x['cases'],list)
 assert set(x)=={'version','families','cases'}
 ids=set();actions=set();strata=set();usefulfamilies=set();fixturepairs={}; wrapperparams=set()
 for r in x['cases']:
  rid=r.get('id','?'); required={'id','family','action','query','acceptable','maxRank','role'}
  assert required <= r.keys(),f'{rid}: missing required fields'
  allowed=required|{'stratum','recents','limit','entries','spanishAliases','forbidden','equivalentTo'}
  assert r.keys()<=allowed,f'{rid}: unsupported field'
  assert rid not in ids;ids.add(rid)
  assert r['family'] in x['families'] and r['action'] in {'search','wrapper','fixture'}
  assert r['role'] in {'useful','control'} and isinstance(r['query'],str)
  assert r['acceptable'] and all(isinstance(t,str) and t for t in r['acceptable'])
  assert type(r['maxRank']) is int and 0<r['maxRank']<=60
  for t in r['acceptable']:
   assert unicodedata.normalize('NFC',t).casefold() not in blocked,f'{rid}: exposed target'
  assert unicodedata.normalize('NFC',r['query']).casefold() not in blocked,f'{rid}: exposed query'
  assert r['family'] not in ex['families'],f'{rid}: exposed family'
  actions.add(r['action'])
  if r['role']=='useful':
   assert r.get('stratum') in {'en','es','mixed'}
   strata.add(r['stratum']);usefulfamilies.add(r['family'])
  if r['action']=='wrapper':
   assert 'recents' in r and 'limit' in r
   assert isinstance(r['recents'],list) and type(r['limit']) is int
   wrapperparams.update({'recents','limit'})
  else: assert not {'recents','limit'}&r.keys()
  if r['action']=='fixture':
   assert r['entries'] and r['spanishAliases']
   for e in r['entries']:assert set(e)=={'emoji','name','keywords'}
   for a in r['spanishAliases']:assert set(a)=={'emoji','name'}
   assert set(r['acceptable'])<=set(e['emoji'] for e in r['entries'])
   key=json.dumps([r['entries'],r['spanishAliases']],sort_keys=True,ensure_ascii=False)
   fixturepairs.setdefault(key,[]).append(r)
  else: assert not {'entries','spanishAliases'}&r.keys()
  if 'forbidden' in r:assert isinstance(r['forbidden'],list)
  if 'equivalentTo' in r:assert isinstance(r['equivalentTo'],str)
 assert actions=={'search','wrapper','fixture'} and strata=={'en','es','mixed'} and len(usefulfamilies)>=6
 assert wrapperparams=={'recents','limit'}
 discriminating=False
 for rows in fixturepairs.values():
  aliases=rows[0]['spanishAliases']
  for a in aliases:
   name=unicodedata.normalize('NFC',a['name'])
   if 'ñ' not in name:continue
   other=next((b for b in aliases if b['name']==name.replace('ñ','n') and b['emoji']!=a['emoji']),None)
   if not other:continue
   queries={r['query']:r for r in rows}
   expected=[(name,a['emoji'],other['emoji']),(other['name'],other['emoji'],a['emoji']),(unicodedata.normalize('NFD',name),a['emoji'],other['emoji'])]
   discriminating=all(q in queries and queries[q]['acceptable']==[target] and banned in queries[q].get('forbidden',[]) for q,target,banned in expected)
 assert discriminating,'missing generic discriminating fresh n/ñ and NFD fixture'
 for r in x['cases']:
  if 'equivalentTo' in r:assert r['equivalentTo'] in ids
 print(json.dumps({'rows':len(ids),'families':len(usefulfamilies),'actions':sorted(actions),'sha256':hashlib.sha256(Path(path).read_bytes()).hexdigest(),'staticValidation':'PASS','payloadNotDisplayed':True}))
if __name__=='__main__':validate(*sys.argv[1:])
