func foldedName(_ s:String)->String { s.folding(options:[.diacriticInsensitive,.caseInsensitive],locale:posix).split(whereSeparator: \.isWhitespace).joined(separator:" ") }
func normalized(_ s:String)->String { s.folding(options:[.diacriticInsensitive,.caseInsensitive],locale:posix).unicodeScalars.map { CharacterSet.alphanumerics.contains($0) ? String($0) : ($0 == "'" || $0 == "\u{2019}" ? "" : " ") }.joined().split(whereSeparator: \.isWhitespace).joined(separator:" ") }
func words(_ s:String)->Set<String> { Set(s.split(separator:" ").map(String.init)) }
func spanish(_ s:String)->[String] {
 let f=s.lowercased(with:posix).decomposedStringWithCanonicalMapping
 var out:Set<String>=[], token="", vowel=false
 func flush() {if !token.isEmpty {out.insert(token.precomposedStringWithCanonicalMapping)};token="";vowel=false}
 for c in f.unicodeScalars {
  switch c.properties.generalCategory {
  case .uppercaseLetter,.lowercaseLetter,.titlecaseLetter,.modifierLetter,.otherLetter:
   if CharacterSet.letters.contains(c) {token += String(c);vowel="aeiou".unicodeScalars.contains(c)} else {flush()}
  case .nonspacingMark,.spacingMark,.enclosingMark:
   if !token.isEmpty && !vowel {token += String(c)}
  default:flush()
  }
 }
 flush();return out.sorted()
}
struct Observed {
 let entries:[Emoji], keys:[[String]], exact:[String:Int]
 init(_ s:EmojiSuggester) {
  let fields=Dictionary(uniqueKeysWithValues:Mirror(reflecting:s).children.compactMap { c in c.label.map {($0,c.value)} })
  entries=fields["emoji"] as! [Emoji];exact=fields["exactNames"] as! [String:Int]
  var ks=Array(repeating:[String](),count:entries.count)
  for term in Mirror(reflecting:fields["index"]!).children {
   let f=Dictionary(uniqueKeysWithValues:Mirror(reflecting:term.value).children.compactMap {c in c.label.map {($0,c.value)}})
   for m in Mirror(reflecting:f["matches"]!).children {
    let a=Dictionary(uniqueKeysWithValues:Mirror(reflecting:m.value).children.compactMap {c in c.label.map {($0,c.value)}})
    ks[a["id"] as! Int].append(f["key"] as! String)
   }
  }
  keys=ks
 }
 func vocab(_ i:Int)->Set<String> {
  var v=Set(([entries[i].name]+(entries[i].keywords ?? [])).flatMap {words(normalized($0))})
  for k in keys[i] {v.insert(k);v.formUnion(words(k))};return v
 }
 func envelope(_ q:String)->Set<Int> {
  let w=words(normalized(q));if w.isEmpty {return []}
  return Set(entries.indices.filter {i in
   if words(normalized(entries[i].name))==w {return true}
   guard keys[i].contains(where:{k in w.contains {k.hasPrefix($0)}}) else {return false}
   let v=vocab(i);return w.subtracting(ignored).allSatisfy {g in v.contains {$0.hasPrefix(g)}}
  })
 }
}
