import Foundation
// Independent normative v4 model. Final I is read from actual baseline via Mirror.
enum Policy {
 static let locale = Locale(identifier:"en_US_POSIX")
 static let ignored:Set<String> = ["a","an","and","for","in","of","on","the","to","with"]
 static func base(_ s:Unicode.Scalar)->Bool {
  guard CharacterSet.letters.contains(s) else {return false}
  switch s.properties.generalCategory {case .uppercaseLetter,.lowercaseLetter,.titlecaseLetter,.modifierLetter,.otherLetter:return true;default:return false}
 }
 static func mark(_ s:Unicode.Scalar)->Bool {switch s.properties.generalCategory {case .nonspacingMark,.spacingMark,.enclosingMark:return true;default:return false}}
 static func folded(_ s:String)->String {s.folding(options:[.caseInsensitive,.diacriticInsensitive],locale:locale)}
 static func exact(_ s:String)->String {folded(s).split(whereSeparator: \.isWhitespace).joined(separator:" ")}
 static func legacy(_ s:String)->String {folded(s).unicodeScalars.map{CharacterSet.alphanumerics.contains($0) ? String($0) : ($0 == "'" || $0 == "\u{2019}" ? "" : " ")}.joined().split(whereSeparator: \.isWhitespace).joined(separator:" ")}
 static func p(_ s:String)->String {String(String.UnicodeScalarView(folded(s).decomposedStringWithCanonicalMapping.unicodeScalars.filter(base)))}
 static func q(_ t:String)->String {p(String(String.UnicodeScalarView(t.decomposedStringWithCanonicalMapping.unicodeScalars.filter{!mark($0)})))}
 static func tokens(_ s:String)->[String] {
  var out:[String]=[],token="",vowel=false
  func emit(){if !token.isEmpty{out.append(token.precomposedStringWithCanonicalMapping)};token="";vowel=false}
  for u in s.lowercased(with:locale).decomposedStringWithCanonicalMapping.unicodeScalars {
   if base(u){token += String(u);vowel="aeiou".unicodeScalars.contains(u)}
   else if mark(u){if !vowel{token += String(u)}}
   else{emit()}
  }
  emit();return Array(Set(out)).sorted()
 }
 static func hasBase(_ tokens:[String])->Bool {tokens.contains{$0.unicodeScalars.contains(where:base)}}
 static func marked(_ tokens:[String])->Bool {tokens.contains{$0.unicodeScalars.first.map(mark) ?? false}}
 static func starts(_ hay:String,_ needle:String)->Bool {Array(hay.unicodeScalars).starts(with:Array(needle.unicodeScalars))}
 static func contains(_ hay:String,_ needle:String)->Bool {
  let h=Array(hay.unicodeScalars),n=Array(needle.unicodeScalars)
  if n.isEmpty{return true};if n.count>h.count{return false}
  return (0...(h.count-n.count)).contains{i in h[i..<(i+n.count)].elementsEqual(n)}
 }
 static let d:Set<String> = {
  var out:Set<String>=[]
  for w in ignored {let u=Array(w.unicodeScalars);for i in u.indices {for j in (i+1)...u.count{out.insert(String(String.UnicodeScalarView(u[i..<j])))}}}
  return out
 }()
 static func required(_ tokens:[String])->Set<String> {Set(tokens.map(q)).subtracting(d).subtracting([""])}
 static func scalars(_ s:String)->[String] {s.unicodeScalars.map{String(format:"U+%04X",$0.value)}}
}
struct Envelope {
 let entries:[Emoji]
 let vocabulary:[[String]]
 let indexed:[[String]]
 let exactNames:[String:Int]
 let rawIndex:[String:[Int]]
 init(_ entries:[Emoji],_ s:EmojiSuggester) {
  self.entries=entries
  vocabulary=entries.map{e in Array(Set(([e.name]+(e.keywords ?? [])).flatMap{raw -> [String] in let n=Policy.legacy(raw),w=n.split(separator:" ").map(String.init);return ([n,w.joined()]+w).map(Policy.p)})).sorted()}
  var terms = [[String]](repeating:[],count:entries.count),ex:[String:Int]=[:],ri:[String:[Int]]=[:]
  for child in Mirror(reflecting:s).children {
   if child.label == "exactNames",let exact=child.value as? [String:Int] {ex=exact;for (k,id) in exact {terms[id].append(k)}}
   if child.label == "index" {for term in Mirror(reflecting:child.value).children {let m=Mirror(reflecting:term.value);let key=m.children.first{$0.label == "key"}!.value as! String;let matches=m.children.first{$0.label == "matches"}!.value;var ids:[Int]=[];for match in Mirror(reflecting:matches).children {let id=Mirror(reflecting:match.value).children.first{$0.label == "id"}!.value as! Int;terms[id].append(key);ids.append(id)};ri[key]=ids}}
  }
  exactNames=ex;rawIndex=ri
  indexed=terms.map{Array(Set($0.map(Policy.p))).sorted()}
 }
 func contains(_ id:Int,_ tokens:[String])->Bool {
  guard !tokens.isEmpty else{return false}
  let a=Set(tokens.map(Policy.q)),r=Policy.required(tokens)
  return r.allSatisfy{t in vocabulary[id].contains{Policy.contains($0,t)}} && (Policy.marked(tokens) || indexed[id].contains{i in a.contains{Policy.starts(i,$0)}})
 }
 func ids(_ tokens:[String])->[Int]{entries.indices.filter{contains($0,tokens)}}
}
func cat(_ es:[Emoji])->EmojiCatalog {EmojiCatalog(version:"fixture",categories:[EmojiCategory(id:"f",name:"Fixture",emoji:es)])}
func entry(_ id:String,_ name:String,_ keys:[String]=[])->Emoji {Emoji(value:id,name:name,version:1,tones:nil,keywords:keys)}
func json(_ object:Any,_ path:String)throws {try JSONSerialization.data(withJSONObject:object,options:[.prettyPrinted,.sortedKeys]).write(to:URL(fileURLWithPath:path))}
