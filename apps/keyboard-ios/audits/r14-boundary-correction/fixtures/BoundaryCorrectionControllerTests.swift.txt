import UIKit
import XCTest
@testable import KeyboardUI

/// R14 protection gates through actual UIKit controller/key routing. Runtime
/// verification requires the parent-owned test device; this sweep only compiles.
@MainActor
final class BoundaryCorrectionControllerTests: XCTestCase {
  private func harness(_ text: String, caret: Int, selection: Int = 0, optimistic: Bool) -> ControllerHarness {
    let harness = ControllerHarness(text: text, optimistic: optimistic, waitsForEmoji: false)
    harness.engine.settings.autoCapitalization = false
    harness.proxy.hostChanges { state in
      state.cursor = caret
      state.selection = selection
    }
    harness.deliver(callbacks: [.textDid, .selectionDid])
    return harness
  }

  func testFragmentsAndPartialSelectionsPreserveTheExistingSuffix() {
    for optimistic in [false, true] {
      for delayed in [false, true] {
        let fragment = harness("use myvalue here", caret: 6, optimistic: optimistic)
        fragment.type("teh ", settling: !delayed)
        fragment.settle()
        XCTAssertEqual(fragment.proxy.host.string, "use myteh value here")

        let selected = harness("use myoldvalue now", caret: 6, selection: 3, optimistic: optimistic)
        selected.type("teh ", settling: !delayed)
        selected.settle()
        XCTAssertEqual(selected.proxy.host.string, "use myteh value now")
      }
    }
  }

  func testRestrictedFieldsKeepTypedLiteralsBeforeExistingText() {
    for optimistic in [false, true] {
      for keyboard in [UIKeyboardType.emailAddress, .URL] {
        let restricted = harness("send tail", caret: 5, optimistic: optimistic)
        restricted.proxy.keyboardTypeValue = keyboard
        restricted.fire([.textDid])
        restricted.type("teh ", settling: false)
        restricted.settle()
        XCTAssertEqual(restricted.proxy.host.string, "send teh tail")
        XCTAssertEqual(restricted.engine.suggester?.model.personal.totalCount, 0)
      }
    }
  }

  func testNumericKeypadInsertsOnlyVisibleDigitsAndConservesTheSuffix() {
    for optimistic in [false, true] {
      let numeric = harness("1299", caret: 2, optimistic: optimistic)
      numeric.proxy.keyboardTypeValue = .numberPad
      numeric.fire([.textDid])
      numeric.type("34", settling: false)
      numeric.settle()
      XCTAssertEqual(numeric.proxy.host.string, "123499")
      XCTAssertEqual(numeric.engine.suggester?.model.personal.totalCount, 0)
    }
  }

  func testBackspaceAndRetypedSpaceConservePrefixSuffixAndOriginalSpelling() throws {
    for optimistic in [false, true] {
      let undo = harness("join meeting", caret: 5, optimistic: optimistic)
      undo.type("teh ", settling: false)
      undo.settle()
      let delete = try XCTUnwrap(undo.controller.rootView.keys.keyViews.first { $0.key.action == .delete })
      XCTAssertTrue(delete.accessibilityActivate())
      undo.settle()
      XCTAssertEqual(undo.proxy.host.string, "join tehmeeting")
      undo.type(" ")
      XCTAssertEqual(undo.proxy.host.string, "join teh meeting")
    }
  }
}
