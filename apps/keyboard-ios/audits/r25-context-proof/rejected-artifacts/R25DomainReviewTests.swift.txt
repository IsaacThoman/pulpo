import Foundation
import Testing

/// Independent domain challenges: synthetic physical edits, no lexical fixtures or engine state.
struct R25DomainReviewTests {
  typealias O = R25ConservationOracle

  struct Host {
    var state: O.Snapshot
    var oracle: O
    init(prefix: String = "👩🏽‍💻 e\u{301} ", suffix: String = "漢字tail") {
      state = .init(text: prefix + suffix, caret: prefix.count)
      oracle = O(actual: state)
    }

    mutating func action(_ action: O.Action, _ primitives: [O.Primitive]) -> O.Verdict {
      let before = state
      var steps: [O.Step] = []
      for primitive in primitives {
        let previous = state
        var left = Array(state.text.prefix(state.caret))
        let right = String(state.text.dropFirst(state.caret + state.selectionLength))
        switch primitive {
        case .deleteBackward:
          if state.selectionLength == 0, !left.isEmpty { left.removeLast() }
        case let .insert(text): left = Array(String(left) + text)
        }
        let text = String(left) + right
        state = .init(text: text, caret: min(left.count, text.count))
        steps.append(.init(primitive: primitive, before: previous, after: state))
      }
      return oracle.observe(.init(action: action, before: before, steps: steps, after: state))
    }

    mutating func tap(_ key: Character, output: String? = nil) {
      #expect(action(.letter(key), [.insert(output ?? String(key))]).failures.isEmpty)
    }

    mutating func rewrite(_ c: String = "qq rr's!", action trigger: O.Action = .explicitSpace) -> O.Verdict {
      let count = oracle.owned?.word.count ?? 2
      return action(trigger, Array(repeating: .deleteBackward, count: count) + [.insert(c), .insert(" ")])
    }
  }

  @Test func casingRelationRemainsIndependentAcrossReportAndModifierHistories() {
    for key: Character in ["q", "Q", "é", "É", "ß", "İ", "Σ"] {
      for output in [String(key), String(key).uppercased(), String(key).lowercased()] {
        for history in 0..<4 {
          var host = Host()
          let inherited = host.oracle.physicalIdentities
          host.tap(key, output: output)
          let ownedIDs = host.oracle.physicalIdentities
          if history > 0 {
            host.oracle.proxyReported(nil)
            host.oracle.proxyReported(.init(text: "unrelated old report", caret: 0))
          }
          if history > 1 {
            #expect(host.action(.modifier, []).failures.isEmpty)
          }
          if history > 2 { host.oracle.proxyReported(host.state) }
          #expect(host.oracle.physicalIdentities == ownedIDs)
          #expect(host.oracle.owned?.word.unicodeScalars.elementsEqual(output.unicodeScalars) == true)
          let verdict = host.rewrite()
          #expect(verdict.permitsSpaceReplacement)
          #expect(host.oracle.physicalIdentities.suffix(inherited.count - "👩🏽‍💻 e\u{301} ".count)
            == inherited.suffix(inherited.count - "👩🏽‍💻 e\u{301} ".count))
        }
      }
    }
  }

  @Test func unrelatedOutputCannotCreateOwnership() {
    for output in ["r", "qq", "Qq", "q ", "q\u{301}"] {
      var host = Host()
      host.tap("q", output: output)
      #expect(host.oracle.owned == nil)
      #expect(!host.rewrite().permitsSpaceReplacement)
    }
  }

  @Test func wrongTriggersNeverReceiveReplacementPermission() {
    for trigger: O.Action in [.punctuation, .returnKey, .swipe, .insertText, .hostSync, .other, .modifier] {
      var host = Host()
      host.tap("q"); host.tap("z")
      let verdict = host.rewrite(action: trigger)
      #expect(!verdict.permitsSpaceReplacement)
      #expect(!verdict.failures.isEmpty)
    }
  }

  @Test func identicalHostReplacementAndSelectionCannotResurrectIDs() {
    for selected in [false, true] {
      var host = Host()
      host.tap("q"); host.tap("z")
      let previous = host.state
      let previousIDs = host.oracle.physicalIdentities
      if selected { host.state.selectionLength = 1 }
      host.oracle.hostChanged(to: host.state)
      #expect(Set(previousIDs).isDisjoint(with: host.oracle.physicalIdentities))
      host.oracle.proxyReported(previous)
      #expect(host.oracle.owned == nil)
      #expect(!host.rewrite().permitsSpaceReplacement)
    }
  }

  @Test func unreportedObservableMoveFailsContinuityEvenAtRepeatedText() {
    var host = Host(prefix: "p ", suffix: "tail p qztail")
    host.tap("q"); host.tap("z")
    host.state.caret = "p qztail p qz".count
    let verdict = host.rewrite()
    #expect(!verdict.permitsSpaceReplacement)
    #expect(verdict.failures.contains("actual document discontinuity"))
  }

  @Test func transientDeletionAndSameTextRepairCannotRecoverInheritedIdentity() {
    var host = Host()
    host.tap("q"); host.tap("z")
    let inheritedIDs = host.oracle.physicalIdentities
    // Delete the owned word AND inherited whitespace, then repair the exact final text.
    let verdict = host.action(.explicitSpace, [.deleteBackward, .deleteBackward, .deleteBackward, .insert(" qq ")])
    #expect(!verdict.permitsSpaceReplacement)
    #expect(verdict.failures.contains { $0.contains("identity") })
    #expect(host.oracle.physicalIdentities != inheritedIDs)
    #expect(O.sameText(host.state.text, "👩🏽‍💻 e\u{301} qq 漢字tail"))
  }

  @Test func exactMultiwordUndoAndWrongSeparatorBoundary() {
    for retainsSeparator in [false, true] {
      var host = Host()
      host.tap("q"); host.tap("z")
      let before = host.state
      let c = "qq rr's!"
      #expect(host.rewrite(c).permitsSpaceReplacement)
      let verdict = host.action(.backspace,
        Array(repeating: .deleteBackward, count: c.count + 1) + [.insert(retainsSeparator ? "qz " : "qz")])
      if retainsSeparator {
        #expect(!verdict.failures.isEmpty)
        #expect(verdict.exception != .exactUndo)
      } else {
        #expect(verdict.failures.isEmpty)
        #expect(verdict.exception == .exactUndo)
        #expect(host.state == before)
        #expect(host.oracle.owned == nil)
      }
    }
  }

  @Test func inheritedGraphemeBoundaryCannotBecomeOwnedThroughCaseMapping() {
    // Prepend letter can join the first inherited combining grapheme after insertion.
    var host = Host(prefix: "", suffix: "\u{301}tail")
    let before = host.oracle.physicalIdentities
    host.tap("q")
    #expect(host.oracle.owned == nil)
    #expect(host.oracle.physicalIdentities != before)
    #expect(!host.rewrite().permitsSpaceReplacement)
  }

  @Test func frozenSchemaRequiresAnExpectedCaret() throws {
    let missingCaret = Data(#"[{"id":"domain-missing-caret","language":"en","category":"schema","intent":"synthetic negative","initial":"","caret":0,"typed":"","expected":""}]"#.utf8)
    // The immutable contract marks expectedCaret REQUIRED. An absent value must
    // not become a wildcard in an otherwise successful full-document score.
    #expect(throws: (any Error).self) {
      try JSONDecoder().decode([R25ExplicitSpaceSweepTests.Case].self, from: missingCaret)
    }
  }

  @Test func ordinaryDoubleSpacePeriodIsNotAFreshWordReplacement() {
    var host = Host(prefix: "p ", suffix: "tail")
    let inherited = host.oracle.physicalIdentities
    host.tap("q"); host.tap("z")
    #expect(host.action(.explicitSpace, [.insert(" ")]).failures.isEmpty)
    // The ordinary second-Space path removes only its freshly inserted separator.
    // No letter or inherited character is deleted; existing spacing policy stays.
    let verdict = host.action(.explicitSpace, [.deleteBackward, .insert(". ")])
    #expect(host.state == .init(text: "p qz. tail", caret: 6))
    #expect(Array(host.oracle.physicalIdentities.prefix(2)) == Array(inherited.prefix(2)))
    #expect(Array(host.oracle.physicalIdentities.suffix(4)) == Array(inherited.suffix(4)))
    #expect(verdict.exception == .none)
    #expect(verdict.failures.isEmpty)
  }
}
