import Foundation

@main
struct R38BehaviorMain {
  @MainActor
  static func main() {
    do {
      let arguments = CommandLine.arguments
      guard arguments.count == 5 || arguments.count == 6 else {
        throw R38Error("Usage: r38-behavior CASES.json CALIBRATION.json LANGUAGE_DATA_DIRECTORY OUTPUT.json [baseline|candidate|development]")
      }
      let runner = try R38BehaviorRunner(calibrationURL: URL(fileURLWithPath: arguments[2]),
                                         languageDataURL: URL(fileURLWithPath: arguments[3], isDirectory: true))
      let output = try runner.run(casesData: Data(contentsOf: URL(fileURLWithPath: arguments[1])),
                                  label: arguments.count == 6 ? arguments[5] : "development")
      try output.write(to: URL(fileURLWithPath: arguments[4]), options: .atomic)
      let object = try JSONSerialization.jsonObject(with: output) as! [String: Any]
      let count = object["failureCount"] as! Int
      print("R38 \(R38BehaviorRunner.platform): \(count) failure(s); \(arguments[4])")
      if count > 0 { exit(1) }
    } catch {
      // Write machine-readable startup/schema failures too. CLI misuse without
      // an output argument can only be reported on stderr.
      if CommandLine.arguments.count >= 5,
         let output = try? JSONSerialization.data(withJSONObject: [
           "version": 1, "runner": "R38-development-behavior", "passed": false,
           "failureCount": 1, "failures": [String(describing: error)], "results": [],
         ], options: [.prettyPrinted, .sortedKeys]) {
        try? output.write(to: URL(fileURLWithPath: CommandLine.arguments[4]), options: .atomic)
      }
      FileHandle.standardError.write(Data("R38: \(error)\n".utf8))
      exit(2)
    }
  }
}
