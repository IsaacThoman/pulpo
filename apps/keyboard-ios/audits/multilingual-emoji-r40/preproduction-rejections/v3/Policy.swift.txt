import Foundation
// Independent executable contract model. Not production code.
enum Policy {
 static let locale = Locale(identifier:"en_US_POSIX")
 static let ignored:Set<String> = ["a","an","and","for","in","of","on","the","to","with"]
 static func base(_ s:Unicode.Scalar)->Bool {
  guard CharacterSet.letters.contains(s) else{return false}
  switch s.properties.generalCategory {case .uppercaseLetter,.lowercaseLetter,.titlecaseLetter,.modifierLetter,.otherLetter:return true;default:return false}
 }
 static func mark(_ s:Unicode.Scalar)->Bool {switch s.properties.generalCategory {case .nonspacingMark,.spacingMark,.enclosingMark:return true;default:return false}}
 static func folded(_ s:String)->String {s.folding(options:[.caseInsensitive,.diacriticInsensitive],locale:locale)}
 static func exact(_ s:String)->String {folded(s).split(whereSeparator: \.isWhitespace).joined(separator:" ")}
 static func legacy(_ s:String)->String {folded(s).unicodeScalars.map{CharacterSet.alphanumerics.contains($0) ? String($0) : ($0 == "'" || $0 == "\u{2019}" ? "" : " ")}.joined().split(whereSeparator: \.isWhitespace).joined(separator:" ")}
 static func project(_ s:String)->String {String(String.UnicodeScalarView(folded(s).decomposedStringWithCanonicalMapping.unicodeScalars.filter(base)))}
 static func tokens(_ s:String, mode:Int = 3)->[String] {
  var out:[String]=[],token="",vowel=false,hasBase=false
  func emit(){if !token.isEmpty && (mode == 3 || hasBase){out.append(token.precomposedStringWithCanonicalMapping)};token="";vowel=false;hasBase=false}
  for u in s.lowercased(with:locale).decomposedStringWithCanonicalMapping.unicodeScalars {
   if base(u){token += String(u);hasBase=true;vowel="aeiou".unicodeScalars.contains(u)}
   else if mark(u){if !vowel && (mode != 1 || hasBase){token += String(u)}}
   else{emit()}
  }
  emit();return Array(Set(out)).sorted()
 }
 static func hasBase(_ tokens:[String])->Bool {tokens.contains{$0.unicodeScalars.contains(where:base)}}
}
struct Envelope {
 let entries:[Emoji]
 let vocabulary:[[String]]
 let indexed:[[String]]
 init(_ entries:[Emoji]) {
  self.entries=entries
  var postings:[String:[Int:Int]]=[:],exact:[String:Int]=[:]
  var vocab = [[String]](repeating:[],count:entries.count)
  func add(_ term:String,_ id:Int,_ rank:Int){guard (3...80).contains(term.count),!Policy.ignored.contains(term) else{return};postings[term,default:[:]][id]=min(postings[term]?[id] ?? Int.max,rank)}
  for (id,e) in entries.enumerated(){
   let f=Policy.exact(e.name);if exact[f] == nil {exact[f]=id}
   for raw in [e.name]+(e.keywords ?? []) {let n=Policy.legacy(raw),w=n.split(separator:" ").map(String.init);vocab[id] += [n,w.joined()]+w}
   let name=Policy.legacy(e.name),w=name.split(separator:" ").map(String.init)
   add(name,id,0)
   if w.count>1 {add(w.joined(),id,10);for start in w.indices{add(w[start],id,40+w.count);if start>0{add(w[start...].joined(separator:" "),id,20+start)}}}
   for keyword in e.keywords ?? []{let term=Policy.legacy(keyword);add(term,id,80);if term.contains(" "){add(term.replacingOccurrences(of:" ",with:""),id,90)}}
  }
  var terms = [[String]](repeating:[],count:entries.count)
  for (key,posts) in postings {for id in posts.keys.sorted(by:{posts[$0]! != posts[$1]! ? posts[$0]! < posts[$1]! : $0 < $1}).prefix(12){terms[id].append(key)}}
  for (key,id) in exact{terms[id].append(key)}
  self.vocabulary=vocab.map{Array(Set($0.map(Policy.project))).sorted()}
  self.indexed=terms.map{Array(Set($0.map(Policy.project))).sorted()}
 }
 func contains(_ id:Int,_ tokens:[String])->Bool {
  guard !tokens.isEmpty else{return false}
  let a=Set(tokens.map(Policy.project)),r=a.subtracting(Policy.ignored)
  return r.allSatisfy{t in t.isEmpty || vocabulary[id].contains{$0.contains(t)}} && indexed[id].contains{i in a.contains{t in t.isEmpty || i.hasPrefix(t)}}
 }
 func ids(_ tokens:[String])->[Int]{entries.indices.filter{contains($0,tokens)}}
}
