import UIKit
import XCTest
import KeyboardUI
@testable import PulpoKeyboard

/// R23 informed ownership controls. No product source intervention.
@MainActor
final class PreviewConsumerOwnershipTests: XCTestCase {
  // Baseline host-storage comparison only: keep its unowned adapter valid for
  // the process lifetime, including asynchronous UIKit/controller callbacks.
  // This separate control never satisfies the standalone host-release contract.
  private static var comparisonHosts: [PreviewHost] = []
  private final class Probes {
    weak var view: UIView?
    weak var controller: KeyboardController?
    weak var engine: KeyboardEngine?
    weak var host: PreviewHost?
  }

  func testPlainViewReleases() {
    let probes = Probes()
    makeUIKitView(probes, input: false, selfSizing: false)
    assertReleased(probes)
  }

  func testPlainInputViewReleases() {
    let probes = Probes()
    makeUIKitView(probes, input: true, selfSizing: false)
    assertReleased(probes)
  }

  func testSelfSizingInputViewReleases() {
    let probes = Probes()
    makeUIKitView(probes, input: true, selfSizing: true)
    assertReleased(probes)
  }

  func testPlainDefaultStyleInputViewReleases() {
    let probes = Probes()
    autoreleasepool {
      let view = UIInputView(frame: .zero, inputViewStyle: .default)
      view.addSubview(UIView())
      probes.view = view
    }
    assertReleased(probes)
  }

  func testResponderHostedInputViewReleases() throws {
    let probes = Probes()
    let editor = UITextView()
    try makeResponderConsumer(probes, editor: editor, preview: false)
    assertReleased(probes)
  }

  func testResponderHostedPreviewInputReleases() throws {
    let probes = Probes()
    let editor = UITextView()
    let document = try makeResponderConsumer(probes, editor: editor, preview: true)
    assertReleased(probes)
    withExtendedLifetime((document, editor)) {}
  }

  func testControllerWithoutInputReleases() {
    let probes = Probes()
    let editor = UITextView()
    let document = makeConsumer(probes, editor: editor, input: false)
    assertReleased(probes)
    withExtendedLifetime((document, editor)) {}
  }

  func testUnpresentedPreviewInputReleases() {
    let probes = Probes()
    let editor = UITextView()
    let document = makeConsumer(probes, editor: editor, input: true)
    assertReleased(probes)
    withExtendedLifetime((document, editor)) {}
  }

  func testPreviewInputReleaseWithHostKeptAliveForBaselineComparison() {
    let editor = UITextView()
    let host = PreviewHost(textView: editor, openURL: { _ in })
    Self.comparisonHosts.append(host)
    let probes = Probes()
    makeLivingHostConsumer(probes, host: host)
    assertReleased(probes)
    withExtendedLifetime((host, editor)) {
      XCTAssertTrue(Self.comparisonHosts.last === host)
    }
  }

  @inline(never)
  private func makeLivingHostConsumer(_ probes: Probes, host: PreviewHost) {
    autoreleasepool {
      let controller = KeyboardController(host: host, dictationBridge: nil)
      let view = PreviewInputView(controller: controller)
      probes.view = view
      probes.controller = controller
      probes.engine = controller.engine
    }
  }

  @inline(never)
  private func makeUIKitView(_ probes: Probes, input: Bool, selfSizing: Bool) {
    autoreleasepool {
      let view: UIView
      if input {
        let keyboard = UIInputView(frame: CGRect(x: 0, y: 0, width: 400, height: 300), inputViewStyle: .keyboard)
        keyboard.allowsSelfSizing = selfSizing
        view = keyboard
      } else {
        view = UIView()
      }
      view.addSubview(UIView())
      probes.view = view
    }
  }

  @inline(never)
  private func makeConsumer(_ probes: Probes, editor: UITextView, input: Bool) -> TextDocument {
    autoreleasepool {
      let host = PreviewHost(textView: editor, openURL: { _ in })
      let controller = KeyboardController(host: host, dictationBridge: nil)
      probes.host = host
      probes.controller = controller
      probes.engine = controller.engine
      if input {
        let view = PreviewInputView(controller: controller)
        probes.view = view
        return withExtendedLifetime(view) { controller.engine.document }
      }
      return controller.engine.document
    }
  }

  @inline(never)
  private func makeResponderConsumer(_ probes: Probes, editor: UITextView, preview: Bool) throws -> TextDocument? {
    let scene = try XCTUnwrap(UIApplication.shared.connectedScenes.compactMap { $0 as? UIWindowScene }.first)
    return autoreleasepool {
      let window = UIWindow(windowScene: scene)
      window.rootViewController = UIViewController()
      editor.frame = CGRect(x: 10, y: 50, width: 300, height: 100)
      window.rootViewController!.view.addSubview(editor)
      let view: UIInputView
      var document: TextDocument?
      if preview {
        let host = PreviewHost(textView: editor, openURL: { _ in })
        let controller = KeyboardController(host: host, dictationBridge: nil)
        view = PreviewInputView(controller: controller)
        probes.host = host
        probes.controller = controller
        probes.engine = controller.engine
        document = controller.engine.document
      } else {
        view = UIInputView(frame: CGRect(x: 0, y: 0, width: 400, height: 300), inputViewStyle: .keyboard)
        view.addSubview(UIView())
      }
      editor.inputView = view
      probes.view = view
      window.makeKeyAndVisible()
      XCTAssertTrue(editor.becomeFirstResponder())
      RunLoop.main.run(until: Date().addingTimeInterval(0.2))
      XCTAssertTrue(editor.isFirstResponder)
      editor.inputView = nil
      editor.reloadInputViews()
      XCTAssertTrue(editor.resignFirstResponder())
      RunLoop.main.run(until: Date().addingTimeInterval(0.2))
      editor.removeFromSuperview()
      window.isHidden = true
      window.rootViewController = nil
      return document
    }
  }

  @inline(never)
  private func assertReleased(_ probes: Probes, file: StaticString = #filePath, line: UInt = #line) {
    let deadline = ProcessInfo.processInfo.systemUptime + 2
    while autoreleasepool(invoking: { probes.view != nil || probes.controller != nil || probes.engine != nil || probes.host != nil }),
          ProcessInfo.processInfo.systemUptime < deadline {
      autoreleasepool { RunLoop.main.run(until: Date().addingTimeInterval(0.01)) }
    }
    XCTAssertNil(probes.view, file: file, line: line)
    XCTAssertNil(probes.controller, file: file, line: line)
    XCTAssertNil(probes.engine, file: file, line: line)
    XCTAssertNil(probes.host, file: file, line: line)
  }

}
