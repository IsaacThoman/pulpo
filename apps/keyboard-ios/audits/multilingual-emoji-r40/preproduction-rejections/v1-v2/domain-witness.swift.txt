import Foundation

// Independent PREPRODUCTION contract controls. No candidate implementation.
@main struct DomainReview {
  struct E { let value:String; let english:String; var keywords:[String]=[]; var spanish:String?=nil }
  struct Source {
    let entries:[E]
    let search:EmojiSuggester
    let projectedVocabulary:[[String]]
    let projectedIndexed:[[String]]
    init(_ entries:[E]) {
      self.entries=entries
      let emojis=entries.map{Emoji(value:$0.value,name:$0.english,version:1,tones:nil,keywords:$0.keywords)}
      search=EmojiSuggester(catalog:EmojiCatalog(version:"independent",categories:[EmojiCategory(id:"x",name:"x",emoji:emojis)]))
      projectedVocabulary=entries.map{e in Array(Set(([e.english]+e.keywords).flatMap{ t -> [String] in
        let n=Policy.legacy(t), w=n.split(separator:" ").map(String.init)
        return [n,w.joined()]+w
      }.map(Policy.project)))}
      // Inspect the actual baseline's final bounded postings. This reads private
      // fields via Mirror in a standalone diagnostic; it does not alter source.
      var indexed=Array(repeating:Set<String>(),count:entries.count)
      let fields=Dictionary(uniqueKeysWithValues:Mirror(reflecting:search).children.compactMap{child -> (String,Any)? in
        guard let k=child.label else{return nil};return(k,child.value)
      })
      for child in Mirror(reflecting:fields["index"]!).children {
        let term=Dictionary(uniqueKeysWithValues:Mirror(reflecting:child.value).children.map{($0.label!,$0.value)})
        let key=term["key"] as! String
        for m in Mirror(reflecting:term["matches"]!).children {
          let fields=Dictionary(uniqueKeysWithValues:Mirror(reflecting:m.value).children.map{($0.label!,$0.value)})
          indexed[fields["id"] as! Int].insert(Policy.project(key))
        }
      }
      for (key,id) in fields["exactNames"] as! [String:Int] {indexed[id].insert(Policy.project(key))}
      projectedIndexed=indexed.map{Array($0).sorted()}
    }
    func envelope(_ tokens:[String])->[String] {
      let a=Set(tokens.map(Policy.project)),r=a.subtracting(DomainReview.ignored)
      return entries.indices.filter{idx in
        r.allSatisfy{t in projectedVocabulary[idx].contains{$0.contains(t)}} &&
        a.contains{t in projectedIndexed[idx].contains{$0.hasPrefix(t)}}
      }.map{entries[$0].value}
    }
  }
  static let ignored:Set<String>=["a","an","and","for","in","of","on","the","to","with"]
  // Precisely the proposed general v2: keep pending leading marks only when a
  // base arrives before a separator/end; otherwise discard pending runs.
  static func canonicalV2(_ text:String)->[String] {
    var words:[String]=[],token="",pending="",vowel=false,hasBase=false
    func flush() {
      if hasBase {words.append(token.precomposedStringWithCanonicalMapping)}
      token="";pending="";vowel=false;hasBase=false
    }
    for s in text.lowercased(with:Policy.locale).decomposedStringWithCanonicalMapping.unicodeScalars {
      if Policy.mark(s) {
        if !hasBase {pending.unicodeScalars.append(s)}
        else if !vowel {token.unicodeScalars.append(s)}
      } else if Policy.base(s) {
        if !hasBase {token=pending;pending=""}
        token.unicodeScalars.append(s);hasBase=true;vowel="aeiou".unicodeScalars.contains(s)
      } else {flush()}
    }
    flush();return Array(Set(words)).sorted()
  }
  static func localized(_ source:Source,_ tokens:[String],_ canon:(String)->[String])->[String] {
    guard !tokens.isEmpty else{return []}
    if tokens.count==1,tokens[0].utf8.count==1{return []}
    var rank:[(Int,Int,Int,Int)]=[]
    for (i,e) in source.entries.enumerated() {
      let es=canon(e.spanish ?? "")
      let en=([e.english]+e.keywords).flatMap{t -> [String] in
        let n=Policy.legacy(t),w=n.split(separator:" ").map(String.init)
        return [n,w.joined()]+w
      }
      guard tokens.contains(where:{t in es.contains{$0.hasPrefix(t)}}),
        tokens.allSatisfy({t in es.contains{$0.hasPrefix(t)} || en.contains{$0.hasPrefix(Policy.legacy(t))}}) else{continue}
      let completion=tokens.filter{t in !es.contains(t) && !en.contains(Policy.legacy(t))}.count
      rank.append((es==tokens ? 0:1,completion,es.count,i))
    }
    return rank.sorted{$0 < $1}.map{source.entries[$0.3].value}
  }
  static func main() throws {
    var rows:[[String:Any]]=[],groups:[[String:Any]]=[]
    var executed=0,subsetFailures=0,capFailures=0,positiveControls=0
    let prefs:[[String:Int]]=[[:],["E0":2,"E79":2,"L0":2]]
    func inspect(_ source:Source,_ queries:[String],_ label:String,_ version:Int,_ expectComplete:Bool) {
      let canon:(String)->[String] = version==1 ? Policy.canonical:canonicalV2
      let first=canon(queries[0]),env=source.envelope(first),eligible=localized(source,first,canon)
      precondition(queries.allSatisfy{canon($0)==first},"unexpected token mismatch \(label)")
      for cap in [1,48,60] {
        let reserved=Set(env),tranche=Array(eligible.filter{!reserved.contains($0)}.prefix(max(0,cap-min(cap,env.count))))
        for (p,preferred) in prefs.enumerated() {
          for q in queries {
            let measured=source.search.search(q,limit:cap,preferring:preferred).map(\.value)
            let complete=Set(measured).isSubset(of:reserved),merged=measured+tranche
            if !complete{subsetFailures += 1};if merged.count>cap{capFailures += 1}
            if expectComplete {precondition(complete && merged.count<=cap && Set(merged).count==merged.count);positiveControls += 1}
            executed += 1
            rows.append(["control":label,"version":version,"query":q,"scalars":q.unicodeScalars.map{String(format:"U+%04X",$0.value)},"canonical":first,"A":Array(Set(first.map(Policy.project))).sorted(),"cap":cap,"preferenceMap":p,"legacyKey":Policy.legacy(q),"measuredBaseline":measured,"draftEnvelope":env,"calculatedLocalizedTranche":tranche,"calculatedMerge":merged,"legacySubsetEnvelope":complete,"withinCap":merged.count<=cap,"calculatedOnly":true])
          }
        }
      }
      groups.append(["control":label,"version":version,"queries":queries,"canonical":first,"entries":source.entries.count,"envelopeCount":env.count,"eligibleLocalizedCount":eligible.count,"allProjectedIndexTerms":source.projectedIndexed])
    }
    // Full-capacity impossibility. Each English entry has a distinct indexed
    // name, avoiding the 12-posting ceiling of a shared term.
    let saturated=Source((0..<80).map{E(value:"E\($0)",english:"ιnfixture"+String(Unicode.Scalar(97+$0/26)!)+String(Unicode.Scalar(97+$0%26)!))}
      + (0..<80).map{E(value:"L\($0)",english:"opaque",spanish:"ñavoru")})
    precondition(saturated.projectedIndexed.flatMap{$0}.allSatisfy{!$0.hasPrefix("n")})
    inspect(saturated,["ñ","Ñ","n\u{0303}","\u{0345}ñ","\u{0345}'ñ","\u{0345}’ñ"],"v1-full-capacity-impossibility",1,false)
    inspect(saturated,["ñ","Ñ","n\u{0303}","\u{0345}'ñ","\u{0345}’ñ","\u{0345}''n\u{0303}"],"v2-full-capacity-apostrophe-impossibility",2,false)
    // v2 also fails when a leading mark-only run is separated by an allowed
    // hyphen/space. Short English words never enter individual postings.
    let short=Source([E(value:"E0",english:"ι n"),E(value:"L0",english:"opaque",spanish:"ñavoru")])
    precondition(!short.projectedIndexed[0].contains(where:{$0.hasPrefix("n")}))
    inspect(short,["ñ","\u{0345}-ñ","\u{0345} ñ"],"v2-short-word-phrase-impossibility",2,false)
    // v2 fixes immediate leading marks, proving the failing apostrophe test is
    // checking the proposed rule rather than accidentally retaining v1.
    precondition(canonicalV2("\u{0345}ñ") != canonicalV2("ñ"))
    precondition(canonicalV2("\u{0345}'ñ") == canonicalV2("ñ"))
    // Existing v5 ñ fixture is repaired by the intended (but incomplete) rule.
    let internalOnly=Source((0..<80).map{E(value:"L\($0)",english:"copper lantern",keywords:["beacon"],spanish:"ñavoru")})
    inspect(internalOnly,["ñ","Ñ","n\u{0303}","ñ ñ"],"ordinary-internal-only-capacity",1,true)
    inspect(internalOnly,["ñ","Ñ","n\u{0303}","ñ ñ"],"ordinary-internal-only-capacity",2,true)
    // Marks, ignored-word anchors, apostrophe joins, exact technical names,
    // reordering and saturation are development controls, never utility rows.
    for (name,queries) in [
      ("foo'bar",["foo'bar","foo bar","bar foo","foo'bar foo'bar"]),
      ("foo’bar",["foo’bar","bar foo","foo bar"]),
      ("the foobar",["the foo","foo the","the foo foo"]),
      ("keycap: #",["keycap: #","keycap","keycap keycap"]),
      ("a\u{093E}b",["a\u{093E}b","ab","ab ab"]),
      ("n\u{093E}b",["n\u{093E}b","n\u{093E}b n\u{093E}b"]),
      ("a\u{20DD}b",["a\u{20DD}b","ab","ab ab"]),
      ("n\u{20DD}b",["n\u{20DD}b","n\u{20DD}b n\u{20DD}b"]),
      ("n\u{034F}b",["n\u{034F}b","n\u{034F}b n\u{034F}b"]),
      ("a\u{105C0}b extra",["a\u{105C0}b extra","extra b a","a b extra a"])
    ] {
      let source=Source([E(value:"E0",english:name),E(value:"L0",english:"opaque",spanish:Policy.canonical(name).joined(separator:" "))])
      for version in [1,2] {inspect(source,queries,"positive-"+name,version,true)}
    }
    // Literal v1 capacity statement must refer to canonical non-ASCII tokens:
    // admitted raw á canonicalizes ASCII a, invoking mandatory legacy-only.
    precondition(Policy.base("á".unicodeScalars.first!))
    precondition(Policy.canonical("á")==["a"])
    precondition(localized(internalOnly,["a"],Policy.canonical).isEmpty)
    let output:[String:Any] = ["scope":"Independent contract controls linked to unchanged baseline, no candidate, utility, costs or native evidence", "versionsReviewed":["exact draft v1","proposed general leading-mark v2"],"executedRows":executed,"legacySubsetEnvelopeFailures":subsetFailures,"calculatedCapViolations":capFailures,"positiveControlRows":positiveControls,"groups":groups,"rows":rows]
    try JSONSerialization.data(withJSONObject:output,options:[.prettyPrinted,.sortedKeys]).write(to:URL(fileURLWithPath:"domain-results.json"))
    print("executed \(executed): \(subsetFailures) incomplete-envelope rows, \(capFailures) calculated cap violations, \(positiveControls) positive controls passed")
  }
}
