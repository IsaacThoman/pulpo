#!/usr/bin/env python3
import datetime,hashlib,json,pathlib,subprocess,sys
from importlib.machinery import SourceFileLoader
here=pathlib.Path(__file__).resolve().parent
check=SourceFileLoader('r47check',str(here/'check.py')).load_module()
input,baseline,candidate,resources,out=map(pathlib.Path,sys.argv[1:])
out.mkdir(parents=True,exist_ok=False)
check.preflight(json.loads(input.read_text()))
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
receipt={'input':sha(input),'executables':{'baseline':sha(baseline),'candidate':sha(candidate)},'resources':{p.name:sha(p) for p in sorted(resources.glob('*.pkdict'))},'process':{}}
for name,binary in [('baseline',baseline),('candidate',candidate)]:
 start=datetime.datetime.now(datetime.timezone.utc).isoformat()
 r=subprocess.run([str(binary.resolve()),str(input.resolve()),str(resources.resolve()),str((out/(name+'.json')).resolve())],capture_output=True,text=True)
 receipt['process'][name]={'exit':r.returncode,'stderr':r.stderr,'stdout':r.stdout,'started':start,'finished':datetime.datetime.now(datetime.timezone.utc).isoformat()}
(out/'receipt.json').write_text(json.dumps(receipt,indent=2)+'\n')
(out/'process.json').write_text(json.dumps(receipt['process'],indent=2)+'\n')
r=subprocess.run([sys.executable,str(here/'check.py'),str(input),str(out/'baseline.json'),str(out/'candidate.json'),str(out/'process.json'),str(out/'report.json')])
print(json.dumps({k:v for k,v in json.loads((out/'report.json').read_text()).items() if k not in ('allChecks','failures')}))
sys.exit(r.returncode)
