#!/usr/bin/env python3
"""Frozen native executor orchestration: serial, UUID pinned, no successful log reads."""
import argparse,hashlib,json,pathlib,subprocess,sys,shutil,time
ROOT=pathlib.Path('/tmp/pulpo-r43-v1')
DEVICE='8E0314EC-CDF8-4606-895F-2C084B98E0F1'
def sha(p): return hashlib.sha256(pathlib.Path(p).read_bytes()).hexdigest()
def invoke(args,log):
 with open(log,'w') as f: p=subprocess.run(args,stdout=f,stderr=subprocess.STDOUT)
 print(json.dumps({'command':args[0:3], 'stage':str(log),'exit':p.returncode}),flush=True)
 return p.returncode
def base_args(variant):
 return ['xcodebuild','-project',str(ROOT/('native-'+variant)/'R43Native.xcodeproj'),'-scheme','R43Native','-destination','platform=iOS Simulator,id='+DEVICE,'-derivedDataPath',str(ROOT/('DerivedData-'+variant)),'-parallel-testing-enabled','NO','-collect-test-diagnostics','never']
def build():
 for variant in ['baseline','candidate']:
  if invoke(base_args(variant)+['build-for-testing'],ROOT/('build-'+variant+'.log')): return 1
 return 0
def evaluate():
 inputs=ROOT/'fresh-input.json'; refs=ROOT/'references.json'
 result=subprocess.run([sys.executable,str(ROOT/'oracle.py'),'validate',str(inputs),str(refs)],capture_output=True,text=True)
 (ROOT/'input-validation.json').write_text(result.stdout)
 if result.returncode: print(result.stdout,flush=True); return 1
 manifest=json.loads((ROOT/'source-manifest.json').read_text())
 for v in ['baseline','candidate']:
  tree=ROOT/'snapshots'/v/'apps/keyboard-ios/PulpoKeyboardKit'
  for name,h in manifest[v].items():
   if sha(tree/name)!=h: raise RuntimeError('source hash changed '+v+'/'+name)
  if sha(ROOT/('native-'+v)/'R43Executor.swift')!=manifest['driver']['R43Executor.swift']: raise RuntimeError('driver mismatch')
 receipt={'frozenAt':time.time(),'device':DEVICE,'sourceManifest':sha(ROOT/'source-manifest.json'),'driver':sha(ROOT/'native/R43Executor.swift'),'orchestrator':sha(__file__),'input':sha(inputs),'references':sha(refs),'oracle':sha(ROOT/'oracle.py'),'variants':['baseline','candidate'],'rerunPolicy':'one full paired evaluation, no gold/source changes after reveal'}
 freeze=ROOT/'evaluation-freeze.json'
 with freeze.open('x') as f: json.dump(receipt,f,indent=2)
 rows=json.loads(inputs.read_text()); locales=list(dict.fromkeys(r['locale'] for r in rows)); status=[]
 for v in ['baseline','candidate']:
  found={}; dicts=None
  for locale in locales:
   selected=[r for r in rows if r['locale']==locale]
   pathlib.Path('/tmp/r43-input.json').write_text(json.dumps(selected,ensure_ascii=False))
   pathlib.Path('/tmp/r43-output.json').unlink(missing_ok=True)
   lang,region=locale.split('_')
   bundle=ROOT/(v+'-'+locale+'.xcresult')
   code=invoke(base_args(v)+['test-without-building','-only-testing:R43Tests/R43Executor/testExecute','-resultBundlePath',str(bundle),'-testLanguage',lang,'-testRegion',region],ROOT/(v+'-'+locale+'.log'))
   summary=ROOT/(v+'-'+locale+'-summary.json')
   with summary.open('w') as f: subprocess.run(['xcrun','xcresulttool','get','test-results','summary','--path',str(bundle),'--format','json'],stdout=f,check=True)
   raw=pathlib.Path('/tmp/r43-output.json')
   if code or not raw.exists(): raise RuntimeError('native evaluation unavailable '+v+' '+locale+'; preserve receipt and fail, no automatic rerun')
   saved=ROOT/(v+'-'+locale+'-raw.json'); shutil.copy2(raw,saved)
   data=json.loads(saved.read_text()); dicts=data['dictionaries'] if dicts is None else dicts
   if dicts!=data['dictionaries']: raise RuntimeError('dictionary changed between groups')
   for r in data['rows']:
    if r['id'] in found: raise RuntimeError('duplicate output '+r['id'])
    found[r['id']]=r
   status.append({'variant':v,'locale':locale,'rows':len(selected),'rawSHA256':sha(saved),'testExit':code})
  out={'schemaVersion':1,'dictionaries':dicts,'rows':[found.get(r['id']) for r in rows]}
  output=ROOT/(v+'-output.json'); output.write_text(json.dumps(out,ensure_ascii=False,indent=2))
  scored=subprocess.run([sys.executable,str(ROOT/'oracle.py'),'score',str(inputs),str(refs),str(output)],capture_output=True,text=True)
  (ROOT/(v+'-verdict.json')).write_text(scored.stdout)
  verdict=json.loads(scored.stdout); print(json.dumps({'variant':v,'oracleExit':scored.returncode,'summary':verdict.get('summary'),'errors':verdict.get('errors')}),flush=True)
 (ROOT/'evaluation-receipt.json').write_text(json.dumps(status,indent=2))
 return 0
if __name__=='__main__': sys.exit(build() if sys.argv[1:] == ['build'] else evaluate())
