#!/usr/bin/env python3
"""Frozen independent R43 literal-protocol validator and fail-closed scorer.
Reads only named command-line JSON artifacts. XCTest exit status is never evidence.
"""
import ctypes
import ctypes.util
import hashlib
import json
import math
import re
import sys

LOCALES = {'en_US':'US', 'es_ES':'ES', 'es_MX':'MX', 'ar_EG':'EG'}
LANGUAGES = {'en_US', 'es', 'en_es'}
KEYBOARDS = {'decimal', 'number', 'asciiNumber', 'standard', 'email', 'url'}
NUMERIC = {'decimal', 'number', 'asciiNumber'}
DOCKS = {'off', 'left', 'right'}
WIDTHS = {320, 375, 393, 430, 852}
KEY_ACTIONS = {'key-numbers', 'key-symbols', 'key-letters', 'key-space', 'key-return'}
FIELDS = {'id','locale','language','keyboard','globe','width','oneHanded','privateField',
          'initialText','actions','expectedText','expectedAdvance','expectedLists'}
NUMBERS = set('1234567890-/:;()$&@".,?!\'')
SYMBOLS = set('[]{}#%^*+=_\\|~<>€£¥•.,?!\'')
LETTERS = set('qwertyuiopasdfghjklzxcvbnm')
EPS = 1e-5

class Invalid(Exception):
    pass

def require(condition, message):
    if not condition:
        raise Invalid(message)

def integer(v):
    return type(v) is int and v >= 0

def number(v):
    return type(v) in (int,float) and math.isfinite(v)

def exact_keys(v, keys, where):
    require(type(v) is dict and set(v) == set(keys), where + ': incorrect schema')

def string(v):
    if type(v) is not str:
        return False
    try:
        v.encode('utf-8')
        return True
    except UnicodeError:
        return False

# ICU character boundaries reproduce native composed-character backspace without
# reading implementation or resource files. Unavailable ICU fails validation.
def boundaries(s):
    require(string(s), 'invalid Unicode string')
    if not s:
        return [0]
    libname = ctypes.util.find_library('icucore') or ctypes.util.find_library('icuuc')
    require(bool(libname), 'ICU unavailable: cannot simulate native backspace')
    lib = ctypes.CDLL(libname)
    def sym(name):
        for suffix in [''] + ['_'+str(n) for n in range(99,49,-1)]:
            try:
                return getattr(lib,name+suffix)
            except AttributeError:
                pass
        raise Invalid('ICU missing '+name)
    op, first, nxt, close = map(sym,['ubrk_open','ubrk_first','ubrk_next','ubrk_close'])
    op.restype = ctypes.c_void_p
    op.argtypes = [ctypes.c_int,ctypes.c_char_p,ctypes.POINTER(ctypes.c_uint16),ctypes.c_int32,ctypes.POINTER(ctypes.c_int32)]
    for f in (first,nxt):
        f.restype=ctypes.c_int32; f.argtypes=[ctypes.c_void_p]
    close.argtypes=[ctypes.c_void_p]; close.restype=None
    raw=s.encode('utf-16-le'); buf=(ctypes.c_uint16*(len(raw)//2)).from_buffer_copy(raw)
    status=ctypes.c_int32(0); iterator=op(0,b'en_US',buf,len(buf),ctypes.byref(status))
    require(bool(iterator) and status.value<=0,'ICU character iterator failed')
    result=[]
    try:
        n=first(iterator)
        while n != -1:
            result.append(n); n=nxt(iterator)
    finally:
        close(iterator)
    return result

def backspace(s):
    if not s:
        return s
    return s.encode('utf-16-le')[:boundaries(s)[-2]*2].decode('utf-16-le')

def load_json(path):
    def pairs(items):
        out={}
        for k,v in items:
            require(k not in out,'duplicate JSON key: '+k)
            out[k]=v
        return out
    with open(path,encoding='utf-8') as f:
        return json.load(f,object_pairs_hook=pairs,parse_constant=lambda x: (_ for _ in ()).throw(Invalid('nonfinite JSON number')))

def validate_references(ref):
    exact_keys(ref,{'locales','dictionaries'},'references')
    exact_keys(ref['locales'],LOCALES,'reference locales')
    for name,region in LOCALES.items():
        entry=ref['locales'][name]
        exact_keys(entry,{'region','separator'},'reference locale '+name)
        require(entry['region']==region,'reference region mismatch '+name)
        sep=entry['separator']
        require(string(sep) and sep and not sep.isspace() and len(boundaries(sep))==2 and not any(c.isalnum() or ord(c)<32 for c in sep),'invalid reference separator '+name)
    exact_keys(ref['dictionaries'],LANGUAGES,'reference dictionaries')
    for name,entry in ref['dictionaries'].items():
        exact_keys(entry,{'available','bytes','sha256'},'reference dictionary '+name)
        require(entry['available'] is True,'dictionary unavailable '+name)
        require(integer(entry['bytes']) and entry['bytes']>0,'empty dictionary '+name)
        require(type(entry['sha256']) is str and re.fullmatch('[0-9a-f]{64}',entry['sha256']) is not None,'invalid dictionary hash '+name)


def capabilities(keyboard,page,globe,sep):
    if keyboard in NUMERIC:
        chars=set('0123456789') | ({sep} if keyboard=='decimal' else set())
        keys=set()
    else:
        chars=(LETTERS.copy() if page=='letters' else NUMBERS.copy() if page=='numbers' else SYMBOLS.copy())
        if keyboard=='email':
            chars|=set('@.')
        if keyboard=='url':
            chars|=set('./')
        keys={'key-return'}
        if keyboard!='url':
            keys.add('key-space')
        keys|=({'key-numbers'} if page=='letters' else {'key-symbols','key-letters'} if page=='numbers' else {'key-numbers','key-letters'})
    return chars,keys


def simulate(row,ref):
    text=row['initialText']; advance=lists=0; keyboard=row['keyboard']; page='letters'; steps=[]
    sep=ref['locales'][row['locale']]['separator']
    for i,action in enumerate(row['actions']):
        exact_keys(action,{'kind','value'} if action.get('kind') in {'tap','key','field'} else {'kind'},f'action {i}')
        kind=action['kind']; before=text; before_keyboard=keyboard; before_page=page
        chars,keys=capabilities(keyboard,page,row['globe'],sep)
        if kind=='tap':
            value=action['value']
            require(string(value) and len(boundaries(value))==2,f'action {i}: tap needs one displayed character')
            require(value in chars,f'action {i}: unavailable tap {value!r} on {keyboard}/{page}')
            text+=value
        elif kind=='key':
            value=action['value']
            require(type(value) is str and value in KEY_ACTIONS,f'action {i}: unknown/forbidden key')
            require(value in keys,f'action {i}: unavailable key {value} on {keyboard}/{page}')
            if value in {'key-numbers','key-symbols','key-letters'}:
                page={'key-numbers':'numbers','key-symbols':'symbols','key-letters':'letters'}[value]
            else:
                require(value!='key-space' or not text.endswith(' '),f'action {i}: double space outside literal protocol')
                text+=' ' if value=='key-space' else '\n'; page='letters'
        elif kind=='delete':
            text=backspace(text)
        elif kind in {'globeTouch','globeAX'}:
            require(row['globe'],f'action {i}: required globe absent')
            if kind=='globeTouch':
                lists+=2
            else:
                advance+=1
        elif kind=='refresh':
            pass
        elif kind=='field':
            value=action['value']; require(type(value) is str and value in KEYBOARDS,f'action {i}: unsupported field')
            if keyboard!=value:
                keyboard=value; page='letters'
        else:
            raise Invalid(f'action {i}: unknown operation {kind!r}')
        steps.append(dict(beforeText=before,afterText=text,advance=advance,lists=lists,
                          keyboard=keyboard,page=page,beforeKeyboard=before_keyboard,beforePage=before_page))
    require(text==row['expectedText'],'invalid gold expectedText: literal protocol produces '+repr(text))
    require(advance==row['expectedAdvance'],'invalid gold expectedAdvance')
    require(lists==row['expectedLists'],'invalid gold expectedLists')
    return steps


def validate_input(rows,ref):
    validate_references(ref)
    require(type(rows) is list and 12<=len(rows)<=24,'input must have 12–24 rows')
    seen=set(); simulations=[]
    for i,row in enumerate(rows):
        try:
            exact_keys(row,FIELDS,'row')
            require(string(row['id']) and bool(row['id']) and row['id'] not in seen,'empty/duplicate id')
            seen.add(row['id'])
            for key,allowed in [('locale',LOCALES),('language',LANGUAGES),('keyboard',KEYBOARDS),('oneHanded',DOCKS)]:
                require(type(row[key]) is str and row[key] in allowed,'invalid '+key)
            require(type(row['globe']) is bool and type(row['privateField']) is bool,'globe/privacy must be Boolean')
            require(number(row['width']) and row['width'] in WIDTHS,'invalid width')
            require(string(row['initialText']) and string(row['expectedText']),'text must be valid Unicode strings')
            require(type(row['actions']) is list and bool(row['actions']),'actions must be nonempty array')
            require(all(type(a) is dict for a in row['actions']),'actions must be objects')
            require(integer(row['expectedAdvance']) and integer(row['expectedLists']),'callbacks must be nonnegative integers')
            simulations.append(simulate(row,ref))
        except Invalid as e:
            raise Invalid(f'row {i} ({row.get("id","?")}): {e}')
    return simulations


def rect(v,name):
    require(type(v) is list and len(v)==4 and all(number(x) for x in v),name+': rectangle must contain four finite numbers')
    require(v[2]>0 and v[3]>0,name+': nonpositive rectangle')
    return v

def inside(inner,outer):
    return inner[0]>=outer[0]-EPS and inner[1]>=outer[1]-EPS and inner[0]+inner[2]<=outer[0]+outer[2]+EPS and inner[1]+inner[3]<=outer[1]+outer[3]+EPS

def overlaps(a,b):
    return min(a[0]+a[2],b[0]+b[2])-max(a[0],b[0])>EPS and min(a[1]+a[3],b[1]+b[3])-max(a[1],b[1])>EPS

def check_stack(v,where):
    require(type(v) is dict,where+': stack missing')
    for key in ('controller','keysView','engine','dictionary'):
        require(string(v.get(key)) and bool(v[key].strip()) and v[key].strip().lower() not in {'nil','null','none','false','0'},where+': missing stack '+key)

def check_languages(v,row,where,requested=False):
    for key in (('requestedLanguage','hostLanguage','loadedLanguage') if requested else ('hostLanguage','loadedLanguage')):
        expected = ('es' if row['language']=='es' else 'en-US') if key=='hostLanguage' else row['language']
        require(v.get(key)==expected,where+': wrong '+key)
    check_stack(v.get('stack'),where)


def check_geometry(g,row,keyboard,page,sep,where):
    require(type(g) is dict,where+': missing geometry')
    require(number(g.get('width')) and g['width']==row['width'],where+': width differs')
    require(number(g.get('height')) and g['height']>0,where+': invalid height')
    grid=rect(g.get('grid'),where+' grid'); bounds=[0,0,g['width'],g['height']]
    require(inside(grid,bounds),where+': grid outside view')
    require(g.get('requestedOneHanded')==row['oneHanded'],where+': wrong requested docking')
    effective=g.get('effectiveOneHanded')
    require(effective in DOCKS,where+': invalid effective docking')
    require(effective==row['oneHanded'] or (effective=='off' and (row['width']==320 or row['width']>430)),where+': unexpected docking unavailable')
    require(number(g.get('effectiveWidth')) and abs(g['effectiveWidth']-grid[2])<=EPS,where+': effective width missing/mismatched')
    keys=g.get('keys'); require(type(keys) is list and bool(keys),where+': no keys')
    ids=[]; targets=[]; texts=[]; short=[]
    for k in keys:
        require(type(k) is dict and string(k.get('id')) and k['id'],where+': key id missing')
        require(k['id'] not in ids,where+': duplicate key id'); ids.append(k['id'])
        require(string(k.get('text')),where+': missing key text'); texts.append(k['text'])
        cap=rect(k.get('cap'),where+' cap '+k['id']); target=rect(k.get('target'),where+' target '+k['id'])
        require(inside(cap,grid),where+': cap outside grid '+k['id'])
        require(inside(target,bounds),where+': target outside view '+k['id'])
        require(target[0]-EPS<=cap[0]+cap[2]/2<=target[0]+target[2]+EPS and target[1]-EPS<=cap[1]+cap[3]/2<=target[1]+target[3]+EPS,where+': target excludes cap center '+k['id'])
        for route in ('centerRoutes','targetRoutes','identityRoutes','rootRoutes'):
            require(k.get(route) is True,where+': routing failed/missing '+route+' '+k['id'])
        require(all(not overlaps(target,t) for t in targets),where+': overlapping target '+k['id']); targets.append(target)
        if target[2]<44-EPS or target[3]<44-EPS:
            short.append({'id':k['id'],'width':target[2],'height':target[3]})
    chars,available=capabilities(keyboard,page,row['globe'],sep)
    for c in chars:
        require(texts.count(c)==1,where+': missing/duplicate displayed character '+repr(c))
    for special in available:
        require(special in ids,where+': missing key '+special)
    require('key-delete' in ids,where+': missing delete')
    require(('key-globe' in ids)==row['globe'],where+': globe visibility differs from host requirement')
    if keyboard in NUMERIC:
        require(sum(t in set('0123456789') for t in texts)==10,where+': expected exactly ten digit controls')
        require((sep in texts)==(keyboard=='decimal'),where+': decimal separator availability wrong')
        allowedchars=set('0123456789') | ({sep} if keyboard=='decimal' else set())
        require(all(t in allowedchars or t=='' for t in texts),where+': unexpected numeric text key')
        for k in keys:
            bottom=k['text']=='0' or (keyboard=='decimal' and k['text']==sep) or k['id'] in {'key-delete','key-globe'}
            if bottom:
                require(k['target'][2]>=44-EPS,where+': numeric bottom target width below 44 '+k['id'])
                if row['width']<=430:
                    require(k['target'][3]>=44-EPS,where+': portrait numeric bottom target height below 44 '+k['id'])
    return ids,short


def check_locale(out,row,ref):
    v=out.get('locale'); require(type(v) is dict,'locale observation missing')
    entry=ref['locales'][row['locale']]
    for prefix in ('current','autoupdating'):
        require(v.get(prefix+'Region')==entry['region'],'wrong '+prefix+' locale region')
        require(v.get(prefix+'DecimalSeparator')==entry['separator'],'wrong '+prefix+' decimal separator')
        identifier=v.get(prefix+'Identifier')
        require(string(identifier) and bool(identifier),'missing '+prefix+' locale identifier')
        # -testLanguage may differ; region is the locale contract. Foundation
        # identifiers may include Unicode / Xcode preference suffixes.
        base=re.split(r'[@.]',identifier)[0].replace('-','_')
        require(re.search(r'(?:^|_)'+re.escape(entry['region'])+r'(?:_|$)',base) is not None,'identifier region mismatch '+prefix)


def check_row(row,out,steps,ref):
    require(type(out) is dict,'missing/nonobject output row')
    require(out.get('id')==row['id'],'row id/order mismatch')
    require(out.get('errors')==[],'row runtime errors missing/nonempty')
    check_locale(out,row,ref); check_languages(out,row,'row',True)
    require(out.get('initialText')==row['initialText'],'initial text mismatch')
    require(type(out.get('initialAdvance')) is int and out['initialAdvance']==0,'initial advance not zero')
    require(type(out.get('initialLists')) is int and out['initialLists']==0,'initial lists not zero')
    observations=out.get('actions'); require(type(observations) is list and len(observations)==len(steps),'action observations omitted/extra')
    previous_count=previous_dirty=None
    for i,(requested,actual,expected) in enumerate(zip(row['actions'],observations,steps)):
        where=f'action {i}'
        require(type(actual) is dict and type(actual.get('index')) is int and actual['index']==i,where+': missing/index mismatch')
        require(actual.get('action')==requested,where+': action differs')
        require(actual.get('errors')==[],where+': runtime errors missing/nonempty')
        for name in ('beforeText','afterText','advance','lists'):
            require(actual.get(name)==expected[name] and (name not in {'advance','lists'} or type(actual.get(name)) is int),where+': wrong '+name)
        require(type(actual.get('documentDidChangeCount')) is int and actual['documentDidChangeCount']==1,where+': missing explicit host callback')
        check_languages(actual,row,where)
        learn=actual.get('learning'); require(type(learn) is dict,where+': learning observer missing')
        for name in ('beforeCount','afterCount'):
            require(integer(learn.get(name)),where+': invalid learning '+name)
        for name in ('beforeDirty','afterDirty'):
            require(type(learn.get(name)) is bool,where+': invalid learning '+name)
        disabled=row['privateField'] or expected['keyboard'] in NUMERIC
        require(learn.get('policy')==('disabled' if disabled else 'allowed'),where+': wrong learning policy')
        if previous_count is not None:
            require(learn['beforeCount']==previous_count and learn['beforeDirty']==previous_dirty,where+': observer continuity lost')
        if disabled or requested['kind'] in {'globeTouch','globeAX'}:
            require(learn['beforeCount']==learn['afterCount'] and learn['beforeDirty']==learn['afterDirty'],where+': prohibited learning/dirty change')
        if requested['kind'] in {'globeTouch','globeAX'}:
            require(actual['beforeText']==actual['afterText'],where+': globe edited document')
        previous_count=learn['afterCount']; previous_dirty=learn['afterDirty']
    for name,gold in [('text','expectedText'),('advance','expectedAdvance'),('lists','expectedLists')]:
        require(out.get(name)==row[gold] and (name=='text' or type(out.get(name)) is int),'final '+name+' mismatch')
    sep=ref['locales'][row['locale']]['separator']
    initial_ids,initial_short=check_geometry(out.get('initialGeometry'),row,row['keyboard'],'letters',sep,'initial')
    require(out.get('initialKeys')==initial_ids,'initialKeys omitted/differ from initial geometry')
    _,final_short=check_geometry(out.get('finalGeometry'),row,steps[-1]['keyboard'],steps[-1]['page'],sep,'final')
    return {'initialShortTargets':initial_short,'finalShortTargets':final_short}


def score(rows,ref,out,steps):
    global_errors=[]
    if type(out) is not dict:
        global_errors.append('output must be object'); out={}
    if type(out.get('schemaVersion')) is not int or out.get('schemaVersion')!=1:
        global_errors.append('missing/unsupported output schemaVersion')
    try:
        validate_references({'locales':ref['locales'],'dictionaries':out.get('dictionaries')})
        require(out['dictionaries']==ref['dictionaries'],'dictionary availability/bytes/hash differ from independent references')
    except (Invalid,TypeError,KeyError) as e:
        global_errors.append(str(e))
    if out.get('errors',[]) != []:
        global_errors.append('top-level runtime errors')
    observations=out.get('rows')
    if type(observations) is not list:
        global_errors.append('rows missing/not array'); observations=[]
    if len(observations)!=len(rows):
        global_errors.append('output row count mismatch/omissions/extras')
    if [r.get('id') if type(r) is dict else None for r in observations] != [r['id'] for r in rows]:
        global_errors.append('output IDs/order differ/omitted/duplicated')
    verdicts=[]
    for i,row in enumerate(rows):
        errors=list(global_errors); details={}
        try:
            details=check_row(row,observations[i] if i<len(observations) else None,steps[i],ref)
        except (Invalid,KeyError,TypeError,ValueError,IndexError) as e:
            errors.append(str(e))
        verdicts.append({'id':row['id'],'passed':not errors,'unchanged':row['expectedText']==row['initialText'],'errors':errors,**details})
    passed=sum(v['passed'] for v in verdicts)
    return {'validInput':True,'passed':passed==len(rows),'summary':{'total':len(rows),'passed':passed,'failed':len(rows)-passed,'unchanged':sum(v['unchanged'] for v in verdicts),'completion':passed/len(rows)},'globalErrors':global_errors,'rows':verdicts,'scope':'hosted workflow completion; installed OS menu and blanket landscape 44pt usability are not established'}


def main(argv):
    require(len(argv) in (4,5) and argv[1] in {'validate','score'},'usage: oracle.py validate INPUT REFERENCES | score INPUT REFERENCES OUTPUT')
    require(len(argv)==(4 if argv[1]=='validate' else 5),'wrong argument count')
    rows=load_json(argv[2]); ref=load_json(argv[3]); steps=validate_input(rows,ref)
    if argv[1]=='validate':
        result={'valid':True,'rows':len(rows),'actions':sum(len(s) for s in steps)}
    else:
        try:
            out=load_json(argv[4])
        except (OSError,ValueError,Invalid) as e:
            out={'errors':['unreadable/invalid output: '+str(e)]}
        result=score(rows,ref,out,steps)
    print(json.dumps(result,ensure_ascii=False,indent=2)); return 0 if result.get('valid',result.get('passed',False)) else 1

if __name__=='__main__':
    try:
        sys.exit(main(sys.argv))
    except (Invalid,OSError,ValueError,TypeError,KeyError,IndexError) as e:
        print(json.dumps({'valid':False,'passed':False,'errors':[str(e)]},ensure_ascii=False,indent=2)); sys.exit(1)
