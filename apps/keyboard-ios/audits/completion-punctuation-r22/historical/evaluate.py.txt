#!/usr/bin/env python3
"""Replay two immutable source refs; reveal paired results only after both runs finish."""
import argparse,hashlib,json,pathlib,subprocess,sys
p=argparse.ArgumentParser();p.add_argument('--baseline',required=True);p.add_argument('--candidate',required=True);p.add_argument('--input',required=True);p.add_argument('--name',required=True);p.add_argument('--sealed',action='store_true');a=p.parse_args()
audit=pathlib.Path(__file__).resolve().parent
source=pathlib.Path('apps/keyboard-ios/PulpoKeyboardKit/Sources/KeyboardCore/Input/KeyboardEngine.swift')
refs={k:subprocess.check_output(['git','rev-parse',v],text=True).strip() for k,v in [('baseline',a.baseline),('candidate',a.candidate)]}
variants={k:subprocess.check_output(['git','show',f'{v}:{source}']) for k,v in refs.items()}
assert source.read_bytes()==variants['candidate'],'Source differs from immutable candidate'
if a.sealed:
 freeze=json.loads((audit/'candidate-freeze.json').read_text())
 assert hashlib.sha256(pathlib.Path(a.input).read_bytes()).hexdigest()==freeze['heldoutSHA256']
 for name,digest in freeze['sha256'].items():
  assert hashlib.sha256(pathlib.Path(name).read_bytes()).hexdigest()==digest,name
outputs={k:pathlib.Path(f'/tmp/r22-{a.name}-{k}.json') for k in refs}
assert not any(p.exists() for p in outputs.values()),'Refuse to overwrite or repeat an evaluated pair'
try:
 for label in ['baseline','candidate']:
  source.write_bytes(variants[label])
  command=[sys.executable,str(audit/'run.py'),'--input',a.input,'--output',str(outputs[label]),'--summary',str(audit/f'{a.name}-{label}-execution.json')]
  if a.sealed:command+=['--sealed']
  subprocess.run(command,check=True)
finally:
 if source.read_bytes()!=variants['candidate']:source.write_bytes(variants['candidate'])
if a.sealed:
 for name,digest in freeze['sha256'].items():
  assert hashlib.sha256(pathlib.Path(name).read_bytes()).hexdigest()==digest,name
subprocess.run([sys.executable,str(audit/'compare.py'),str(outputs['baseline']),str(outputs['candidate']),str(audit/f'{a.name}-paired.json')],check=True)
