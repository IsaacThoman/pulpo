import Foundation

@testable import KeyboardCore

private func r37SHA(_ data: Data) -> String { SHA256.hash(data: data).map { String(format: "%02x", $0) }.joined() }
private func r37Null<T>(_ value: T?) -> Any { value.map { $0 as Any } ?? NSNull() }
private func r37Diagnostic(_ value: Any) -> Any {
  let mirror = Mirror(reflecting: value)
  if mirror.displayStyle == .optional { return mirror.children.first.map { r37Diagnostic($0.value) } ?? NSNull() }
  if let v = value as? String { return v }
  if let v = value as? Int { return v }
  if let v = value as? Bool { return v }
  if let v = value as? Double, v.isFinite { return v }
  if let v = value as? UUID { return v.uuidString }
  if mirror.displayStyle == .struct {
    return Dictionary(uniqueKeysWithValues: mirror.children.compactMap { child -> (String, Any)? in
      child.label.map { ($0, r37Diagnostic(child.value)) }
    })
  }
  return String(describing: value)
}

private struct R37Resources {
  let directory: URL
  let hashes: [String: String]
  init(path: String) throws {
    guard path.hasPrefix("/") else { throw R37Failure("R37_RESOURCE_DIR must be an explicit absolute directory (also mandatory for baseline)") }
    directory = URL(fileURLWithPath: path, isDirectory: true).standardizedFileURL
    var hashes: [String: String] = [:]
    for mode in R37Input.modes {
      hashes[mode] = r37SHA(try Data(contentsOf: directory.appendingPathComponent(mode + ".pkdict")))
    }
    self.hashes = hashes
  }
  func url(_ mode: String) -> URL { directory.appendingPathComponent(mode + ".pkdict") }
}

private final class R37Session {
  let row: R37Case
  let resources: R37Resources
  let document: R37Document
  let engine: KeyboardEngine
  var stack: LanguageStack
  var stackOrdinal = 0
  var fieldPolicy = "normal"
  var captures: [Suggestion] = []
  var records: [[String: Any]] = []
  var failures: [String] = []
  var lastSlots: [Suggestion] = []
  let loadMilliseconds: Double
  init(row: R37Case, mode: String, resources: R37Resources) throws {
    self.row = row
    self.resources = resources
    document = R37Document(row)
    let start = ProcessInfo.processInfo.systemUptime
    stack = try LanguageStack(dictionaryURL: resources.url(mode), personalURL: nil)
    loadMilliseconds = (ProcessInfo.processInfo.systemUptime - start) * 1000
    var settings = KeyboardSettings()
    settings.language = KeyboardLanguage(rawValue: mode)!
    settings.predictions = true
    settings.learnWords = false
    settings.autoCapitalization = false
    settings.smartPunctuation = false
    settings.doubleSpacePeriod = false
    engine = KeyboardEngine(document: document, suggester: stack.suggester, glide: stack.glide, settings: settings)
    var traits = InputTraits()
    traits.capitalization = .none
    engine.traits = traits
    engine.documentDidChange()
    guard stack.personal.totalCount == 0, stack.personal.knownWords.isEmpty, !stack.personal.hasUnsavedChanges,
          stack.suggester.textReplacements.isEmpty else { throw R37Failure("personal dictionary must start empty") }
  }
  func identities() -> [String: Any] {
    ["stackOrdinal": stackOrdinal, "stack": String(describing: ObjectIdentifier(stack)),
     "lexicon": String(describing: ObjectIdentifier(stack.lexicon)), "model": String(describing: ObjectIdentifier(stack.model)),
     "personal": String(describing: ObjectIdentifier(stack.personal)), "suggester": String(describing: ObjectIdentifier(stack.suggester)),
     "corrector": String(describing: ObjectIdentifier(stack.corrector)), "glide": String(describing: ObjectIdentifier(stack.glide)),
     "touch": String(describing: ObjectIdentifier(stack.touch))]
  }
  func state() -> [String: Any] {
    let p = engine.policy
    return ["document": document.state, "composingText": engine.composingText,
            "logicalContextBeforeCursor": engine.contextBeforeCursor,
            "mode": engine.settings.language.rawValue, "resourceSHA256": resources.hashes[engine.settings.language.rawValue]!,
            "resourcePath": resources.url(engine.settings.language.rawValue).path, "objects": identities(),
            "fieldPolicy": fieldPolicy, "predictions": engine.settings.predictions, "learnWords": engine.settings.learnWords,
            "policy": ["showsSuggestions": p.showsSuggestions, "learns": p.learns, "autocorrects": p.autocorrects,
                       "appliesTextReplacements": p.appliesTextReplacements, "capitalization": String(describing: p.capitalization),
                       "smartQuotes": p.smartQuotes, "smartSpacing": p.smartSpacing, "autoSpaces": p.autoSpaces],
            "personal": ["totalCount": stack.personal.totalCount, "knownWordCount": stack.personal.knownWords.count,
                         "hasUnsavedChanges": stack.personal.hasUnsavedChanges],
            "languageContextDiagnostic": r37Diagnostic(engine.languageContext())]
  }
  func slot(_ suggestion: Suggestion, index: Int, count: Int) -> [String: Any] {
    let rankOrder = count >= 2 ? [1, 0, 2] : [0]
    return ["displayIndex": index, "modelRank": rankOrder.firstIndex(of: index)! + 1,
            "text": suggestion.text, "kind": String(describing: suggestion.kind),
            "isAutocorrection": suggestion.isAutocorrection, "accessibilityLabel": r37Null(suggestion.accessibilityLabel),
            "sourceContext": r37Null(suggestion.sourceContext?.uuidString)]
  }
  func observe() -> [String: Any] {
    let start = ProcessInfo.processInfo.systemUptime
    // Do not reconcile: silent-host stale-tap tests depend on this exact API.
    let set = engine.suggestions
    let elapsed = (ProcessInfo.processInfo.systemUptime - start) * 1000
    lastSlots = set.slots
    let captureStart = captures.count
    captures.append(contentsOf: set.slots) // Actual values, including hidden UUID.
    return ["slots": set.slots.enumerated().map { slot($0.element, index: $0.offset, count: set.slots.count) },
            "autocorrection": r37Null(set.autocorrection), "captureStart": captureStart,
            "captureCount": set.slots.count, "elapsedMilliseconds": elapsed]
  }
  func check(_ expected: R37Expectation, primitiveCount: Int) throws {
    guard r37Exact(document.text, expected.document), document.caret == expected.caret,
          document.selectionLength == expected.selectionLength, primitiveCount == expected.primitiveCount else {
      throw R37Failure("expected doc=\(String(reflecting: expected.document)) caret=\(expected.caret) selection=\(expected.selectionLength) primitives=\(expected.primitiveCount); got doc=\(String(reflecting: document.text)) caret=\(document.caret) selection=\(document.selectionLength) primitives=\(primitiveCount)")
    }
    if let count = expected.slotCount {
      // Explicit expectation reads the bar without reconciliation, just as observe.
      guard engine.suggestions.slots.count == count else { throw R37Failure("expected slotCount \(count)") }
    }
    if let count = expected.personalCount, stack.personal.totalCount != count { throw R37Failure("personal count differs") }
    if let dirty = expected.personalDirty, stack.personal.hasUnsavedChanges != dirty { throw R37Failure("personal dirty differs") }
  }
  func perform(_ action: R37Action) {
    let before = state()
    let primitiveStart = document.primitives.count
    let start = ProcessInfo.processInfo.systemUptime
    var detail: [String: Any] = [:]
    var error: String?
    do {
      switch action.action {
      case "observe": detail = observe()
      case "type": for character in action.text! { engine.insertCharacter(String(character)) }
      case "space": engine.insertSpace()
      case "return": engine.insertReturn()
      case "backspace": engine.deleteBackward()
      case "setMode":
        let next = try LanguageStack(dictionaryURL: resources.url(action.mode!), sharing: stack)
        engine.settings.language = KeyboardLanguage(rawValue: action.mode!)!
        engine.useLanguageStack(next)
        stack = next
        stackOrdinal += 1
      case "hostMove":
        try r37Bounds(document.text, caret: action.caret!, selection: action.selectionLength!)
        document.caret = action.caret!
        document.selectionLength = action.selectionLength!
        if action.callback! { engine.documentDidChange() }
      case "hostReplace":
        try r37Bounds(action.document!, caret: action.caret!, selection: action.selectionLength!)
        document.text = action.document!
        document.caret = action.caret!
        document.selectionLength = action.selectionLength!
        if action.newDocumentID == true { document.documentIdentifier = UUID() }
        if action.callback! { engine.documentDidChange() }
      case "setPolicy":
        var traits = InputTraits()
        traits.capitalization = .none
        switch action.policy! {
        case "normal": break
        case "private": traits.isSecure = true
        case "url": traits.keyboard = .url
        case "numeric": traits.keyboard = .numberPad
        default: throw R37Failure("unsupported policy")
        }
        engine.traits = traits
        engine.settings.predictions = action.predictions!
        engine.settings.learnWords = action.learnWords!
        fieldPolicy = action.policy!
      case "acceptSlot":
        let actual = engine.suggestions.slots
        guard actual.indices.contains(action.index!) else { throw R37Failure("missing displayed slot \(action.index!)") }
        let choice = actual[action.index!]
        detail["selectedSuggestion"] = slot(choice, index: action.index!, count: actual.count)
        engine.select(choice)
      case "replayCapturedSlot":
        guard captures.indices.contains(action.capture!) else { throw R37Failure("capture index out of bounds") }
        let choice = captures[action.capture!]
        detail["capturedSuggestion"] = ["text": choice.text, "kind": String(describing: choice.kind),
                                       "sourceContext": r37Null(choice.sourceContext?.uuidString)]
        engine.select(choice)
      default: throw R37Failure("unsupported action")
      }
      let host = action.action == "hostMove" || action.action == "hostReplace"
      if host {
        guard primitiveStart == document.primitives.count else { throw R37Failure("host action caused engine primitive") }
      } else {
        try document.verifyTrace(from: primitiveStart, initial: before["document"] as! [String: Any])
      }
      if let expected = action.expect { try check(expected, primitiveCount: document.primitives.count - primitiveStart) }
      // Every learning-off or private action must preserve all personal counters.
      let oldPersonal = before["personal"] as! [String: Any]
      let wasLearning = (before["policy"] as! [String: Any])["learns"] as! Bool
      if !wasLearning {
        guard oldPersonal["totalCount"] as! Int == stack.personal.totalCount,
              oldPersonal["knownWordCount"] as! Int == stack.personal.knownWords.count,
              oldPersonal["hasUnsavedChanges"] as! Bool == stack.personal.hasUnsavedChanges else { throw R37Failure("learning-off/privacy personal state changed") }
      }
    } catch let failure { error = String(describing: failure); failures.append("\(action.action): \(failure)") }
    records.append(["action": action.action, "arguments": r37Diagnostic(action), "before": before, "after": state(),
                    "primitives": Array(document.primitives.dropFirst(primitiveStart)), "detail": detail,
                    "elapsedMilliseconds": (ProcessInfo.processInfo.systemUptime - start) * 1000, "error": r37Null(error)])
  }
  func stage() throws {
    guard !row.typed.isEmpty else { return }
    let data = try JSONSerialization.data(withJSONObject: ["action": "type", "text": row.typed])
    perform(try JSONDecoder().decode(R37Action.self, from: data))
  }
  func isolatedAcceptance(index: Int, observed: [Suggestion]) -> [String: Any] {
    let before = state()
    let first = document.primitives.count
    var error: String?
    var chosen: Any = NSNull()
    var expected: Any = NSNull()
    let start = ProcessInfo.processInfo.systemUptime
    do {
      guard observed.indices.contains(index) else { throw R37Failure("missing displayed slot \(index); full-denominator acceptance failure") }
      let choice = observed[index] // Same exact value returned by this engine's initial observation.
      chosen = slot(choice, index: index, count: observed.count)
      guard choice.sourceContext != nil, document.selectionLength == 0,
            r37Exact(document.documentContextBeforeInput!, row.prefix + row.typed),
            r37Exact(engine.composingText, row.typed) else { throw R37Failure("unexpected primary initial context/composition") }
      let space = row.suffix.first?.isWhitespace == true ? "" : " "
      let expectedDocument = row.prefix + choice.text + space + row.suffix
      let expectedCaret = r37CaretAfterUTF16Insertion(expectedDocument, target: (row.prefix + choice.text + space).utf16.count)
      let count: Int
      if choice.kind == .literal { count = space.isEmpty ? 0 : 1 }
      else { count = row.typed.count + 1 }
      expected = ["document": expectedDocument, "caret": expectedCaret, "selectionLength": 0, "primitiveCount": count, "unchangedSuffix": row.suffix]
      engine.select(choice)
      try document.verifyTrace(from: first, initial: before["document"] as! [String: Any])
      guard r37Exact(document.text, expectedDocument), document.caret == expectedCaret, document.selectionLength == 0,
            document.primitives.count - first == count,
            r37Exact(document.documentContextAfterInput!, row.suffix), stack.personal.totalCount == 0,
            !stack.personal.hasUnsavedChanges else { throw R37Failure("isolated acceptance document/range/suffix/primitive/privacy oracle failed") }
    } catch let failure { error = String(describing: failure); failures.append("accept slot \(index): \(failure)") }
    return ["displayIndex": index, "selectedSuggestion": chosen, "expected": expected, "before": before, "after": state(),
            "primitives": Array(document.primitives.dropFirst(first)), "error": r37Null(error),
            "elapsedMilliseconds": (ProcessInfo.processInfo.systemUptime - start) * 1000]
  }
}

struct R37ExecutorTests {
  @Test func runStrictEvaluation() throws {
    let env = ProcessInfo.processInfo.environment
    guard let inputPath = env["R37_INPUT"] else { return } // Opt-in external workload, never bundled target data.
    guard inputPath.hasPrefix("/"), let outputPath = env["R37_OUTPUT"], outputPath.hasPrefix("/"),
          let resourcePath = env["R37_RESOURCE_DIR"], let variant = env["R37_VARIANT"], ["baseline", "candidate"].contains(variant) else {
      throw R37Failure("set absolute R37_INPUT, R37_OUTPUT, R37_RESOURCE_DIR and label R37_VARIANT=baseline|candidate; no resource fallback")
    }
    let audit = URL(fileURLWithPath: #filePath).deletingLastPathComponent().deletingLastPathComponent()
      .deletingLastPathComponent().deletingLastPathComponent().appendingPathComponent("audits/sentence-nextword-r37")
    let data = try Data(contentsOf: URL(fileURLWithPath: inputPath))
    let schemaData = try Data(contentsOf: audit.appendingPathComponent("public-schema.json"))
    let input = try R37Input.load(data: data, schema: schemaData) // Entire input admitted BEFORE resource/model/document creation.
    let resources = try R37Resources(path: resourcePath)
    var rows: [[String: Any]] = []
    var allFailures: [String] = []
    for row in input.cases {
      for mode in R37Input.modes {
        var result: [String: Any] = ["id": row.id, "familyId": row.familyId, "contentLabel": row.contentLabel,
                                   "oracle": row.oracle, "route": row.route, "mode": mode, "acceptable": row.acceptableByMode[mode]!]
        var failures: [String] = []
        var acceptance: [[String: Any]] = []
        do {
          let session = try R37Session(row: row, mode: mode, resources: resources)
          try session.stage()
          for action in row.actions { session.perform(action) }
          result["loadMilliseconds"] = session.loadMilliseconds
          result["actions"] = session.records
          result["final"] = session.state()
          result["initialSlots"] = session.lastSlots.enumerated().map { session.slot($0.element, index: $0.offset, count: session.lastSlots.count) }
          failures += session.failures
          if row.route != "control" {
            let gold = row.acceptableByMode[mode]!
            let hits = session.lastSlots.map { suggestion in gold.contains { r37Exact($0, suggestion.text) } }
            let ranked = session.lastSlots.count >= 2 ? [1, 0, 2].filter { $0 < session.lastSlots.count } : Array(session.lastSlots.indices)
            result["top3Hit"] = row.oracle == "scored" ? hits.contains(true) as Any : NSNull()
            result["centerHit"] = row.oracle == "scored" ? (!hits.isEmpty && hits[hits.count >= 2 ? 1 : 0]) as Any : NSNull()
            result["reciprocalRank"] = row.oracle == "scored" ? (ranked.firstIndex(where: { hits[$0] }).map { 1.0 / Double($0 + 1) } ?? 0) as Any : NSNull()
            // Each branch starts with a fresh real stack/engine/document. No prior acceptance affects it.
            for index in 0..<3 {
              do {
                let branch = try R37Session(row: row, mode: mode, resources: resources)
                try branch.stage()
                let observation = branch.observe()
                var accepted = branch.isolatedAcceptance(index: index, observed: branch.lastSlots)
                accepted["observation"] = observation
                accepted["stagingActions"] = branch.records
                if !branch.lastSlots.map(\.text).elementsEqual(session.lastSlots.map(\.text), by: r37Exact) {
                  branch.failures.append("independent branch initial suggestions differ")
                }
                acceptance.append(accepted)
                failures += branch.failures
              } catch { failures.append("branch \(index): \(error)"); acceptance.append(["displayIndex": index, "error": String(describing: error)]) }
            }
          } else {
            result["top3Hit"] = NSNull(); result["centerHit"] = NSNull(); result["reciprocalRank"] = NSNull()
          }
        } catch { failures.append(String(describing: error)) }
        result["acceptance"] = acceptance
        result["failures"] = failures
        rows.append(result)
        allFailures += failures.map { "\(row.id)/\(mode): \($0)" }
      }
    }
    var equivalenceResults: [[String: Any]] = []
    for pair in input.equivalences {
      for mode in R37Input.modes {
        let left = rows.first { $0["id"] as? String == pair.left && $0["mode"] as? String == mode }!
        let right = rows.first { $0["id"] as? String == pair.right && $0["mode"] as? String == mode }!
        func observable(_ result: [String: Any]) -> [String: Any] {
          let final = result["final"] as? [String: Any] ?? [:]
          let doc = final["document"] as? [String: Any] ?? [:]
          let slots = (result["initialSlots"] as? [[String: Any]] ?? []).map { slot in
            slot.filter { $0.key != "sourceContext" }
          }
          return ["document": doc.filter { $0.key != "documentID" }, "slots": slots,
                  "composingText": final["composingText"] ?? NSNull(), "logicalContextBeforeCursor": final["logicalContextBeforeCursor"] ?? NSNull(),
                  "mode": final["mode"] ?? NSNull(), "policy": final["policy"] ?? NSNull(), "personal": final["personal"] ?? NSNull()]
        }
        let l = try JSONSerialization.data(withJSONObject: observable(left), options: [.sortedKeys])
        let r = try JSONSerialization.data(withJSONObject: observable(right), options: [.sortedKeys])
        let passed = l == r && (left["failures"] as! [String]).isEmpty && (right["failures"] as! [String]).isEmpty
        equivalenceResults.append(["left": pair.left, "right": pair.right, "mode": mode, "passed": passed])
        if !passed { allFailures.append("equivalence \(pair.left)/\(pair.right)/\(mode) failed") }
      }
    }
    var artifacts = ["input": r37SHA(data), "public-schema.json": r37SHA(schemaData)]
    for name in ["public-contract.md", "public-api-map.json"] {
      artifacts[name] = r37SHA(try Data(contentsOf: audit.appendingPathComponent(name)))
    }
    let testDirectory = URL(fileURLWithPath: #filePath).deletingLastPathComponent()
    for name in ["R37Contract.swift", "R37Document.swift", "R37ExecutorTests.swift"] {
      artifacts[name] = r37SHA(try Data(contentsOf: testDirectory.appendingPathComponent(name)))
    }
    let afterResources = try R37Resources(path: resourcePath)
    if afterResources.hashes != resources.hashes { allFailures.append("resource bytes changed during execution") }
    let output: [String: Any] = ["schemaVersion": 1, "variant": variant, "cohort": input.cohort,
                               "inputPath": inputPath, "input": try JSONSerialization.jsonObject(with: data), "equivalences": equivalenceResults, "resourceDirectory": resources.directory.path,
                               "artifactSHA256": artifacts, "resourceSHA256": resources.hashes,
                               "rowCount": rows.count, "rows": rows, "failures": allFailures,
                               "costDefinition": "wall-clock milliseconds; no per-slot ranking scores or model candidate pool exposed"]
    let raw = try JSONSerialization.data(withJSONObject: output, options: [.prettyPrinted, .sortedKeys, .withoutEscapingSlashes])
    try raw.write(to: URL(fileURLWithPath: outputPath), options: .atomic)
    #expect(allFailures.isEmpty, "R37 failures (full raw output retained): \(allFailures.joined(separator: "; "))")
  }
}
