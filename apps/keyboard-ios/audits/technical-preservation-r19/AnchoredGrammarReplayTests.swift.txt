import Foundation
import Testing
@testable import KeyboardCore

/// Opt-in synthetic R19 replay. Each row gets a fresh actual language stack.
struct AnchoredGrammarReplayTests {
  struct Input: Decodable { let cases: [Case] }
  struct Case: Decodable {
    let id: String
    let category: String
    let before: String
    let after: String
    let operations: [Operation]
    let expected: String
    let expectedByMode: [String: String]?
  }
  struct Operation: Decodable {
    let op: String
    let text: String?
    let count: Int?
    let offset: Int?
    let start: Int?
    let end: Int?
    let enabled: Bool?
    let expectDocument: String?
    let expectDocumentByMode: [String: String]?
    let expectNoLearning: Bool?
    let expectNoAutocorrection: Bool?
    let expectAutocorrection: String?
    let expectCursor: Int?
    let expectSelection: Int?
    let expectUndoAvailable: Bool?
    let expectLearns: Bool?
    let expectAutocorrects: Bool?
    let expectLearningUnchanged: Bool?
  }
  final class Document: TextDocument {
    var documentIdentifier: UUID? = UUID()
    var text: String
    var cursor: Int
    var selected = 0
    init(_ item: Case) { text = item.before + item.after; cursor = item.before.count }
    func index(_ offset: Int) -> String.Index { text.index(text.startIndex, offsetBy: offset) }
    var documentContextBeforeInput: String? { String(text[..<index(cursor)]) }
    var documentContextAfterInput: String? { String(text[index(cursor + selected)...]) }
    var selectedText: String? { selected == 0 ? nil : String(text[index(cursor)..<index(cursor + selected)]) }
    var hasText: Bool { !text.isEmpty }
    func insertText(_ inserted: String) {
      text.replaceSubrange(index(cursor)..<index(cursor + selected), with: inserted)
      cursor += inserted.count
      selected = 0
    }
    func deleteBackward() {
      if selected > 0 {
        text.removeSubrange(index(cursor)..<index(cursor + selected)); selected = 0
      } else if cursor > 0 {
        text.removeSubrange(index(cursor - 1)..<index(cursor)); cursor -= 1
      }
    }
  }
  struct Snapshot: Codable {
    let operation: Int
    let document: String
    let cursor: Int
    let selected: Int
    let alternatives: [String]
    let autocorrection: String?
    let undoAvailable: Bool
    let learned: [String]
    let learnedCounts: [String: Int]
    let personalCount: Int
    let touchWeight: Double
    let learns: Bool
    let autocorrects: Bool
    let milliseconds: Double
  }
  struct Outcome: Codable {
    let id: String
    let category: String
    let mode: String
    let expected: String
    let output: String
    let correct: Bool
    let oracleErrors: [String]
    let snapshots: [Snapshot]
    let finalState: Snapshot?
    let unavailable: String?
    let keyMilliseconds: [Double]
  }
  static func mode(_ language: KeyboardLanguage) -> String {
    switch language { case .english: "en"; case .spanish: "es"; case .bilingual: "bi" }
  }
  static func replay(_ item: Case, language: KeyboardLanguage, schemaErrors: [String]) throws -> Outcome {
    var settings = KeyboardSettings()
    settings.language = language
    let stack = try LanguageStack(dictionaryURL: Fixtures.languageData.appendingPathComponent(language.rawValue + ".pkdict"), personalURL: nil)
    let document = Document(item)
    let engine = KeyboardEngine(document: document, suggester: stack.suggester, glide: stack.glide, settings: settings)
    var now = 100.0
    engine.clock = { now }
    engine.documentDidChange()
    var snapshots: [Snapshot] = []
    var errors = schemaErrors
    var privateLearning: ([String: Int], Double)?
    func learnedCounts() -> [String: Int] { Dictionary(uniqueKeysWithValues: stack.personal.usedEntries.map { ($0.text, stack.personal.count(of: $0.text)) }) }
    func snapshot(_ operation: Int, _ milliseconds: Double) -> Snapshot {
      let suggestions = engine.suggestions
      return Snapshot(operation: operation, document: document.text, cursor: document.cursor, selected: document.selected, alternatives: suggestions.slots.map(\.text), autocorrection: suggestions.autocorrection, undoAvailable: engine.deleteUndoesLastEdit, learned: stack.personal.usedEntries.map(\.text), learnedCounts: learnedCounts(), personalCount: stack.personal.totalCount, touchWeight: stack.touch.weight, learns: engine.policy.learns, autocorrects: engine.policy.autocorrects, milliseconds: milliseconds)
    }
    var times: [Double] = []
    for (index, op) in item.operations.enumerated() {
      let start = ProcessInfo.processInfo.systemUptime
      switch op.op {
      case "insert":
        guard let text = op.text else { errors.append("malformed: insert missing text"); continue }
        for character in text {
          now += 0.25
          let keyStart = ProcessInfo.processInfo.systemUptime
          switch character {
          case " ": engine.insertSpace(tap: KeyPoint(x: 5, y: 3))
          case "\n": engine.insertReturn()
          default:
            let key = String(character).lowercased()
            if character.isUppercase, !engine.shift.isActive { engine.shiftDown(); engine.shiftUp() }
            let point = character.isLetter && key.utf8.count == 1 ? key.utf8.first.flatMap(LetterGeometry.qwerty.center) : nil
            engine.insertCharacter(key, tap: point)
          }
          _ = engine.suggestions
          times.append((ProcessInfo.processInfo.systemUptime - keyStart) * 1_000)
        }
      case "backspace":
        guard (op.count ?? 1) >= 0 else { errors.append("malformed: negative backspace count"); continue }
        for _ in 0..<(op.count ?? 1) { now += 0.25; engine.deleteBackward() }
      case "undo": engine.deleteBackward()
      case "move":
        guard let offset = op.offset else { errors.append("malformed: move missing offset"); continue }
        let destination = document.cursor + offset
        guard (0...document.text.count).contains(destination) else { errors.append("malformed: move out of bounds"); continue }
        document.cursor = destination; document.selected = 0; engine.documentDidChange()
      case "select":
        guard let lower = op.start, let upper = op.end, lower >= 0, upper >= lower, upper <= document.text.count else { errors.append("malformed: select out of bounds"); continue }
        document.cursor = lower; document.selected = upper - lower; engine.documentDidChange()
      case "reset":
        document.text = op.text ?? ""; document.cursor = document.text.count; document.selected = 0
        document.documentIdentifier = UUID(); engine.documentDidChange()
      case "private":
        if op.enabled ?? true { privateLearning = (learnedCounts(), stack.touch.weight) }
        var traits = engine.traits; traits.isSecure = op.enabled ?? true; engine.traits = traits
        engine.documentDidChange()
      case "sync": engine.documentDidChange()
      case "save": engine.save(inBackground: false)
      default: errors.append("unsupported operation: \(op.op)")
      }
      let suggestions = engine.suggestions
      if let expected = op.expectDocumentByMode?[mode(language)] ?? op.expectDocument, document.text != expected { errors.append("document checkpoint \(index): expected \(expected), got \(document.text)") }
      if op.expectNoLearning == true, !stack.personal.usedEntries.isEmpty || stack.touch.weight != 0 { errors.append("learning checkpoint \(index)") }
      if op.expectNoAutocorrection == true, suggestions.autocorrection != nil { errors.append("unexpected autocorrection checkpoint \(index)") }
      if let expected = op.expectAutocorrection, suggestions.autocorrection != expected { errors.append("autocorrection checkpoint \(index): expected \(expected)") }
      if let expected = op.expectCursor, document.cursor != expected { errors.append("cursor checkpoint \(index)") }
      if let expected = op.expectSelection, document.selected != expected { errors.append("selection checkpoint \(index)") }
      if let expected = op.expectUndoAvailable, engine.deleteUndoesLastEdit != expected { errors.append("undo checkpoint \(index)") }
      if let expected = op.expectLearns, engine.policy.learns != expected { errors.append("learns checkpoint \(index)") }
      if let expected = op.expectAutocorrects, engine.policy.autocorrects != expected { errors.append("autocorrects checkpoint \(index)") }
      if op.expectLearningUnchanged == true {
        if let original = privateLearning {
          if original.0 != learnedCounts() || original.1 != stack.touch.weight { errors.append("private learning delta checkpoint \(index)") }
        } else { errors.append("malformed: no private learning checkpoint") }
      }
      snapshots.append(snapshot(index, (ProcessInfo.processInfo.systemUptime - start) * 1_000))
    }
    engine.save(inBackground: false)
    if item.operations.last?.expectNoLearning == true, stack.personal.totalCount != 0 || stack.touch.weight != 0 { errors.append("learning after final save") }
    if item.operations.last?.expectLearningUnchanged == true, let original = privateLearning, original.0 != learnedCounts() || original.1 != stack.touch.weight { errors.append("private learning delta after final save") }
    let expected = item.expectedByMode?[mode(language)] ?? item.expected
    return Outcome(id: item.id, category: item.category, mode: mode(language), expected: expected, output: document.text,
                   correct: document.text == expected && errors.isEmpty, oracleErrors: errors, snapshots: snapshots, finalState: snapshot(-1, 0), unavailable: nil, keyMilliseconds: times)
  }
  @Test func frozenReplay() throws {
    let env = ProcessInfo.processInfo.environment
    guard let input = env["PK_R19_CASES"] else { return }
    let report = try #require(env["PK_R19_REPORT"])
    let data = try Data(contentsOf: URL(fileURLWithPath: input))
    let cases = try JSONDecoder().decode(Input.self, from: data).cases
    let raw = try #require(JSONSerialization.jsonObject(with: data) as? [String: Any])
    let rawCases = try #require(raw["cases"] as? [[String: Any]])
    let caseKeys: Set<String> = ["id", "category", "before", "after", "operations", "expected", "expectedByMode", "notes"]
    let opKeys: Set<String> = ["op", "text", "count", "offset", "start", "end", "enabled", "expectDocument", "expectDocumentByMode", "expectNoLearning", "expectNoAutocorrection", "expectAutocorrection", "expectCursor", "expectSelection", "expectUndoAvailable", "expectLearns", "expectAutocorrects", "expectLearningUnchanged"]
    let modeKeys: Set<String> = ["en", "es", "bi"]
    let categories: Set<String> = ["literal", "ordinary", "control", "safety", "intentional-miss"]
    func schemaErrors(_ row: [String: Any]) -> [String] {
      var errors = Set(raw.keys).subtracting(["cases"]).map { "schema: unknown top key \($0)" }
      errors += Set(row.keys).subtracting(caseKeys).map { "schema: unknown case key \($0)" }
      if !categories.contains(row["category"] as? String ?? "") { errors.append("schema: unknown category") }
      if let modes = row["expectedByMode"] as? [String: Any], !Set(modes.keys).isSubset(of: modeKeys) { errors.append("schema: unknown expected mode") }
      let arguments: [String: Set<String>] = ["insert": ["text"], "backspace": ["count"], "move": ["offset"], "select": ["start", "end"], "reset": ["text"], "private": ["enabled"], "sync": [], "save": [], "undo": []]
      let dataKeys: Set<String> = ["text", "count", "offset", "start", "end", "enabled"]
      for op in row["operations"] as? [[String: Any]] ?? [] {
        if let allowed = arguments[op["op"] as? String ?? ""] {
          for key in Set(op.keys).intersection(dataKeys).subtracting(allowed) { errors.append("malformed: unexpected argument \(key)") }
        }
        errors += Set(op.keys).subtracting(opKeys).map { "schema: unknown operation key \($0)" }
        if let modes = op["expectDocumentByMode"] as? [String: Any], !Set(modes.keys).isSubset(of: modeKeys) { errors.append("schema: unknown checkpoint mode") }
      }
      return errors.sorted()
    }
    try #require(!cases.isEmpty)
    try #require(Set(cases.map(\.id)).count == cases.count)
    var outcomes: [Outcome] = []
    for language in KeyboardLanguage.allCases {
      for (index, item) in cases.enumerated() {
        do { outcomes.append(try Self.replay(item, language: language, schemaErrors: schemaErrors(rawCases[index]))) }
        catch { outcomes.append(Outcome(id: item.id, category: item.category, mode: Self.mode(language), expected: item.expectedByMode?[Self.mode(language)] ?? item.expected, output: "", correct: false, oracleErrors: ["unavailable: \(error)"], snapshots: [], finalState: nil, unavailable: String(describing: error), keyMilliseconds: [])) }
      }
    }
    let encoder = JSONEncoder()
    encoder.outputFormatting = [.prettyPrinted, .sortedKeys, .withoutEscapingSlashes]
    try encoder.encode(outcomes).write(to: URL(fileURLWithPath: report), options: .atomic)
    print("R19 exact \(outcomes.filter(\.correct).count)/\(outcomes.count), oracle errors \(outcomes.reduce(0) { $0 + $1.oracleErrors.count })")
  }
}
