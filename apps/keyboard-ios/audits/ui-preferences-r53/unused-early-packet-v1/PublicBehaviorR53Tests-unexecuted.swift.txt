  // Override public UITouch observations; production KeysView owns every action.
  @MainActor
  private final class Contact: UITouch {
    unowned let surface: UIView
    var point: CGPoint
    var suppliedPhase: UITouch.Phase = .began
    var suppliedTime = ProcessInfo.processInfo.systemUptime
    init(_ surface: UIView, _ point: CGPoint) {
      self.surface = surface
      self.point = point
      super.init()
    }
    override var view: UIView? { surface }
    override var window: UIWindow? { surface.window }
    override var phase: UITouch.Phase { suppliedPhase }
    override var timestamp: TimeInterval { suppliedTime }
    override var tapCount: Int { 1 }
    override func location(in view: UIView?) -> CGPoint { surface.convert(point, to: view) }
    override func previousLocation(in view: UIView?) -> CGPoint { location(in: view) }
    func move(_ point: CGPoint) {
      self.point = point
      suppliedPhase = .moved
      suppliedTime += 0.012 // Input samples, never a timing/recognition-quality oracle.
      surface.touchesMoved([self], with: nil)
    }
    func end() {
      suppliedPhase = .ended
      suppliedTime += 0.012
      surface.touchesEnded([self], with: nil)
    }
  }

  // The sender supplies state/location to the REAL installed chooser target/action.
  // This does not claim UIKit's physical long-press recognizer recognized a touch.
  @MainActor
  private final class ToneContact: UILongPressGestureRecognizer {
    var suppliedState: UIGestureRecognizer.State = .possible
    var point = CGPoint.zero // In the mounted collection's coordinates.
    override var state: UIGestureRecognizer.State {
      get { suppliedState }
      set { suppliedState = newValue }
    }
    override func location(in view: UIView?) -> CGPoint {
      self.view?.convert(point, to: view) ?? point
    }
  }

    func key(_ matches: (Key) -> Bool) throws -> KeyView {
      guard let cap = keys.keyViews.first(where: { matches($0.key) }),

    func begin(_ cap: KeyView) -> Contact {
      contacts += 1
      let point = cap.convert(CGPoint(x: cap.bounds.midX, y: cap.bounds.midY), to: keys)
      notes.append("CONTACT key=\(cap.accessibilityLabel ?? cap.key.identifier) frame=\(cap.frame) hit=\(String(describing: keys.hitTest(point, with: nil)))")
      let contact = Contact(keys, point)
      keys.touchesBegan([contact], with: nil)
      return contact
    }

    func descendants(_ view: UIView) -> [UIView] {
      [view] + view.subviews.flatMap { descendants($0) }
    }
    func visible(_ view: UIView) -> Bool {
      guard view.window != nil, !view.bounds.isEmpty else { return false }
      var node: UIView? = view
      while let current = node {
        if current.isHidden || current.alpha <= 0.01 { return false }
        if current.clipsToBounds,
           !current.bounds.intersects(view.convert(view.bounds, to: current)) { return false }
        node = current.superview
      }
      return true
    }
    func control(_ identifier: String) throws -> UIControl {
      guard let control = descendants(controller.rootView).compactMap({ $0 as? UIControl })
        .first(where: { $0.accessibilityIdentifier == identifier && visible($0) && $0.isEnabled }) else {
        throw BindingError.unavailable("Visible enabled UIControl \(identifier)")
      }
      return control
    }
    func press(_ control: UIControl) throws {
      guard visible(control), control.isEnabled, !control.allTargets.isEmpty else {
        throw BindingError.unavailable("Unbound/hidden control")
      }
      let bindings = control.allTargets.flatMap { target in
        control.actions(forTarget: target.base, forControlEvent: .touchUpInside) ?? []
      }
      guard !bindings.isEmpty else { throw BindingError.unavailable("No installed touchUpInside action") }
      controlActions += 1
      notes.append("CONTROL id=\(control.accessibilityIdentifier ?? control.accessibilityLabel ?? "category") frame=\(control.frame) actions=\(bindings)")
      control.sendActions(for: .touchUpInside)
      layout()
    }
    func press(_ identifier: String) throws { try press(control(identifier)) }

    func chooseRegular(excluding excluded: Set<String> = [], variable: String) throws -> String {
      try category(recents: false)
      let grid = try collection()
      // At most four ordinary smiley viewports. No full-catalog glyph enumeration.
      for page in 0..<4 {
        if page > 0 {
          grid.setContentOffset(CGPoint(x: min(grid.contentOffset.x + grid.bounds.width * 0.8,
            max(0, grid.contentSize.width - grid.bounds.width)), y: 0), animated: false)
          grid.layoutIfNeeded()
        }
        for path in grid.indexPathsForVisibleItems.sorted(by: { ($0.section, $0.item) < ($1.section, $1.item) }) where path.section > 0 {
          guard let cell = grid.cellForItem(at: path) as? EmojiCell,
                let value = cell.emoji, visible(cell), cell.imageView.image != nil,
                !excluded.contains(value), !(cell.accessibilityIdentifier ?? "").hasPrefix("emoji-👍") else { continue }
          return try select(path, variable: variable)
        }
      }
      throw BindingError.unavailable("Visible ordinary non-recent/distinct fixture; no oracle substitution")
    }

    func tone(cancel: Bool, preserving text: String) throws -> String? {
      try category(recents: false)
      let grid = try collection()
      var targetPath: IndexPath?
      // Known public catalog base B=thumbs up; navigate bounded UI pages to it.
      for page in 0..<16 {
        targetPath = grid.indexPathsForVisibleItems.first { path in
          path.section > 0 && grid.cellForItem(at: path)?.accessibilityIdentifier == "emoji-👍"
        }
        if targetPath != nil { break }
        if page < 15 {
          grid.setContentOffset(CGPoint(x: min(grid.contentOffset.x + grid.bounds.width * 0.8,
            max(0, grid.contentSize.width - grid.bounds.width)), y: 0), animated: false)
          grid.layoutIfNeeded()
        }
      }
      let path = try XCTUnwrap(targetPath, "Known tone-capable public B must be rendered")
      let cell = try show(path)
      notes.append("B base=👍 displayed=\(String(reflecting: cell.emoji ?? "")) before tone interaction path=\(path)")
      let actual = try XCTUnwrap(grid.gestureRecognizers?.compactMap { $0 as? UILongPressGestureRecognizer }.first)
      // UIKit has no public recognizer target-list API. Read ONLY its installed
      // target/action metadata (not model/state). Never invoke a guessed panel callback.
      let entries = try XCTUnwrap(actual.value(forKey: "_targets") as? [NSObject])
      var binding: (NSObject, Selector)?
      for entry in entries {
        guard let target = entry.value(forKey: "_target") as? NSObject, target === panel,
              let ivar = class_getInstanceVariable(type(of: entry), "_action"),
              let encoding = ivar_getTypeEncoding(ivar), String(cString: encoding) == ":" else { continue }
        let selector = Unmanaged.passUnretained(entry).toOpaque()
          .advanced(by: ivar_getOffset(ivar)).load(as: Selector.self)
        if target.responds(to: selector) { binding = (target, selector); break }
      }
      guard let (target, action) = binding else {
        throw BindingError.unavailable("Installed skin-tone recognizer target/action metadata unavailable")
      }
      notes.append("TONE installed=\(type(of: actual)) target=\(type(of: target)) action=\(NSStringFromSelector(action)); synthetic recognizer sender, mounted collection coordinates")
      let sender = ToneContact(target: nil, action: nil)
      grid.addGestureRecognizer(sender)
      defer { grid.removeGestureRecognizer(sender) }
      sender.point = CGPoint(x: cell.frame.midX, y: cell.frame.midY)
      func dispatch(_ state: UIGestureRecognizer.State, _ name: String) {
        sender.suppliedState = state
        toneStates.append(name)
        XCTAssertTrue(UIApplication.shared.sendAction(action, to: target, from: sender, for: nil))
        layout()
      }
      dispatch(.began, "began")
      hostState(text, label: "tone chooser open; no selection yet")
      guard let panel, let popup = descendants(panel).compactMap({ $0 as? KeyPopupView }).first(where: { visible($0) && $0.options.count > 1 }) else {
        throw BindingError.unavailable("Rendered native tone chooser")
      }
      let labels = descendants(popup).compactMap { $0 as? UILabel }.filter { visible($0) && $0.text != nil }
      notes.append("TONE displayed choices=\(labels.compactMap(\.text)) frames=\(labels.map(\.frame))")
      if cancel {
        dispatch(.cancelled, "cancelled")
        XCTAssertTrue(popup.isHidden)
        return nil
      }
      let label = try XCTUnwrap(labels.first(where: { label in
        (label.text ?? "").unicodeScalars.contains { (0x1F3FB...0x1F3FF).contains($0.value) }
      }), "Displayed non-default T")
      let t = try XCTUnwrap(label.text)
      notes.append("T=\(String(reflecting: t)) recorded BEFORE change/end")
      // Finger movement is relative to the original key. Use rendered label pitch
      // and the displayed default index, then read the resulting highlighted choice.
      let ordered = labels.sorted { $0.frame.midX < $1.frame.midX }
      let base = try XCTUnwrap(ordered.firstIndex(where: { $0.text == "👍" }))
      let chosen = try XCTUnwrap(ordered.firstIndex(where: { $0 === label }))
      guard ordered.count > 1 else { throw BindingError.unavailable("Tone label pitch") }
      let pitch = ordered[1].frame.midX - ordered[0].frame.midX
      sender.point.x += CGFloat(chosen - base) * pitch
      dispatch(.changed, "changed")
      hostState(text, label: "tone highlight changed; before release")
      XCTAssertEqual(popup.selectedOption, t, "Rendered selected T before release")
      dispatch(.ended, "ended")
      XCTAssertTrue(popup.isHidden)
      return t
    }

    func moveCursor(_ contact: Contact, to caret: Int, preserving text: String) throws {
      // One-point input increments; terminate on the actual UITextView selection.
      for _ in 0..<600 {
        if editor.selectedRange == NSRange(location: caret, length: 0) { return }
        let direction: CGFloat = editor.selectedRange.location > caret ? -1 : 1
        contact.move(CGPoint(x: contact.point.x + direction, y: contact.point.y))
        XCTAssertEqual(editor.text, text, "Full host invariant during cursor movement")
      }
      throw BindingError.unavailable("Gesture did not reach requested visible caret; no assumed pixel mapping")
    }
    func trailVisible() -> Bool {
      (keys.layer.sublayers ?? []).compactMap { $0 as? CAShapeLayer }.contains { layer in
        // Inspect the mounted drawing layer, never a diagnostic counter.
        // Synchronous synthetic moves may precede the next CA presentation commit.
        let shown = layer
        return !shown.isHidden && shown.opacity > 0.01 && shown.lineWidth > 0 &&
          (shown.strokeColor?.alpha ?? 0) > 0 && !(shown.path?.isEmpty ?? true) &&
          !(shown.path?.boundingBoxOfPath.isEmpty ?? true)
      }
    }
    func slide(_ letters: String, trail: Bool? = nil) throws {
      let caps = try letters.map { letter in try key { $0.action == .character(String(letter)) } }
      let contact = begin(try XCTUnwrap(caps.first))
      var sawTrail = false
      var samples = 0
      for cap in caps.dropFirst() {
        let start = contact.point
        let end = CGPoint(x: cap.frame.midX, y: cap.frame.midY)
        for sample in 1...12 {
          let fraction = CGFloat(sample) / 12
          contact.move(CGPoint(x: start.x + (end.x - start.x) * fraction,
                               y: start.y + (end.y - start.y) * fraction))
          let drawn = trailVisible()
          samples += 1
          sawTrail = sawTrail || drawn
          if trail == false { XCTAssertFalse(drawn, "No rendered trail at moving sample \(samples)") }
        }
      }
      if trail == true { XCTAssertTrue(sawTrail, "Rendered stroked CAShapeLayer during contact") }
      notes.append("SLIDE path=\(letters) movingSamples=\(samples) sawRenderedTrail=\(sawTrail) expected=\(String(describing: trail)); no diagnostics/delegate telemetry read")
      contact.end()
      layout()
    }
  }

  private func run(_ id: String, language: KeyboardLanguage, text: String = "", caret: Int? = nil,
                   configure: (inout KeyboardSettings) -> Void = { _ in },
                   body: @MainActor (Rig) async throws -> Void) async throws {
    let rig = try Rig(language: language, text: text, caret: caret, configure: configure)
    defer {
      rig.notes.append("DENOMINATORS case=\(id) authoredCaseCount=16 boundMethodCount=16 host+caretChecks=\(rig.hostChecks) queryChecks=\(rig.queryChecks) renderedControlActions=\(rig.controlActions) collectionSelections=\(rig.selections) keyContacts=\(rig.contacts) chooserStates=\(rig.toneStates)")
      rig.notes.append("Physical click audibility/tactile haptics: NOT OBSERVABLE/NOT CLAIMED by this executor (Simulator or hardware). OS gesture delivery/installed extension and actual app Settings UI are separately owned by parent. Assertions executed before throw are retained; a thrown binding error is a failed/unbound prerequisite, never a pass or case omission.")
      let receipt = XCTAttachment(string: rig.notes.joined(separator: "\n"))
      receipt.name = "R53 \(id) native public binding receipt"
      receipt.lifetime = .keepAlways
      add(receipt)
      rig.restore()
    }
    try rig.preconditions(language)
    rig.hostState(text, caret, label: "initial")
    try await body(rig)
    let denominator: [String: (host: Int, query: Int)] = [
      "EN-01": (2, 0), "ES-01": (2, 0), "BI-01": (2, 0),
      "EN-02": (3, 0), "ES-02": (4, 0), "BI-02": (4, 0),
      "EN-03": (6, 4), "ES-03": (7, 5), "BI-03": (4, 2),
      "EN-04": (4, 0), "ES-04": (6, 0), "BI-04": (7, 2),
      "EN-05": (3, 0), "ES-05": (4, 0), "BI-05": (3, 0), "ALL-01": (7, 0)
