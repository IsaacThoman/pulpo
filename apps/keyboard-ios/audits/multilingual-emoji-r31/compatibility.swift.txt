import Foundation
@main struct Compatibility {
 static func main() throws {
  let a=CommandLine.arguments
  let catalog=try EmojiCatalog.load(url:URL(fileURLWithPath:a[1]));let search=EmojiSuggester(catalog:catalog)
  let queries=try JSONDecoder().decode([String].self,from:Data(contentsOf:URL(fileURLWithPath:a[2])))
  var result:[[String:Any]]=[]
  for q in queries { result.append(["query":q,"search":search.search(q).map(\.value),"suggestions":search.suggestions(for:q).map(\.value)]) }
  try JSONSerialization.data(withJSONObject:result,options:[.sortedKeys,.withoutEscapingSlashes]).write(to:URL(fileURLWithPath:a[3]))
  print("queries=\(queries.count)")
 }
}
