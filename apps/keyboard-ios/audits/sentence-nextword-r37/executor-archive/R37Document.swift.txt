import Foundation
@testable import KeyboardCore

final class R37Document: TextDocument {
  var text: String
  var caret: Int
  var selectionLength: Int
  var documentIdentifier: UUID? = UUID()
  var primitives: [[String: Any]] = []
  init(_ row: R37Case) {
    text = row.prefix + row.selected + row.suffix
    caret = row.prefix.count
    selectionLength = row.selected.count
  }
  var documentContextBeforeInput: String? { String(text.prefix(caret)) }
  var documentContextAfterInput: String? { String(text.dropFirst(caret + selectionLength)) }
  var selectedText: String? { selectionLength == 0 ? nil : String(text.dropFirst(caret).prefix(selectionLength)) }
  var hasText: Bool { !text.isEmpty }
  var state: [String: Any] {
    ["document": text, "caret": caret, "selectionLength": selectionLength,
     "documentID": documentIdentifier?.uuidString ?? NSNull() as Any,
     "beforeCursor": documentContextBeforeInput!, "selected": selectedText ?? "", "afterCursor": documentContextAfterInput!]
  }
  func insertText(_ inserted: String) { primitive("insertText", inserted: inserted) }
  func deleteBackward() { primitive("deleteBackward", inserted: "") }
  private func primitive(_ operation: String, inserted: String) {
    let before = state
    let start = operation == "deleteBackward" && selectionLength == 0 ? max(0, caret - 1) : caret
    let length = selectionLength > 0 ? selectionLength : (operation == "deleteBackward" && caret > 0 ? 1 : 0)
    let removed = String(text.dropFirst(start).prefix(length))
    let left = String(text.prefix(start)) + inserted
    let right = String(text.dropFirst(start + length))
    text = left + right
    // UIKit-style insertion position is UTF-16 based. Re-find a real String
    // boundary after concatenation; inserted Characters can merge with either side.
    let target = left.utf16.count
    caret = r37CaretAfterUTF16Insertion(text, target: target)
    selectionLength = 0
    primitives.append(["operation": operation, "inserted": inserted,
                       "deleted": operation == "deleteBackward" ? removed : "",
                       "replaced": operation == "insertText" ? removed : "",
                       "rangeStart": start, "rangeLength": length, "utf16InsertionEnd": target, "before": before, "after": state])
  }
  /// Independent destination/range replay of every actual document primitive.
  func verifyTrace(from index: Int, initial: [String: Any]) throws {
    var current = initial
    for primitive in primitives.dropFirst(index) {
      let before = primitive["before"] as! [String: Any]
      let after = primitive["after"] as! [String: Any]
      guard r37Exact(current["document"] as! String, before["document"] as! String),
            current["caret"] as! Int == before["caret"] as! Int,
            current["selectionLength"] as! Int == before["selectionLength"] as! Int else { throw R37Failure("primitive trace discontinuity") }
      let source = before["document"] as! String
      let oldCaret = before["caret"] as! Int
      let oldSelection = before["selectionLength"] as! Int
      let deleting = primitive["operation"] as! String == "deleteBackward"
      let expectedStart = deleting && oldSelection == 0 ? max(0, oldCaret - 1) : oldCaret
      let expectedLength = oldSelection > 0 ? oldSelection : (deleting && oldCaret > 0 ? 1 : 0)
      let removed = primitive[deleting ? "deleted" : "replaced"] as! String
      let inserted = primitive["inserted"] as! String
      let characters = Array(source)
      guard expectedStart == primitive["rangeStart"] as! Int,
            expectedLength == primitive["rangeLength"] as! Int,
            r37Exact(removed, String(characters[expectedStart..<(expectedStart + expectedLength)])) else { throw R37Failure("primitive destination/removal mismatch") }
      let expectedPrefix = String(characters[..<expectedStart]) + inserted
      let expected = expectedPrefix + String(characters[(expectedStart + expectedLength)...])
      guard r37Exact(expected, after["document"] as! String),
            after["caret"] as! Int == r37CaretAfterUTF16Insertion(expected, target: expectedPrefix.utf16.count),
            after["selectionLength"] as! Int == 0 else { throw R37Failure("primitive replay document/range mismatch") }
      current = after
    }
    guard r37Exact(current["document"] as! String, text), current["caret"] as! Int == caret,
          current["selectionLength"] as! Int == selectionLength else { throw R37Failure("primitive trace final mismatch") }
  }
}

/// Snap forward only when insertion coalesces across the right boundary.
/// A combining mark appended to a base therefore leaves caret 1, not 2.
func r37CaretAfterUTF16Insertion(_ text: String, target: Int) -> Int {
  for (offset, index) in text.indices.enumerated() where index.utf16Offset(in: text) >= target { return offset }
  return text.count
}
