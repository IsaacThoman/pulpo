import Foundation
import Testing
@testable import KeyboardCore

/// Independent source-review probes, not a new lexical accuracy corpus.
@Suite(.serialized)
struct R16SourceReviewTests {
  private final class Rig {
    let engine: KeyboardEngine
    let host: ExplicitSpaceDocument
    var time = 100.0

    init(_ language: KeyboardLanguage, prefix: String = "old ", suffix: String = "suffix",
         selection: String = "", timing: String = "immediate", capitalizes: Bool = false) throws {
      let lexicon = try Lexicon(url: Fixtures.languageData.appendingPathComponent(language.rawValue + ".pkdict"))
      let model = LanguageModel(lexicon: lexicon, personal: PersonalDictionary(url: nil))
      let corrector = SpellCorrector(lexicon: lexicon)
      corrector.touch = TouchModel(url: nil)
      var settings = KeyboardSettings()
      settings.language = language
      host = ExplicitSpaceDocument(prefix + selection + suffix, caret: prefix.count,
                                   selection: selection.count, timing: timing)
      engine = KeyboardEngine(document: host, suggester: Suggester(model: model, corrector: corrector),
                              glide: GlideDecoder(model: model), settings: settings)
      var traits = InputTraits()
      if !capitalizes { traits.capitalization = .none }
      engine.traits = traits
      engine.clock = { [unowned self] in self.time }
      engine.documentDidChange()
      #expect(model.personal.totalCount == 0)
      #expect(corrector.touch?.weight == 0)
    }

    func type(_ word: String, uppercaseFirst: Bool = false) {
      for (index, letter) in word.enumerated() {
        time += host.timing == "old-own" ? 2 : 0.04
        host.perform(.letter(letter, casing: uppercaseFirst && index == 0 ? .uppercase : .asEntered)) {
          engine.type(String(letter))
        }
        if host.timing == "old-own" {
          // Acknowledge the first write; then keep the report one whole key
          // behind. Reporting just before each key would merely acknowledge
          // the current expected state and would never exercise actual lag.
          if index == 0 { host.reported = host.state; engine.documentDidChange() }
          else { host.reportNext(); engine.documentDidChange() }
        }
        if host.timing == "immediate" { engine.documentDidChange() }
        if host.timing == "optimistic", !host.inbox.isEmpty {
          host.reportNext(); engine.documentDidChange()
          host.flush(); engine.documentDidChange()
        }
      }
    }

    func space() { host.perform(.explicitSpace) { engine.insertSpace() } }
    func flush() { host.flush(); engine.documentDidChange() }
    func hostChange(_ text: String, caret: Int, selection: Int = 0, callback: Bool = true) {
      host.flush()
      host.state = .init(text: Array(text), caret: caret, selection: selection)
      host.hostChanged(); host.flush()
      if callback { engine.documentDidChange() }
    }
  }

  private func same(_ lhs: String, _ rhs: String) -> Bool {
    lhs.unicodeScalars.elementsEqual(rhs.unicodeScalars)
  }

  /// Independently inspect EVERY physical snapshot, beyond final-state equality.
  private func bounded(_ steps: ArraySlice<ExplicitSpaceDocument.Primitive>, prefix: String, suffix: String) {
    for step in steps {
      for state in [step.before, step.after] {
        #expect(state.selection == 0)
        #expect(state.caret >= prefix.count)
        #expect(same(String(state.text.prefix(prefix.count)), prefix))
        #expect(same(state.after, suffix))
      }
    }
  }

  @Test(arguments: KeyboardLanguage.allCases)
  func pendingTypoThenSwipeWithoutSpaceNeverDeletesTypedOrInheritedGlyphs(language: KeyboardLanguage) throws {
    for suffix in ["teh", "hello", "hola", "42tail", "e\u{301}clair"] {
      let rig = try Rig(language, suffix: suffix)
      rig.type("teh")
      let start = rig.host.primitives.count
      var random = SeededGenerator(state: 123)
      let word = language == .spanish ? "hola" : "hello"
      var candidates: [GlideCandidate] = []
      rig.host.perform(.swipe) {
        candidates = rig.engine.insertGlide(SyntheticSwipe.points(for: word, noise: 0, generator: &random))
      }
      rig.flush()
      #expect(!candidates.isEmpty)
      #expect(rig.host.primitives[start...].allSatisfy { $0.inserted != nil })
      #expect(rig.host.state.before.hasPrefix("old teh "))
      bounded(rig.host.primitives[start...], prefix: "old teh", suffix: suffix)
      #expect(rig.host.conservationFailures.isEmpty)
    }
  }

  @Test(arguments: KeyboardLanguage.allCases, [",", ".", "\n", "external"])
  func genericTerminatorsKeepPendingTypo(language: KeyboardLanguage, terminator: String) throws {
    let rig = try Rig(language, suffix: "tehñandú")
    rig.type("teh")
    let start = rig.host.primitives.count
    if terminator == "external" {
      rig.host.perform(.insertText) { rig.engine.insertText("🙂") }
    } else if terminator == "\n" {
      rig.host.perform(.returnKey) { rig.engine.insertReturn() }
    } else {
      rig.host.perform(.punctuation) { rig.engine.insertCharacter(terminator) }
    }
    rig.flush()
    #expect(rig.host.primitives[start...].allSatisfy { $0.inserted != nil })
    #expect(same(String(rig.host.state.text), "old teh" + (terminator == "external" ? "🙂" : terminator) + "tehñandú"))
    bounded(rig.host.primitives[start...], prefix: "old teh", suffix: "tehñandú")
    #expect(rig.host.conservationFailures.isEmpty)
  }

  @Test(arguments: KeyboardLanguage.allCases)
  func inheritedFragmentsExternalInsertionAndSelectionOriginCannotBecomeOwned(language: KeyboardLanguage) throws {
    let fragment = try Rig(language, prefix: "old t")
    fragment.type("eh"); fragment.space(); fragment.flush()
    #expect(String(fragment.host.state.text) == "old teh suffix")
    #expect(fragment.host.primitives.allSatisfy { $0.inserted != nil })

    let external = try Rig(language)
    external.host.perform(.insertText) { external.engine.insertText("teh") }
    external.space(); external.flush()
    #expect(String(external.host.state.text) == "old teh suffix")
    #expect(external.host.primitives.allSatisfy { $0.inserted != nil })

    let selection = try Rig(language, selection: "SELECTED")
    selection.type("teh"); selection.space(); selection.flush()
    #expect(String(selection.host.state.text) == "old teh suffix")
    #expect(selection.host.state.caret == 8)
    #expect(selection.host.state.selection == 0)
    #expect(selection.host.primitives.allSatisfy { $0.inserted != nil })
    for rig in [fragment, external, selection] { #expect(rig.host.conservationFailures.isEmpty) }
  }

  @Test(arguments: KeyboardLanguage.allCases, [false, true])
  func realHostChangesRevokeOwnershipWithAndWithoutCallback(language: KeyboardLanguage, callback: Bool) throws {
    for changed in [("changed tehsuffix", 11), ("old tehDIFFERENT", 7)] {
      let rig = try Rig(language)
      rig.type("teh")
      rig.hostChange(changed.0, caret: changed.1, callback: callback)
      let start = rig.host.primitives.count
      rig.space(); rig.flush()
      #expect(rig.host.primitives[start...].allSatisfy { $0.inserted != nil })
      #expect(rig.host.state.before.hasSuffix("teh "))
      #expect(rig.host.conservationFailures.isEmpty)
    }
    let selected = try Rig(language)
    selected.type("teh")
    selected.hostChange("old tehsuffix", caret: 7, selection: 2, callback: callback)
    let start = selected.host.primitives.count
    selected.space(); selected.flush()
    #expect(String(selected.host.state.text) == "old teh ffix")
    #expect(selected.host.primitives[start...].allSatisfy { $0.inserted != nil })
    #expect(selected.host.conservationFailures.isEmpty)
  }

  @Test(arguments: KeyboardLanguage.allCases)
  func defaultSentenceCapitalizationPreservesInheritedUnicode(language: KeyboardLanguage) throws {
    let rig = try Rig(language, prefix: "", suffix: "e\u{301}clair", capitalizes: true)
    rig.type("teh", uppercaseFirst: true)
    let start = rig.host.primitives.count
    rig.space(); rig.flush()
    bounded(rig.host.primitives[start...], prefix: "", suffix: "e\u{301}clair")
    #expect(rig.host.state.before.first?.isUppercase == true)
    #expect(rig.host.state.before.hasSuffix(" "))
    #expect(rig.host.conservationFailures.isEmpty)
  }

  @Test(arguments: KeyboardLanguage.allCases)
  func unreportedMoveIntoRepeatedInheritedSuffixCannotDeleteOldWord(language: KeyboardLanguage) throws {
    let rig = try Rig(language, suffix: "tailold tehtail")
    rig.type("teh")
    // Actual caret moves past inherited "tailold teh" before the callback.
    // Both short live contexts resemble the prior context window: the new
    // before ends in "old teh" and the new after "tail" prefixes the old after.
    let inheritedPrefix = "old tehtailold teh"
    rig.hostChange(inheritedPrefix + "tail", caret: inheritedPrefix.count, callback: false)
    let start = rig.host.primitives.count
    rig.space(); rig.flush()
    #expect(rig.host.primitives[start...].allSatisfy { $0.inserted != nil })
    #expect(same(String(rig.host.state.text), "old tehtailold teh tail"))
    #expect(rig.host.state.caret == inheritedPrefix.count + 1)
    #expect(rig.host.conservationFailures.isEmpty)
  }
}
