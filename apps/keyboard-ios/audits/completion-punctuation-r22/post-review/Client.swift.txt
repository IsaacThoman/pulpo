import Foundation
import KeyboardCore
struct Item: Codable { let id, before, prefix, suffix, expected: String; let smart: Bool? }
struct Row: Codable { let id, stack, expected, output: String; let exact, available, suffixPreserved: Bool; let caret, selection: Int; let slots: [String] }
final class Document: TextDocument {
 var text: [Character]; var cursor: Int; var selection = 0
 let documentIdentifier: UUID? = UUID()
 init(_ item: Item) { text = Array(item.before + item.suffix); cursor = item.before.count }
 var documentContextBeforeInput: String? { String(text.prefix(cursor)) }
 var documentContextAfterInput: String? { String(text.dropFirst(cursor + selection)) }
 var selectedText: String? { selection > 0 ? String(text.dropFirst(cursor).prefix(selection)) : nil }
 var hasText: Bool { !text.isEmpty }
 func insertText(_ value: String) { text.replaceSubrange(cursor..<(cursor + selection), with: Array(value)); cursor += value.count; selection = 0 }
 func deleteBackward() { if selection > 0 { text.removeSubrange(cursor..<(cursor + selection)); selection = 0 } else if cursor > 0 { text.remove(at: cursor - 1); cursor -= 1 } }
}
let args = CommandLine.arguments
let items = try JSONDecoder().decode([Item].self, from: Data(contentsOf: URL(fileURLWithPath: args[1])))
var rows: [Row] = []
for language in KeyboardLanguage.allCases {
 for item in items {
  let stack = try LanguageStack(dictionaryURL: URL(fileURLWithPath: args[2]).appendingPathComponent(language.rawValue + ".pkdict"), personalURL: nil)
  let document = Document(item)
  var settings = KeyboardSettings(); settings.language = language; settings.autoCapitalization = false
  if let smart = item.smart { settings.smartPunctuation = smart }
  let engine = KeyboardEngine(document: document, suggester: stack.suggester, glide: stack.glide, settings: settings)
  engine.documentDidChange()
  for character in item.prefix { engine.insertCharacter(String(character)) }
  let slots = engine.suggestions.slots
  let chosen = slots.first { $0.kind == .literal }
  if let chosen { engine.select(chosen) }
  let output = String(document.text)
  rows.append(Row(id: item.id, stack: language.rawValue, expected: item.expected, output: output, exact: chosen != nil && output.utf8.elementsEqual(item.expected.utf8), available: chosen != nil, suffixPreserved: document.documentContextAfterInput == item.suffix, caret: document.cursor, selection: document.selection, slots: slots.map { "\($0.kind)|\($0.text)|\($0.isAutocorrection)" }))
 }
}
let encoder = JSONEncoder(); encoder.outputFormatting = [.prettyPrinted,.sortedKeys,.withoutEscapingSlashes]
try encoder.encode(rows).write(to: URL(fileURLWithPath: args[3]), options: .atomic)
print("\(rows.filter(\.exact).count)/\(rows.count) exact; \(args[3])")
