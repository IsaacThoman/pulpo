import Foundation
import Testing
@testable import KeyboardCore

/// Opt-in frozen, synthetic R15 engine replay; no runtime/device dependency.
struct TechnicalPreservationReplayTests {
  struct Case: Decodable {
    let id: String
    let group: String
    let before: String
    let selected: String
    let after: String
    let typed: String
    let expected: String
    let field: String?
    let delete: Int?
    let moveBefore: String?
    let moveAfter: String?
    let undo: Bool?
  }

  final class Document: TextDocument {
    let documentIdentifier: UUID? = UUID()
    var prefix: String
    var selection: String
    var suffix: String
    init(_ item: Case) { prefix = item.before; selection = item.selected; suffix = item.after }
    var documentContextBeforeInput: String? { prefix }
    var documentContextAfterInput: String? { suffix }
    var selectedText: String? { selection.isEmpty ? nil : selection }
    var hasText: Bool { !text.isEmpty }
    var text: String { prefix + selection + suffix }
    func insertText(_ text: String) { selection = ""; prefix += text }
    func deleteBackward() {
      if !selection.isEmpty { selection = "" }
      else if !prefix.isEmpty { prefix.removeLast() }
    }
  }

  struct Replay: Codable {
    let output: String
    let beforeDelimiter: String
    let alternatives: [String]
    let autocorrection: String?
    let learned: [String]
    let touchWeight: Double
    let keyMilliseconds: [Double]
  }
  struct Outcome: Codable {
    let id: String
    let group: String
    let mode: String
    let expected: String
    let correct: Bool
    let rawExpected: String
    let rawCorrect: Bool
    let privacySafe: Bool?
    let automatic: Replay
    let raw: Replay
  }

  static func tap(_ character: Character, engine: KeyboardEngine) {
    if character == " " { engine.insertSpace(tap: KeyPoint(x: 5, y: 3)); return }
    if character == "\n" { engine.insertReturn(); return }
    let key = String(character).lowercased()
    if character.isUppercase, !engine.shift.isActive { engine.shiftDown(); engine.shiftUp() }
    let point = character.isLetter && key.utf8.count == 1
      ? key.utf8.first.flatMap(LetterGeometry.qwerty.center) : nil
    engine.insertCharacter(key, tap: point)
  }

  static func replay(_ item: Case, language: KeyboardLanguage, autocorrect: Bool) throws -> Replay {
    var settings = KeyboardSettings()
    settings.language = language
    settings.autocorrect = autocorrect
    let stack = try LanguageStack(dictionaryURL: Fixtures.languageData.appendingPathComponent(language.rawValue + ".pkdict"), personalURL: nil)
    let document = Document(item)
    let engine = KeyboardEngine(document: document, suggester: stack.suggester, glide: stack.glide, settings: settings)
    var traits = InputTraits()
    switch item.field {
    case "secure": traits.isSecure = true
    case "email": traits.keyboard = .email
    case "url": traits.keyboard = .url
    case "oneTimeCode": traits.contentType = "oneTimeCode"
    default: break
    }
    engine.traits = traits
    engine.documentDidChange()
    for _ in 0..<(item.delete ?? 0) { engine.deleteBackward() }
    if let prefix = item.moveBefore {
      document.prefix = prefix
      document.selection = ""
      document.suffix = item.moveAfter ?? ""
      engine.documentDidChange()
    }
    var times: [Double] = []
    var alternatives: [String] = []
    var correction: String?
    var beforeDelimiter = document.text
    for (index, character) in item.typed.enumerated() {
      if index == item.typed.count - 1 {
        beforeDelimiter = document.text
        alternatives = engine.suggestions.slots.map(\.text)
        correction = engine.suggestions.autocorrection
      }
      let start = ProcessInfo.processInfo.systemUptime
      tap(character, engine: engine)
      _ = engine.suggestions
      times.append((ProcessInfo.processInfo.systemUptime - start) * 1_000)
    }
    if item.undo == true { engine.deleteBackward() }
    engine.save(inBackground: false)
    return Replay(output: document.text, beforeDelimiter: beforeDelimiter, alternatives: alternatives,
                  autocorrection: correction, learned: stack.personal.usedEntries.map(\.text), touchWeight: stack.touch.weight, keyMilliseconds: times)
  }

  /// Separate literal-edit oracle. Sentence casing is retained in raw mode;
  /// these synthetic inputs contain no quote/double-space transforms.
  static func rawExpected(_ item: Case) -> String {
    var prefix = item.before
    var selection = item.selected
    for _ in 0..<(item.delete ?? 0) {
      if !selection.isEmpty { selection = "" }
      else if !prefix.isEmpty { prefix.removeLast() }
    }
    prefix = item.moveBefore ?? prefix
    let suffix = item.moveBefore == nil ? item.after : item.moveAfter ?? ""
    for character in item.typed {
      var text = String(character)
      if character.isLetter, item.field != "email", item.field != "url" {
        // Ignore sentence-opening punctuation before examining the boundary.
        var boundary = prefix
        while let last = boundary.last, "([{¿¡“‘".contains(last) { boundary.removeLast() }
        let hasSpace = boundary.last?.isWhitespace == true
        let followsNewline = boundary.reversed().prefix(while: \.isWhitespace).contains(where: \.isNewline)
        boundary = boundary.trimmingCharacters(in: .whitespacesAndNewlines)
        while let last = boundary.last, ")]}'\"’”".contains(last) { boundary.removeLast() }
        if boundary.isEmpty || followsNewline ||
            (hasSpace && boundary.last.map { ".!?".contains($0) } == true) {
          text = text.uppercased()
        }
      }
      prefix += text
    }
    if item.undo == true, !prefix.isEmpty { prefix.removeLast() }
    return prefix + suffix
  }

  @Test func frozenReplay() throws {
    let environment = ProcessInfo.processInfo.environment
    guard let path = environment["PK_R15_CASES"] else { return }
    let reportPath = try #require(environment["PK_R15_REPORT"])
    let cases = try JSONDecoder().decode([Case].self, from: Data(contentsOf: URL(fileURLWithPath: path)))
    #expect(Set(cases.map(\.id)).count == cases.count)
    var outcomes: [Outcome] = []
    for language in KeyboardLanguage.allCases {
      for item in cases {
        let automatic = try Self.replay(item, language: language, autocorrect: true)
        let raw = try Self.replay(item, language: language, autocorrect: false)
        let rawExpected = Self.rawExpected(item)
        let privacySafe: Bool? = item.field == nil ? nil : automatic.learned.isEmpty && automatic.touchWeight == 0
        if let privacySafe { #expect(privacySafe, "privacy/learning: \(item.id)/\(language.rawValue)") }
        outcomes.append(Outcome(id: item.id, group: item.group, mode: language.rawValue,
                                expected: item.expected, correct: automatic.output == item.expected,
                                rawExpected: rawExpected, rawCorrect: raw.output == rawExpected, privacySafe: privacySafe,
                                automatic: automatic, raw: raw))
      }
    }
    let encoder = JSONEncoder()
    encoder.outputFormatting = [.prettyPrinted, .sortedKeys, .withoutEscapingSlashes]
    try encoder.encode(outcomes).write(to: URL(fileURLWithPath: reportPath), options: .atomic)
    print("R15: \(outcomes.filter(\.correct).count)/\(outcomes.count) exact, raw \(outcomes.filter(\.rawCorrect).count)/\(outcomes.count); \(reportPath)")
  }
}
