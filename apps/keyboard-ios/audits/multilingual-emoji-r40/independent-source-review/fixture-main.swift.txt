import Foundation
func child(_ value: Any, _ key: String) -> Any { Mirror(reflecting: value).children.first { $0.label == key }!.value }
func indexRows(_ suggester: EmojiSuggester) -> [[String: Any]] {
  Mirror(reflecting: child(suggester,"index")).children.map { item in
    let key = child(item.value,"key") as! String
    let matches = Mirror(reflecting: child(item.value,"matches")).children.map { m in ["id": child(m.value,"id") as! Int,"rank":child(m.value,"rank") as! Int] }
    return ["key":key,"matches":matches]
  }
}
let args = CommandLine.arguments
let catalog = try EmojiCatalog.load(url: URL(fileURLWithPath: args[2]))
let suggester = EmojiSuggester(catalog: catalog)
let entries = catalog.categories.flatMap(\.emoji)
func emit(_ value: Any) { let d = try! JSONSerialization.data(withJSONObject:value,options:[.sortedKeys,.fragmentsAllowed]); FileHandle.standardOutput.write(d); FileHandle.standardOutput.write(Data([10])) }
let preferences: [[String:Int]] = [[:],Dictionary(entries.enumerated().map { ($0.element.value,$0.offset % 3) },uniquingKeysWith:max),Dictionary(entries.enumerated().map { ($0.element.value, $0.offset % 2 == 0 ? Int.max : Int.min) },uniquingKeysWith:max)]
switch args[1] {
case "index":
  emit(["index":indexRows(suggester),"exactNames":child(suggester,"exactNames"),"byValue":child(suggester,"byValue")])
case "compat":
  let queries = try JSONDecoder().decode([String].self,from:Data(contentsOf:URL(fileURLWithPath:args[3])))
  for q in queries {
    let contextual = suggester.suggestions(for:q,limit:12).map(\.value)
    let exact = suggester.suggestions(for:q,limit:12,prefixes:false).map(\.value)
    for cap in [1,48,60] { for (p,pref) in preferences.enumerated() {
      emit(["q":q,"cap":cap,"pref":p,"search":suggester.search(q,limit:cap,preferring:pref).map(\.value),"contextual":contextual,"exactContextual":exact])
    }}
  }
case "fixtures":
  let rows = try JSONSerialization.jsonObject(with:Data(contentsOf:URL(fileURLWithPath:args[3]))) as! [[String:Any]]
  for row in rows {
    let c = try JSONDecoder().decode(EmojiCatalog.self,from:JSONSerialization.data(withJSONObject:row["catalog"]!))
    let s = EmojiSuggester(catalog:c)
    let q = row["q"] as! String
    let cap = row["cap"] as! Int
    let pref = row["preferred"] as? [String:Int] ?? [:]
    emit(["id":row["id"]!,"q":q,"cap":cap,"search":s.search(q,limit:cap,preferring:pref).map(\.value),"index":indexRows(s),"exactNames":child(s,"exactNames"),"oracle":policy(row,indexRows(s))])
  }
default: fatalError("unknown mode")
}
