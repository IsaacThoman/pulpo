import XCTest

/// Frozen public-v2 workload, source base 9ff451f4. No preview/controller harness.
/// Exactly twelve independent tests: four EN, four ES, four BI.
///
/// This is an explicitly HUMAN-ASSISTED native fixture. Before a future run, set
/// PK_R53_V2_HANDOFF_DIR to a directory shared with the visual observer. Each
/// uniquely named *.request.json and *.png is published there; the observer
/// atomically writes the corresponding *.response.json. Requests describe the
/// required fields. Every response includes observer, evidence (screenshot/video
/// reference and adjudication), and available: true. available: false plus an
/// obstacle records an unavailable binding and skips, rather than passing it.
/// No response also skips as mechanically unavailable; the timeout is an operator
/// handoff timeout, NOT an application performance oracle. No supplied responses,
/// observations, or test results are bundled with this fixture.
///
/// Human cursor gestures are ONE continuous hold/slide/release stopped by the
/// visible host caret, never a calibrated distance, count, or repair afterwards.
/// Emoji values are transcribed IN FULL before tapping the observed cell. Grid
/// identifiers are locators only, never the displayed-variant oracle. Trail
/// observations require an actual recording covering the synthesized gesture.
/// The only event synthesis used is the existing, expressly permitted TouchPath.
/// Host text is read solely from system-text's public accessibility value.
@MainActor
final class PublicBehaviorR53V2Tests: XCTestCase {
  private var app: XCUIApplication!
  private var caseID = ""
  private var handoff: URL!
  private var settings: [String: Bool] = [:]
  private let feedbackAndSlide = ["Slide to Type", "Show Slide Trail", "Sound", "Haptic Feedback"]

  override func setUp() {
    continueAfterFailure = false
    app = XCUIApplication()
  }

  private var host: XCUIElement { app.textViews["system-text"] }
  private var text: String { host.value as? String ?? "" }
  private func literalEqual(_ lhs: String, _ rhs: String) -> Bool {
    lhs.utf16.elementsEqual(rhs.utf16)
  }
  private func ui(_ id: String) -> XCUIElement {
    app.descendants(matching: .any).matching(identifier: id).firstMatch
  }
  private func require(_ condition: Bool, _ message: String) throws {
    guard condition else {
      XCTFail(message)
      throw FixtureStop.failed
    }
  }
  private enum FixtureStop: Error { case failed }
  private func note(_ name: String, _ value: String) {
    let attachment = XCTAttachment(string: value)
    attachment.name = "\(caseID)-\(name)"
    attachment.lifetime = .keepAlways
    add(attachment)
  }
  private func image(_ name: String) {
    let attachment = XCTAttachment(screenshot: XCUIScreen.main.screenshot())
    attachment.name = "\(caseID)-\(name)"
    attachment.lifetime = .keepAlways
    add(attachment)
  }

  /// No app mutation or hidden inspection occurs while awaiting human input.
  /// A unique nonce prevents reuse of an earlier checkpoint's response.
  private func human(_ stage: String, _ instruction: String,
                     fields: String, details: [String: Any] = [:]) throws -> [String: Any] {
    let token = "\(caseID)-\(stage)-\(UUID().uuidString)"
    let requestURL = handoff.appendingPathComponent(token + ".request.json")
    let responseURL = handoff.appendingPathComponent(token + ".response.json")
    let screenshotURL = handoff.appendingPathComponent(token + ".png")
    try XCUIScreen.main.screenshot().pngRepresentation.write(to: screenshotURL, options: .atomic)
    var request = details
    request["token"] = token
    request["case"] = caseID
    request["stage"] = stage
    request["instruction"] = instruction
    request["requiredResponseFields"] = fields
    request["commonResponseFields"] = "token (exact request token), available (Bool), observer (nonempty String), evidence (nonempty String); if unavailable, obstacle (nonempty String)."
    request["screenshot"] = screenshotURL.path
    request["responsePath"] = responseURL.path
    let requestData = try JSONSerialization.data(withJSONObject: request, options: [.prettyPrinted, .sortedKeys])
    note(stage + "-request", String(decoding: requestData, as: UTF8.self))
    try requestData.write(to: requestURL, options: .atomic)
    let deadline = Date().addingTimeInterval(1800)
    while !FileManager.default.fileExists(atPath: responseURL.path), Date() < deadline {
      RunLoop.current.run(until: Date().addingTimeInterval(0.25))
    }
    guard FileManager.default.fileExists(atPath: responseURL.path) else {
      throw XCTSkip("Unavailable human observation/action binding: \(stage); request \(requestURL.path). No behavioral pass is claimed.")
    }
    let data = try Data(contentsOf: responseURL)
    note(stage + "-human-record", String(decoding: data, as: UTF8.self))
    let response = try JSONSerialization.jsonObject(with: data) as? [String: Any] ?? [:]
    try require(response["token"] as? String == token, "Wrong human handoff token")
    try require(!(response["observer"] as? String ?? "").isEmpty, "Missing observer")
    try require(!(response["evidence"] as? String ?? "").isEmpty, "Missing visual evidence/adjudication")
    guard response["available"] as? Bool == true else {
      let obstacle = response["obstacle"] as? String ?? ""
      try require(!obstacle.isEmpty, "Unavailable observation must identify its exact obstacle")
      throw XCTSkip("Unavailable public binding at \(stage): \(obstacle). No behavioral pass is claimed.")
    }
    return response
  }

  private func checkpoint(_ stage: String, before: String, after: String = "") throws {
    let expected = before + after
    try require(host.value is String, "Host's public literal value is unavailable")
    let condition = NSPredicate { [weak self] _, _ in
      guard let self else { return false }
      return self.literalEqual(self.text, expected)
    }
    _ = XCTWaiter.wait(for: [XCTNSPredicateExpectation(predicate: condition, object: nil)], timeout: 4)
    try require(literalEqual(text, expected), "\(stage): exact host mismatch. Expected \(expected.debugDescription), got \(text.debugDescription)")
    image(stage)
    let response = try human(stage + "-caret", "Observe the ordinary host UI immediately, without changing focus, text, or selection. Record the actual collapsed caret as the full text before and after it. A screenshot alone does not prove a caret when its blink is invisible; cite a visible frame or live observation. Do not infer the caret from host length or the expected value.", fields: "observedHostText (String), caretBefore (String), caretAfter (String), selectionCollapsed (Bool)", details: ["expectedHost": expected, "expectedCaretBefore": before, "expectedCaretAfter": after])
    for field in ["observedHostText", "caretBefore", "caretAfter"] {
      try require(response[field] is String, "Missing explicit visual observation: \(field)")
    }
    try require(literalEqual(response["observedHostText"] as? String ?? "", expected), "Human host observation differs")
    try require(literalEqual(response["caretBefore"] as? String ?? "", before) && literalEqual(response["caretAfter"] as? String ?? "", after) && response["selectionCollapsed"] as? Bool == true, "Visible host caret/selection differs")
    try require(literalEqual(text, expected), "Observation changed host text")
  }

  private func tap(_ id: String) throws {
    let element = ui(id)
    try require(element.waitForExistence(timeout: 5) && element.isHittable, "Public control unavailable: \(id)")
    element.tap()
  }
  private func letters(_ value: String) throws {
    for character in value { try tap(character == " " ? "key-space" : "key-\(character)") }
  }
  private func delete(_ count: Int = 1) throws {
    for _ in 0..<count { try tap("key-delete") }
  }
  private func selectPulpo() throws {
    // The preview is never accepted as an installed-extension substitute.
    try require(!ui("preview-text").exists, "Unexpected in-app preview")
    for _ in 0..<8 {
      if ui("key-q").waitForExistence(timeout: 1) { break }
      let next = app.buttons["Next keyboard"].firstMatch
      guard next.exists && next.isHittable else { break }
      next.tap()
    }
    try require(ui("key-q").waitForExistence(timeout: 4), "Installed Pulpo unavailable via Next keyboard; no fallback")
  }
  private func backToSettings() throws {
    // App navigation, not a URL, controller, or preference-store write.
    let back = app.navigationBars.buttons.element(boundBy: 0)
    try require(back.exists && back.isHittable, "No public Try It back button")
    back.tap()
    try require(app.navigationBars["Keyboard"].waitForExistence(timeout: 5), "App Settings did not open")
  }
  private func scrollTop() { for _ in 0..<12 { app.swipeDown() } }
  private func setting(_ label: String) throws -> XCUIElement {
    // SwiftUI combines the exact title and its explanatory subtitle in AX.
    let element = app.switches.matching(NSPredicate(format: "label == %@ OR label BEGINSWITH %@ OR label BEGINSWITH %@", label, label + ",", label + "\n")).firstMatch
    scrollTop()
    for _ in 0..<16 {
      if element.exists {
        // Retained Show Slide Trail remains an observable value when disabled.
        // Reading it must not require that the disabled switch accept a tap.
        let visibleDisabled = !element.isEnabled && !element.frame.isEmpty && app.frame.contains(element.frame)
        if element.isHittable || visibleDisabled { return element }
      }
      app.swipeUp()
    }
    try require(false, "Public app switch unavailable: \(label)")
    return element
  }
  private func set(_ label: String, _ on: Bool) throws {
    let element = try setting(label)
    let value = on ? "1" : "0"
    if element.value as? String != value {
      try require(element.isEnabled, "Cannot change disabled app switch \(label)")
      element.tap()
    }
    try require(element.value as? String == value, "Displayed \(label) differs from \(value)")
    settings[label] = on
    image("setting-\(label)-\(value)")
  }
  private func observeSettings() throws {
    for label in feedbackAndSlide {
      let row = try setting(label)
      try require(row.value as? String == (settings[label] == true ? "1" : "0"), "Independent displayed setting changed: \(label)")
      image("displayed-\(label)")
    }
  }
  private func returnToHost(_ stage: String) throws {
    scrollTop()
    try tap("try-it")
    try require(host.waitForExistence(timeout: 5), "Installed host unavailable on return")
    let focus = try human(stage + "-focus", "If the host is not editing, activate it through ordinary host/accessibility editing while retaining its saved insertion point (for example VoiceOver edit activation). Do NOT place or repair the caret, replace text, or type. Do NOT select a new language or change settings. If the available public focus method moves the caret, record that obstacle as unavailable; do not restore the expected position. The next checkpoint observes the actual caret.", fields: "focusPreservedSelection (Bool)")
    try require(focus["focusPreservedSelection"] as? Bool == true, "Public focus did not preserve the host selection")
    // Native activation must preserve the selection; never a coordinate tap
    // followed by repair. Select Pulpo via the public OS switcher if necessary.
    try selectPulpo()
  }

  private func begin(_ id: String, language: String, initial: String, caretBefore: String,
                     slide: Bool = true, trail: Bool = true) throws {
    caseID = id
    guard let directory = ProcessInfo.processInfo.environment["PK_R53_V2_HANDOFF_DIR"], !directory.isEmpty else {
      throw XCTSkip("Human public-UI binding unavailable: PK_R53_V2_HANDOFF_DIR is unset. Fixture is not a fully automated caret/trail test.")
    }
    handoff = URL(fileURLWithPath: directory, isDirectory: true)
    try FileManager.default.createDirectory(at: handoff, withIntermediateDirectories: true)
    // No PKResetLearning, history deletion, account route, transcript, or seed.
    app.launchArguments = ["-PKPreview", "-PKSystemKeyboard", "-PKResetSettings"]
    app.launch()
    try require(host.waitForExistence(timeout: 8), "Installed plain-text Try It host missing")
    try backToSettings()
    let prerequisites = try human("setup-public-system-controls", "Before numbered actions, verify the installed Pulpo keyboard is enabled and has Full Access so app switches can reach it. Through ordinary iOS Settings > General > Keyboard UI, establish host/system Auto-Capitalization, Auto-Correction, Smart Punctuation and the period shortcut OFF. Return to this app's Settings. Do not clear learned words, emoji history, any defaults domain, account, or transcript. If those public controls/permissions are unavailable, report the precise obstacle. This prepares the authored fixture; it does not adjudicate candidate behavior.", fields: "installedEnabled (Bool), fullAccess (Bool), hostTransformsOff (Bool)")
    for field in ["installedEnabled", "fullAccess", "hostTransformsOff"] {
      try require(prerequisites[field] as? Bool == true, "Public setup not established: \(field)")
    }
    for label in ["Auto-Correction", "Auto-Capitalization", "“.” Shortcut", "Smart Punctuation", "Learn New Words"] { try set(label, false) }
    // Preparation may enable Slide briefly to establish the disabled trail row.
    try set("Slide to Type", true)
    try set("Show Slide Trail", trail)
    try set("Slide to Type", slide)
    try set("Sound", false)
    try set("Haptic Feedback", false)
    try observeSettings()
    try returnToHost("setup")
    try tap("keyboard-language")
    let choice = app.buttons[language].firstMatch
    try require(choice.waitForExistence(timeout: 4) && choice.isHittable, "Exact public language choice unavailable: \(language)")
    choice.tap()
    try require(ui("keyboard-language").label.contains(language), "Displayed language mismatch")
    // Fresh scene normally has an empty draft; public Clear handles any draft.
    let clear = app.navigationBars.buttons["Clear"]
    if clear.exists && clear.isEnabled { clear.tap() }
    try require(text.isEmpty, "Public Clear did not empty host")
    try letters(initial)
    let prepared = try human("setup-ordinary-host-caret", "Prepare ONLY the initial caret through ordinary host editing: use the text view's normal touch/magnifier/selection UI and collapse at the stated boundary. Do not use a delegate, controller, private state, space-slide feature, or preference write. Text must remain literal. Finish with Pulpo letter keys visible. This is setup before action 1.", fields: "ordinaryHostEditing (Bool)", details: ["initialHostText": initial, "caretBefore": caretBefore, "caretAfter": String(initial.dropFirst(caretBefore.count))])
    try require(prepared["ordinaryHostEditing"] as? Bool == true, "Initial caret was not prepared in ordinary host UI")
    try require(ui("key-q").isHittable && ui("key-numbers").exists, "Initial letter layout unavailable")
    try checkpoint("initial", before: caretBefore, after: String(initial.dropFirst(caretBefore.count)))
  }

  private func visit(_ action: Int, changes: [(String, Bool)], before: String, after: String = "") throws {
    let original = text
    try backToSettings()
    for (label, value) in changes { try set(label, value) }
    // Read all four independent displayed values; mutate only named controls.
    try observeSettings()
    try returnToHost("a\(action)")
    try require(literalEqual(text, original), "Settings visit changed host text")
    try checkpoint("a\(action)", before: before, after: after)
  }
  private func cursor(_ action: Int, direction: String, before: String, after: String) throws {
    let original = text
    let response = try human("a\(action)-cursor-gesture", "Perform exactly ONE public gesture: hold visible key-space, continuously slide \(direction) until the ordinary host caret visibly reaches the requested boundary, then release. Stop by the visible caret, with no fixed displacement/duration oracle. Do not tap/reposition the host, insert a witness, retry, repair, or use a cursor accessibility custom action. Record the actual gesture and caret in evidence. If continuous human device driving is unavailable, report that obstacle.", fields: "continuousSpaceHoldSlideRelease (Bool), direction (String), stoppedByVisibleCaret (Bool)", details: ["caretBefore": before, "caretAfter": after])
    try require(response["continuousSpaceHoldSlideRelease"] as? Bool == true && response["direction"] as? String == direction && response["stoppedByVisibleCaret"] as? Bool == true, "Authored cursor action not performed")
    try require(literalEqual(text, original), "Cursor movement inserted/deleted text")
    try checkpoint("a\(action)", before: before, after: after)
  }
  private func search() throws {
    try tap("key-emoji")
    try tap("emoji-search-open")
    try require(ui("emoji-search-field").waitForExistence(timeout: 4), "Public local search field missing")
  }
  private func query(_ expected: String) throws {
    let field = ui("emoji-search-field")
    try require(field.exists, "Query field disappeared")
    try require((field.value as? String ?? "") == expected, "Displayed local query differs from \(expected.debugDescription)")
    image("query-\(expected.isEmpty ? "empty" : expected)")
  }
  private func lettersLayout() throws {
    // Explicit public return from picker/search; no toggle opening a new picker.
    if ui("emoji-panel").exists { try tap("emoji-abc") }
    else if ui("emoji-search-field").exists { try tap("key-letters") }
    try require(ui("key-q").isHittable && !ui("emoji-panel").exists && !ui("emoji-search-field").exists, "Letter layout did not return")
  }

  private struct DisplayedEmoji {
    let value: String
    let element: XCUIElement
    let identifier: String
    let label: String
    let axValue: String
    let frame: CGRect
  }
  private func visibleEmojiCandidates(searchResult: Bool) throws -> [XCUIElement] {
    let container = ui(searchResult ? "emoji-search-results" : "emoji-panel")
    try require(container.waitForExistence(timeout: 5), "Visible emoji container unavailable")
    let candidates: [XCUIElement]
    if searchResult {
      candidates = container.descendants(matching: .any).matching(NSPredicate(format: "identifier BEGINSWITH %@", "emoji-result-")).allElementsBoundByIndex
    } else {
      candidates = container.descendants(matching: .cell).matching(NSPredicate(format: "identifier BEGINSWITH %@", "emoji-")).allElementsBoundByIndex
    }
    let viewport = searchResult ? container.frame : container.collectionViews.firstMatch.frame
    // A partially clipped cell can still show its entire glyph. Do not impose
    // a new cell-geometry threshold; the observer must see the full glyph.
    return candidates.filter { $0.isHittable && !$0.frame.isEmpty && $0.frame.intersects(viewport) }
  }
  private func availableSearchResult() throws {
    let visible = try visibleEmojiCandidates(searchResult: true)
    try require(!visible.isEmpty, "Authored ordinary query has no visible selectable result")
    image("ordinary-query-visible-result")
  }
  private func observeEmoji(_ stage: String, searchResult: Bool = false,
                            matching expected: String? = nil, recent: Bool = false) throws -> DisplayedEmoji {
    if recent, let expected {
      let browse = try human(stage + "-browse-public-recents", "Show the recorded full emoji in the actual Frequently Used section using only ordinary public recents scrolling if needed. Do not select anything or change category, history, or host. No ordering/count/uniqueness/empty-history assumption. Report found false if absent; available false is reserved for inability to observe/drive the UI, not a failed membership oracle. Leave the matching entry fully visible for the next preselection observation.", fields: "found (Bool), inRecentSection (Bool)", details: ["recordedEmoji": expected])
      try require(browse["found"] as? Bool == true && browse["inRecentSection"] as? Bool == true, "Recorded emoji is not visibly present in recents")
    }
    let visible = try visibleEmojiCandidates(searchResult: searchResult)
    try require(!visible.isEmpty, "No visible selectable emoji; no identity/query substitution")
    let identities = visible.map { element in
      DisplayedEmoji(value: "", element: element, identifier: element.identifier, label: element.label, axValue: element.value as? String ?? "", frame: element.frame)
    }
    let descriptions: [[String: Any]] = visible.enumerated().map { index, element in
      ["index": index, "locatorIdentifier": element.identifier, "accessibleName": element.label,
       "accessibleValue": element.value as? String ?? "", "frame": NSStringFromCGRect(element.frame)]
    }
    let response = try human(stage + "-emoji-before-selection", "Observe the displayed selectable glyphs, and record ONE full actually displayed emoji as a literal Unicode String (every variation selector, modifier and joined component). Choose its listed index. Locator identifiers and semantic names alone do not establish its rendered variant. Use ordinary accessibility inspection as corroboration and cite the visible glyph screenshot. Do not tap, scroll, or alter UI at this handoff. \(recent ? "This must visibly belong to the Frequently Used section, not merely an adjacent category; record inRecentSection true only after observing membership." : "No identity/ranking/count requirement.") If invisible selectors/joiners cannot be established from public presentation and corroborating public accessibility, report the exact observation obstacle; never infer the value from later insertion.", fields: "index (Int), fullEmoji (String), fullSequenceObserved (Bool)\(recent ? ", inRecentSection (Bool)" : "")", details: ["visibleCandidates": descriptions, "matchingRecordedEmoji": expected ?? "no fixed identity"])
    guard let index = response["index"] as? Int, visible.indices.contains(index) else {
      try require(false, "Invalid observed visible emoji index")
      throw FixtureStop.failed
    }
    let value = response["fullEmoji"] as? String ?? ""
    try require(!value.isEmpty && response["fullSequenceObserved"] as? Bool == true, "Full displayed emoji was not observed before selection")
    if let expected { try require(value.unicodeScalars.elementsEqual(expected.unicodeScalars), "Observed recent variant differs from recorded full emoji") }
    if recent { try require(response["inRecentSection"] as? Bool == true, "Visible recent membership not established") }
    let original = identities[index]
    let result = DisplayedEmoji(value: value, element: original.element, identifier: original.identifier, label: original.label, axValue: original.axValue, frame: original.frame)
    try verifyEmoji(result)
    note(stage + "-recorded-full-emoji", "value=\(value)\nscalars=\(value.unicodeScalars.map { String(format: "U+%04X", $0.value) }.joined(separator: " "))")
    return result
  }
  private func verifyEmoji(_ emoji: DisplayedEmoji) throws {
    try require(emoji.element.exists && emoji.element.isHittable && emoji.element.identifier == emoji.identifier && emoji.element.label == emoji.label && (emoji.element.value as? String ?? "") == emoji.axValue && emoji.element.frame == emoji.frame, "Observed emoji cell changed before tap; no replacement")
  }
  private func choose(_ emoji: DisplayedEmoji) throws {
    try verifyEmoji(emoji)
    emoji.element.tap()
  }
  private func recents() throws {
    try tap("key-emoji")
    let recent = app.buttons["Frequently Used"].firstMatch
    try require(recent.waitForExistence(timeout: 4) && recent.isHittable, "Public recent category unavailable")
    recent.tap()
    image("visible-recents")
  }

  private func glide(_ action: Int, path: String, trail: Bool) throws -> String {
    let ready = try human("a\(action)-recording-ready", "Start a screen recording showing host caret and the visible key path before allowing the native gesture. Return its absolute runner-accessible recording file path now. Keep recording through gesture completion, then finalize it for the following trail/caret adjudication. No host or keyboard input at this handoff.", fields: "recording (absolute file path String), recordingActive (Bool)")
    let recording = ready["recording"] as? String ?? ""
    try require(!recording.isEmpty && ready["recordingActive"] as? Bool == true, "No recording for transient trail observation")
    let points = try path.map { character -> CGPoint in
      let key = ui("key-\(character)")
      try require(key.exists && key.isHittable, "Named visible slide key missing: \(character)")
      return CGPoint(x: key.frame.midX, y: key.frame.midY)
    }
    // A mechanical touch duration, not a speed/latency/recognition threshold.
    // No straight-line substitute if permitted multi-point synthesis is unavailable.
    try TouchPath.swipe(through: points)
    image("a\(action)-after-glide")
    let response = try human("a\(action)-trail-and-commit", "Review the actual continuous gesture recording, including while the finger is down. Record whether a visible trail followed it and the actual nonempty committed text span at the host caret. No exact word/spelling/ranking/prefix/boundary-spacing requirement. A post-release screenshot cannot prove trail absence or presence. Do not type, repair, retry, or dismiss the keyboard. Record the actual resulting entire host text separately.", fields: "recording (same String), trailVisible (Bool), committedSpan (nonempty String), insertedAtCaret (Bool), observedHostText (String)", details: ["recording": recording, "namedVisibleKeyPath": path, "expectedTrailVisible": trail])
    try require(response["recording"] as? String == recording, "Trail observation lacks the original gesture recording")
    guard recording.hasPrefix("/"), FileManager.default.fileExists(atPath: recording) else {
      throw XCTSkip("Transient trail observation unavailable: original gesture recording is not accessible at \(recording). No trail pass is claimed.")
    }
    let video = XCTAttachment(contentsOfFile: URL(fileURLWithPath: recording))
    video.name = "\(caseID)-a\(action)-continuous-gesture-recording"
    video.lifetime = .keepAlways
    add(video)
    try require(response["trailVisible"] as? Bool == trail, "Recorded visual trail differs from authored oracle")
    try require(!(response["committedSpan"] as? String ?? "").isEmpty && response["insertedAtCaret"] as? Bool == true, "No nonempty slide-typed span at caret")
    let actual = text
    try require(literalEqual(response["observedHostText"] as? String ?? "", actual), "Recorded actual glide host differs")
    try checkpoint("a\(action)", before: actual)
    note("a\(action)-actual-glide-host", actual)
    return actual
  }
  private func unavailablePhysicalFeedback(_ action: Int) {
    note("a\(action)-physical-feedback", "Physical system-click audibility: unavailable. Tactile haptics: unavailable. Simulator does not provide these physical observations; this fixture asserts displayed independent controls and editing continuity. No sound/haptic pass or impossible physical release prerequisite is claimed.")
  }

  func testEN01LiteralMiddleInsertion() throws {
    try begin("EN-01", language: "English", initial: "red blue", caretBefore: "red")
    try letters("sun") // 1
    try checkpoint("a1", before: "redsun", after: " blue")
    try delete() // 2; insertion above witnesses the original middle caret.
    try checkpoint("a2", before: "redsu", after: " blue")
    try letters(" on") // 3
    try checkpoint("a3", before: "redsu on", after: " blue")
  }

    try letters("x") // 2: insertion witness for action 1.
    try checkpoint("a2", before: "lax", after: "ke")

  }

    try search() // 1

    try letters("cat") // 2
