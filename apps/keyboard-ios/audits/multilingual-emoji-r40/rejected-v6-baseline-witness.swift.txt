// Audit-only observation of the UNCHANGED baseline's final index via Mirror.
// No localized candidate, production edit, index reconstruction, or projection.
import Foundation

private let locale = Locale(identifier: "en_US_POSIX")
private let ignored: Set<String> = ["a", "an", "and", "for", "in", "of", "on", "the", "to", "with"]
private func normalized(_ text: String) -> String {
  let folded = text.folding(options: [.diacriticInsensitive, .caseInsensitive], locale: locale)
  let pieces = folded.unicodeScalars.map { s -> String in
    if CharacterSet.alphanumerics.contains(s) { return String(s) }
    return s == "'" || s == "\u{2019}" ? "" : " "
  }.joined()
  return pieces.split(whereSeparator: \.isWhitespace).joined(separator: " ")
}
private func words(_ s: String) -> Set<String> { Set(s.split(separator: " ").map(String.init)) }
// Independent test interpretation of the normative T. The observed index and
// envelope below are independent of this localized token calculation.
private func spanish(_ text: String) -> Set<String> {
  var token = "", result: Set<String> = [], admitted = false, vowel = false
  for s in text.lowercased(with: locale).decomposedStringWithCanonicalMapping.unicodeScalars {
    switch s.properties.generalCategory {
    case .nonspacingMark, .spacingMark, .enclosingMark:
      if admitted && !vowel { token.unicodeScalars.append(s) }
    case .uppercaseLetter, .lowercaseLetter, .titlecaseLetter, .modifierLetter, .otherLetter:
      token.unicodeScalars.append(s); admitted = true
      vowel = "aeiou".unicodeScalars.contains(s)
    default:
      if !token.isEmpty { result.insert(token.precomposedStringWithCanonicalMapping) }
      token = ""; admitted = false; vowel = false
    }
  }
  if !token.isEmpty { result.insert(token.precomposedStringWithCanonicalMapping) }
  return result
}
private func child(_ value: Any, _ label: String) -> Any {
  Mirror(reflecting: value).children.first { $0.label == label }!.value
}

@main struct Witness {
  static func main() throws {
    let args = CommandLine.arguments
    let catalog = try EmojiCatalog.load(url: URL(fileURLWithPath: args[1]))
    let suite = try JSONSerialization.jsonObject(with: Data(contentsOf: URL(fileURLWithPath: args[2]))) as! [String: Any]
    var output: [[String: Any]] = []
    for row in suite["cases"] as! [[String: Any]] {
      let entries: [Emoji]
      let selected: EmojiCatalog
      if row["action"] as! String == "fixture" {
        let fixtures = row["entries"] as! [[String: Any]]
        let data = try JSONSerialization.data(withJSONObject: ["version": "fixture", "categories": [["id": "fixture", "name": "Fixture", "emoji": fixtures.map {
          ["e": $0["emoji"]!, "n": $0["name"]!, "v": 1, "k": $0["keywords"]!] as [String: Any]
        }]]])
        selected = try JSONDecoder().decode(EmojiCatalog.self, from: data)
        entries = selected.categories.flatMap(\.emoji)
      } else { selected = catalog; entries = selected.categories.flatMap(\.emoji) }
      let suggester = EmojiSuggester(catalog: selected)
      let index = child(suggester, "index")
      var keys = Array(repeating: Set<String>(), count: entries.count)
      for term in Mirror(reflecting: index).children {
        let key = child(term.value, "key") as! String
        let postings = Mirror(reflecting: child(term.value, "matches")).children
        precondition(postings.count <= 12, "baseline posting bound changed")
        for posting in postings { keys[child(posting.value, "id") as! Int].insert(key) }
      }
      let q = row["query"] as! String, w = words(normalized(q)), g = w.subtracting(ignored)
      var envelope: [String] = [], witnesses: [[String: Any]] = []
      for (id, e) in entries.enumerated() {
        let n = words(normalized(e.name))
        var v = Set(([e.name] + (e.keywords ?? [])).flatMap { words(normalized($0)) })
        v.formUnion(keys[id]); v.formUnion(keys[id].flatMap { words($0) })
        let exact = !w.isEmpty && n == w
        let anchors = keys[id].filter { k in w.contains { k.hasPrefix($0) } }.sorted()
        let missing = g.filter { word in !v.contains { $0.hasPrefix(word) } }.sorted()
        if exact || (!w.isEmpty && !anchors.isEmpty && missing.isEmpty) { envelope.append(e.value) }
        if row["action"] as! String == "fixture" {
          witnesses.append(["emoji": e.value, "N": n.sorted(), "B": keys[id].sorted(), "V": v.sorted(),
                            "exactWordSet": exact, "anchorKeys": anchors, "missingNonignoredPrefixes": missing])
        }
      }
      let legacy = suggester.search(q, limit: row["limit"] as? Int ?? 48, preferring: row["preferred"] as? [String: Int] ?? [:]).map(\.value)
      let escaped = legacy.filter { !envelope.contains($0) }
      precondition(escaped.isEmpty, "observed legacy entry escaped E: \(row["id"]!) \(escaped)")
      output.append(["id": row["id"]!, "T": spanish(q).sorted(), "W": w.sorted(), "E": envelope,
                     "legacyDirect": legacy, "escapedLegacy": escaped, "entries": witnesses])
    }
    try JSONSerialization.data(withJSONObject: output, options: [.prettyPrinted, .sortedKeys]).write(to: URL(fileURLWithPath: args[3]))
  }
}
