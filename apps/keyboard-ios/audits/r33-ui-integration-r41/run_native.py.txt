#!/usr/bin/env python3
"""R41 EXPOSED replay on its single owned device; requires an existing test build.
No production hooks, global preferences, broad data removal or raw-log reading.
"""
import argparse, hashlib, json, pathlib, subprocess, sys
ROOT = pathlib.Path(__file__).resolve().parents[4]  # repository root
RAW = pathlib.Path('/tmp/pulpo-r41-validation')
DEVICE = 'F3191384-9F35-4AC4-A739-37E6C7C8D584'
DD = RAW / 'DerivedData'
PROJECT = ROOT / 'apps/keyboard-ios/PulpoKeyboard.xcodeproj'
FOCUSED = ['LearningResetResultUITests', 'SuggestionForgetFeedbackTests', 'LearningForgetControllerTests', 'LearningForgetResultUITests']

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('label')
    parser.add_argument('--broad', action='store_true')
    parser.add_argument('--scope', action='store_true')
    parser.add_argument('--regression', action='store_true')
    args = parser.parse_args()
    receipt = json.loads((RAW / 'baseline-fixture-receipt.json').read_text())
    assert receipt['deviceUDID'] == DEVICE and receipt['purpose'] == 'R41-owned-test-store'
    group = pathlib.Path(receipt['groupContainer'])
    receipt['origin'] = 'R41-only device, all stores created during this recorded round'
    receipt['initialStoreFiles'] = {p.name:hashlib.sha256(p.read_bytes()).hexdigest() for p in group.iterdir() if p.name.startswith(('PersonalDictionary.json','TouchModel.json'))}
    receipt_path = RAW / (args.label + '-fixture-receipt.json')
    receipt_path.write_text(json.dumps(receipt, indent=2)+'\n')
    files = ['App/AppModel.swift','App/PulpoKeyboardApp.swift','PulpoKeyboardKit/Sources/KeyboardUI/KeyboardController.swift','PulpoKeyboardKit/Sources/KeyboardUI/SuggestionBarView.swift','PulpoKeyboardKit/Sources/KeyboardCore/Dictionary/PersonalDictionary.swift','PulpoKeyboardKit/Sources/KeyboardCore/Shared/LearnedDataPersistence.swift']
    files += [str(p.relative_to(ROOT/'apps/keyboard-ios')) for p in (ROOT/'apps/keyboard-ios/UITests').glob('Learning*.swift')]
    files += ['UITests/SuggestionForgetFeedbackTests.swift','UITests/SettingsUsabilitySweepTests.swift']
    freeze = {'head':subprocess.check_output(['git','rev-parse','HEAD'],cwd=ROOT,text=True).strip(),'files':{p:hashlib.sha256((ROOT/'apps/keyboard-ios'/p).read_bytes()).hexdigest() for p in files},'binaries':{str(p):hashlib.sha256(p.read_bytes()).hexdigest() for p in (DD/'Build/Products').rglob('*') if p.is_file() and p.name in ('PulpoKeyboard','PulpoKeyboardUITests','PulpoKeyboardAppTests')}}
    (RAW/(args.label+'-source-binary-hashes.json')).write_text(json.dumps(freeze,indent=2)+'\n')
    gate_log = open(RAW/(args.label+'-gate.log'),'w')
    gate = subprocess.Popen([sys.executable,str(pathlib.Path(__file__).with_name('host_fault_gate.py.txt')),'--group-container',str(group),'--fixture-receipt',str(receipt_path),'--idle-seconds','600'],stdout=gate_log,stderr=subprocess.STDOUT)
    import time
    control=pathlib.Path('/tmp/pulpo-r33-settings-reset-native-control.json')
    for _ in range(200):
        if control.exists(): break
        if gate.poll() is not None: raise RuntimeError('Gate rejected fixture')
        time.sleep(.02)
    session=json.loads(control.read_text())
    (RAW/(args.label+'-gate-session.json')).write_text(json.dumps(session,indent=2)+'\n')
    tests = ['-only-testing:PulpoKeyboardUITests/'+name for name in FOCUSED]
    if args.scope: tests += ['-only-testing:PulpoKeyboardUITests/LearningClearScopeUITests']
    if args.broad:
        tests += ['-only-testing:PulpoKeyboardAppTests']
        tests += ['-only-testing:PulpoKeyboardUITests/'+name for name in ['ControllerHarnessTests','ControllerFuzzTests','SettingsUsabilitySweepTests','DraftNavigationTests']]
    if args.regression and not args.broad:
        tests += ['-only-testing:PulpoKeyboardAppTests/CompanionWorkflowR24Tests']
        tests += ['-only-testing:PulpoKeyboardUITests/'+name for name in ['SettingsUsabilitySweepTests','DraftNavigationTests']]
    cmd=['xcodebuild','test-without-building','-project',str(PROJECT),'-scheme','PulpoKeyboard','-destination','platform=iOS Simulator,id='+DEVICE,'-derivedDataPath',str(DD),'-parallel-testing-enabled','NO','-collect-test-diagnostics','never',*tests,'-resultBundlePath',str(RAW/(args.label+'.xcresult'))]
    with open(RAW/(args.label+'.log'),'w') as log:
        code=subprocess.call(cmd,cwd=ROOT,stdout=log,stderr=subprocess.STDOUT)
    try: gatecode=gate.wait(timeout=20)
    except subprocess.TimeoutExpired:
        gate.terminate();gatecode=gate.wait(timeout=10)
    gate_log.close()
    summary=subprocess.check_output(['xcrun','xcresulttool','get','test-results','summary','--path',str(RAW/(args.label+'.xcresult')),'--compact'],text=True)
    (RAW/(args.label+'-summary.json')).write_text(summary+'\n')
    j=json.loads(summary)
    result={'label':args.label,'exit':code,'gateExit':gatecode,**{k:j.get(k) for k in ('totalTestCount','passedTests','failedTests','skippedTests')},'testFailures':j.get('testFailures',[])}
    (RAW/(args.label+'-outcome.json')).write_text(json.dumps(result,indent=2)+'\n')
    print(json.dumps(result))
if __name__=='__main__':main()
