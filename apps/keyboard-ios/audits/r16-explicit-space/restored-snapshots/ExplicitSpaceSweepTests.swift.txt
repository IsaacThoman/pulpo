import Foundation
import Testing
@testable import KeyboardCore

/// Full-document benchmark, opt-in so report generation cannot masquerade as acceptance.
@Suite(.serialized)
struct ExplicitSpaceSweepTests {
  struct Action: Decodable {
    let kind: String
    let text: String?
    let offset: Int?
    let start: Int?
    let length: Int?
  }
  struct Case: Decodable {
    let id: String
    let language: String
    let category: String
    let intent: String
    let initial: String
    let caret: Int
    let selectionLength: Int?
    let typed: String
    let expected: String
    let field: String?
    let operations: [Action]?
    let ambiguous: Bool?
    let expectedCaret: Int?
    let expectedSelectionLength: Int?
    let capitalization: String?
  }
  struct Row: Encodable {
    let id: String
    let category: String
    let language: String
    let timing: String
    let ambiguous: Bool
    let expected: String
    let actual: String
    let expectedCaret: Int?
    let actualCaret: Int
    let actualSelectionLength: Int
    let primitiveCount: Int
    let queuedRequests: Int
    let queuedApplied: Int
    let passed: Bool
    let unavailable: String?
    let edits: Int
    let lagReports: Int
    let externalChanges: Int
    let personalCount: Int
    let conservationFailures: [String]
    let trace: [String]
  }
  enum Invalid: Error { case contract(String) }

  @Test func fullDocumentPipeline() throws {
    let env = ProcessInfo.processInfo.environment
    guard let source = env["PULPO_R16_CASES"], let output = env["PULPO_R16_RESULTS"] else { return }
    let cases = try JSONDecoder().decode([Case].self, from: Data(contentsOf: URL(fileURLWithPath: source)))
    #expect(Set(cases.map(\.id)).count == cases.count)
    let lexicons = try Dictionary(uniqueKeysWithValues: KeyboardLanguage.allCases.map {
      ($0.rawValue, try Lexicon(url: Fixtures.languageData.appendingPathComponent($0.rawValue + ".pkdict")))
    })
    var rows: [Row] = []
    for task in cases {
      for timing in ["immediate", "queued", "optimistic", "old-own"] {
        let language: KeyboardLanguage
        switch task.language {
        case "en": language = .english
        case "es": language = .spanish
        case "bilingual": language = .bilingual
        default: throw Invalid.contract("language: " + task.language)
        }
        let lexicon = lexicons[language.rawValue]!
        let model = LanguageModel(lexicon: lexicon, personal: PersonalDictionary(url: nil))
        let corrector = SpellCorrector(lexicon: lexicon)
        corrector.touch = TouchModel(url: nil)
        let suggester = Suggester(model: model, corrector: corrector)
        var settings = KeyboardSettings()
        settings.language = language
        let document = ExplicitSpaceDocument(task.initial, caret: task.caret, selection: task.selectionLength ?? 0, timing: timing)
        let engine = KeyboardEngine(document: document, suggester: suggester, glide: GlideDecoder(model: model), settings: settings)
        var traits = InputTraits()
        if task.capitalization == "none" { traits.capitalization = .none }
        switch task.field ?? "normal" {
        case "normal": break
        case "password": traits.isSecure = true; document.hidesContext = true
        case "email": traits.keyboard = .email
        case "url": traits.keyboard = .url
        case "number": traits.keyboard = .numberPad
        default: throw Invalid.contract("field: " + (task.field ?? ""))
        }
        engine.traits = traits
        var time = 100.0
        engine.clock = { time }
        engine.documentDidChange()
        var trace: [String] = []
        var unavailable: String?
        func type(_ text: String) {
          for character in text {
            time += timing == "old-own" ? 2 : 0.04
            if timing == "old-own", !document.inbox.isEmpty {
              document.reportNext()
              engine.documentDidChange()
            }
            let action: R16ConservationOracle.Action
            let lower = String(character).lowercased()
            if character.isLetter, let key = lower.first, lower.count == 1 {
              let capital = character.isUppercase || (task.capitalization != "none" && Self.sentenceStart(document.logical.before))
              action = .letter(key, casing: capital ? .uppercase : .asEntered)
            } else if character == " " { action = .explicitSpace }
            else if character == "\n" { action = .returnKey }
            else { action = .punctuation }
            document.perform(action) { engine.type(String(character)) }
            if timing == "immediate" { engine.documentDidChange() }
            if timing == "optimistic", !document.inbox.isEmpty {
              // Optimistic proxies can later report earlier primitive own states.
              document.reportNext()
              engine.documentDidChange()
              document.flush()
              engine.documentDidChange()
            }
            trace.append(document.logical.description)
          }
        }
        type(task.typed)
        for operation in task.operations ?? [] {
          switch operation.kind {
          case "type": type(operation.text ?? "")
          case "swipe":
            var rng = SeededGenerator(state: 123)
            document.perform(.swipe) { engine.insertGlide(SyntheticSwipe.points(for: operation.text ?? "", noise: 0, generator: &rng)) }
          case "insertText": document.perform(.insertText) { engine.insertText(operation.text ?? "") }
          case "host":
            document.flush()
            let text = operation.text ?? ""
            let start = operation.start ?? -1
            let length = operation.length ?? 0
            guard start >= 0 && length >= 0 && start + length <= text.count else { unavailable = "invalid host caret"; break }
            document.state = .init(text: Array(text), caret: start, selection: length)
            document.hostChanged(); document.flush(); engine.documentDidChange()
          case "flush": document.flush(); engine.documentDidChange()
          case "move":
            document.flush()
            let next = document.state.caret + (operation.offset ?? 0)
            if next < 0 || next > document.state.text.count { unavailable = "invalid move offset"; break }
            document.state.caret = next
            document.state.selection = 0
            document.hostChanged(); document.flush(); engine.documentDidChange()
          case "select":
            document.flush()
            let start = operation.start ?? -1
            let length = operation.length ?? -1
            if start < 0 || length < 0 || start + length > document.state.text.count { unavailable = "invalid selection"; break }
            document.state.caret = start; document.state.selection = length
            document.hostChanged(); document.flush(); engine.documentDidChange()
          case "undo": document.perform(.backspace) { engine.deleteBackward() }
          default: unavailable = "unsupported operation " + operation.kind
          }
          trace.append(operation.kind + ": " + document.state.description)
          if unavailable != nil { break }
        }
        document.flush(); engine.documentDidChange()
        let actual = String(document.state.text)
        rows.append(Row(id: task.id, category: task.category, language: task.language, timing: timing,
          ambiguous: task.ambiguous ?? false, expected: task.expected, actual: actual,
          expectedCaret: task.expectedCaret, actualCaret: document.state.caret, actualSelectionLength: document.state.selection,
          primitiveCount: document.primitives.count, queuedRequests: document.queuedRequests, queuedApplied: document.queuedApplied,
          passed: unavailable == nil && R16ConservationOracle.sameText(actual, task.expected) && (task.expectedCaret == nil || task.expectedCaret == document.state.caret) && document.state.selection == (task.expectedSelectionLength ?? 0) && document.conservationFailures.isEmpty, unavailable: unavailable,
          edits: document.edits, lagReports: engine.laggingContextUpdates, externalChanges: engine.externalContextChanges,
          personalCount: model.personal.totalCount, conservationFailures: document.conservationFailures, trace: trace))
      }
    }
    #expect(rows.count == cases.count * 4)
    let encoder = JSONEncoder()
    encoder.outputFormatting = [.prettyPrinted, .sortedKeys, .withoutEscapingSlashes]
    try encoder.encode(rows).write(to: URL(fileURLWithPath: output))
    let scored = rows.filter { !$0.ambiguous }
    print("R16 REPORT ONLY: \(scored.filter(\.passed).count)/\(scored.count) exact, \(rows.filter(\.ambiguous).count) ambiguous, \(rows.filter { $0.unavailable != nil }.count) unavailable")
  }
  // Independent contract model for the sentence capitalization used by these cases.
  // No reads of engine shift/composition/suggestions or candidate output.
  static func sentenceStart(_ before: String) -> Bool {
    var text = before[...]
    while let last = text.last, last.isWhitespace && last != "\n" { text = text.dropLast() }
    while let last = text.last, ")]}'\"’”".contains(last) { text = text.dropLast() }
    return text.isEmpty || text.last == "\n" || ".!?".contains(text.last!)
  }

}

/// True host text and proxy-reported text are separate, including real selection replacement.
final class ExplicitSpaceDocument: TextDocument {
  struct State: Equatable {
    var text: [Character]
    var caret: Int
    var selection: Int
    var before: String { String(text[..<caret]) }
    var after: String { String(text[(caret + selection)...]) }
    var selected: String? { selection == 0 ? nil : String(text[caret..<(caret + selection)]) }
    var oracleSnapshot: R16ConservationOracle.Snapshot { .init(text: String(text), caret: caret, selectionLength: selection) }
    var description: String { before + "|[" + (selected ?? "") + "]|" + after }
  }
  struct Primitive {
    let inserted: String? // nil means one backspace
    let before: State
    let after: State
  }
  var primitives: [Primitive] = []
  var oracle: R16ConservationOracle
  var pendingTraces: [R16ConservationOracle.Trace] = []
  var conservationFailures: [String] = []
  var queued: State?
  var queuedWrites: [String?] = []
  var queuedRequests = 0
  var queuedApplied = 0
  var state: State
  var reported: State
  var inbox: [State] = []
  let timing: String
  var hidesContext = false
  let documentIdentifier: UUID? = UUID()
  var edits = 0
  init(_ text: String, caret: Int, selection: Int, timing: String) {
    precondition(caret >= 0 && selection >= 0 && caret + selection <= text.count)
    state = State(text: Array(text), caret: caret, selection: selection)
    reported = state
    oracle = R16ConservationOracle(actual: state.oracleSnapshot)
    self.timing = timing
  }
  var documentContextBeforeInput: String? { hidesContext ? nil : reported.before }
  var documentContextAfterInput: String? { hidesContext ? nil : reported.after }
  var selectedText: String? { hidesContext ? nil : reported.selected }
  var hasText: Bool { !state.text.isEmpty }
  var logical: State { queued ?? state }
  func perform<T>(_ action: R16ConservationOracle.Action, _ operation: () -> T) {
    let before = logical.oracleSnapshot
    let start = primitives.count
    _ = operation()
    let steps = primitives[start...].map {
      R16ConservationOracle.Step(primitive: $0.inserted.map { .insert($0) } ?? .deleteBackward,
                                before: $0.before.oracleSnapshot, after: $0.after.oracleSnapshot)
    }
    let trace = R16ConservationOracle.Trace(action: action, before: before, steps: steps, after: logical.oracleSnapshot)
    if timing == "queued" { pendingTraces.append(trace) } else { observe(trace) }
  }
  func observe(_ trace: R16ConservationOracle.Trace) {
    let verdict = oracle.observe(trace)
    conservationFailures += verdict.failures
    let protectedAction: Bool
    switch trace.action {
    case .explicitSpace, .punctuation, .returnKey, .swipe, .insertText: protectedAction = true
    default: protectedAction = false
    }
    // Retain the blanket before-suffix prohibition for every unproven action.
    // Inspect actual deleted characters, not a candidate's autocorrect flag.
    let deletedLetters = trace.steps.contains {
      $0.primitive == .deleteBackward && $0.before.selectionLength == 0 && $0.before.before.last?.isLetter == true
    }
    if protectedAction, deletedLetters,
       trace.before.after.first.map({ $0.isLetter || $0.isNumber }) == true,
       !verdict.permitsSpaceReplacement {
      conservationFailures.append("unowned or non-space replacement before inherited suffix")
    }
  }
  func hostChanged() { oracle.hostChanged(to: state.oracleSnapshot) }
  func insertText(_ text: String) {
    let previous = logical
    var next = previous
    let left = next.before + text
    next.text = Array(left + next.after)
    next.caret = min(left.count, next.text.count); next.selection = 0
    edited(previous, next, inserted: text)
  }
  func deleteBackward() {
    let previous = logical
    var next = previous
    if next.selection > 0 {
      next.text.removeSubrange(next.caret..<(next.caret + next.selection)); next.selection = 0
    } else if next.caret > 0 {
      next.text.remove(at: next.caret - 1); next.caret -= 1
    }
    edited(previous, next, inserted: nil)
  }
  private func edited(_ previous: State, _ next: State, inserted: String?) {
    edits += 1
    primitives.append(Primitive(inserted: inserted, before: previous, after: next))
    inbox.append(next)
    if timing == "queued" {
      queuedRequests += 1
      queuedWrites.append(inserted)
      queued = next
    } else {
      state = next
    }
    if timing == "immediate" || timing == "optimistic" { reported = next }
  }
  func reportNext() {
    if !inbox.isEmpty { reported = inbox.removeFirst() }
    oracle.proxyReported(reported.oracleSnapshot)
  }
  func flush() {
    if let queued {
      // Replay every queued primitive against the unchanged physical host.
      let requestedSteps = pendingTraces.flatMap(\.steps)
      precondition(requestedSteps.count == queuedWrites.count)
      for (index, inserted) in queuedWrites.enumerated() {
        precondition(state.oracleSnapshot == requestedSteps[index].before)
        if let inserted {
          let left = state.before + inserted
          state.text = Array(left + state.after)
          state.caret = min(left.count, state.text.count); state.selection = 0
        } else if state.selection > 0 {
          state.text.removeSubrange(state.caret..<(state.caret + state.selection)); state.selection = 0
        } else if state.caret > 0 {
          state.text.remove(at: state.caret - 1); state.caret -= 1
        }
        queuedApplied += 1
        precondition(state.oracleSnapshot == requestedSteps[index].after)
      }
      precondition(state == queued, "Queued primitive replay must equal requested host state")
      queuedWrites.removeAll()
      self.queued = nil
    }
    for trace in pendingTraces { observe(trace) }
    pendingTraces.removeAll()
    reported = state; inbox.removeAll()
  }
}
