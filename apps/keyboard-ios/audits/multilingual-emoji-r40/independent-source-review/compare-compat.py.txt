from pathlib import Path
import json,itertools
r=Path('/tmp/pulpo-r40-source-review');fails=[];changed=[];rows=0
for b,c in itertools.zip_longest(open(r/'baseline-compat.jsonl'),open(r/'candidate-compat.jsonl')):
 assert b is not None and c is not None
 b=json.loads(b);c=json.loads(c);rows+=1
 errors=[]
 assert (b['q'],b['cap'],b['pref'])==(c['q'],c['cap'],c['pref'])
 if c['search'][:len(b['search'])]!=b['search']:errors.append('legacy-prefix')
 if c['contextual']!=b['contextual']:errors.append('contextual')
 if c['exactContextual']!=b['exactContextual']:errors.append('exact-contextual')
 if len(c['search'])>c['cap']:errors.append('cap')
 if len(set(c['search']))!=len(c['search']):errors.append('dedupe')
 if errors:fails.append({'baseline':b,'candidate':c,'errors':errors})
 if b['search']!=c['search']:changed.append({'q':b['q'],'cap':b['cap'],'pref':b['pref'],'baseline':b['search'],'candidate':c['search']})
assert rows==7374*3*3
s={'queries':7374,'caps':[1,48,60],'preferenceMaps':['empty','catalog ordinal modulo3','alternating Int.max/Int.min'],'pairedRows':rows,'failures':len(fails),'changedRawArraysAcrossAll9Configurations':len(changed),'mechanicalOccurrencesAddedAcrossAll9Configurations':sum(len(x['candidate'])-len(x['baseline']) for x in changed),'EnglishIndexExact':json.load(open(r/'baseline-index.json'))==json.load(open(r/'candidate-index.json'))}
for name,obj in [('compat-failures.json',fails),('compat-changed-rows.json',changed),('compat-summary.json',s)]: (r/name).write_text(json.dumps(obj,ensure_ascii=False,indent=2)+'\n')
print(json.dumps(s))
