// copy temporarily into AppTests with BehaviorNativeTests.swift. No test policy
// is inferred from the implementation: every oracle comes from the case JSON.

struct R38Suite: Decodable {
  let version: Int
  let cases: [R38Case]
}

struct R38Case: Decodable {
  let id: String
  let intent: String
  let languages: [String: R38Variant]
}

struct R38Variant: Decodable {
  let initialText: String
  let selection: [Int]
  let field: R38Field
  let steps: [R38Step]
}

struct R38Field: Codable {
  enum Route: String, Codable { case native, legacy, custom }
  enum Keyboard: String, Codable { case standard, numberPad, email, url }
  let id: String
  let route: Route
  let keyboard: Keyboard
  let secure: Bool
}

struct R38Step: Decodable {
  enum Op: String, Decodable {
    case type, insert, space, delete, field, host, capture, choose
    case guardOrigin, invalidateGuard, destination, automatic, confirm
  }
  enum Kind: String, Decodable { case literal, completion, correction, prediction }
  let op: Op
  let text: String?
  let selection: [Int]?
  let field: R38Field?
  let newIdentity: Bool?
  let kind: Kind?
  let expect: R38Expected
}

struct R38Expected: Codable {
  let text: String
  let selection: [Int]
  let `private`: Bool?
  let plain: Bool?
  let suggestionsEmpty: Bool?
  let knownCounts: [String: Int]?
  let totalCount: Int?
  let touchWeight: Double?
  let touchPositive: Bool?
  let allowsConfirmation: Bool?
  let automaticCore: Bool?
  let automaticNative: Bool?
  let acceptedCore: Bool?
  let acceptedNative: Bool?
}

struct R38Error: Error, CustomStringConvertible {
  let description: String
  init(_ message: String) { description = message }
}

private func r38JSON<T: Encodable>(_ value: T) throws -> Any {
  try JSONSerialization.jsonObject(with: JSONEncoder().encode(value), options: [.fragmentsAllowed])
}

private func r38Nullable<T>(_ value: T?) -> Any { value.map { $0 as Any } ?? NSNull() }

// Selection coordinates are UTF-16, including non-BMP characters. Reject ranges
// which bisect surrogate pairs rather than silently clamping the test input.
private func r38Range(_ selection: [Int], in text: String) throws -> NSRange {
  let string = text as NSString
  guard selection.count == 2, selection[0] >= 0, selection[1] >= 0,
        selection[0] <= string.length, selection[1] <= string.length - selection[0] else {
    throw R38Error("Invalid UTF16 selection \(selection) for length \(string.length)")
  }
  for offset in [selection[0], selection[0] + selection[1]] where offset > 0 && offset < string.length {
    let a = string.character(at: offset - 1), b = string.character(at: offset)
    if (0xD800...0xDBFF).contains(a) && (0xDC00...0xDFFF).contains(b) {
      throw R38Error("UTF16 selection splits a surrogate pair at \(offset)")
    }
  }
  return NSRange(location: selection[0], length: selection[1])
}

#if !canImport(UIKit)
private final class R38CoreDocument: TextDocument {
  var text = ""
  var selection = NSRange(location: 0, length: 0)
  var identity = UUID()
  var documentIdentifier: UUID? { identity }
  var documentContextBeforeInput: String? { (text as NSString).substring(to: selection.location) }
  var documentContextAfterInput: String? { (text as NSString).substring(from: NSMaxRange(selection)) }
  var selectedText: String? {
    selection.length == 0 ? nil : (text as NSString).substring(with: selection)
  }
  var hasText: Bool { !text.isEmpty }
  func replace(text: String, selection: [Int], newIdentity: Bool) throws {
    let range = try r38Range(selection, in: text)
    self.text = text
    self.selection = range
    if newIdentity { identity = UUID() }
  }
  func insertText(_ inserted: String) {
    text = (text as NSString).replacingCharacters(in: selection, with: inserted)
    selection = NSRange(location: selection.location + inserted.utf16.count, length: 0)
  }
  func deleteBackward() {
    if selection.length == 0 {
      guard selection.location > 0 else { return }
      selection = (text as NSString).rangeOfComposedCharacterSequence(at: selection.location - 1)
    }
    insertText("")
  }
}

@MainActor
final class R38BehaviorRunner {
  static let languages = ["en_US", "es", "en_es"]
  static let privateIDs: Set<String> = [
    "username", "password", "newPassword", "oneTimeCode", "emailAddress", "URL", "telephoneNumber",
    "creditCardNumber", "creditCardSecurityCode", "creditCardExpiration", "creditCardExpirationMonth",
    "creditCardExpirationYear", "creditCardName", "creditCardGivenName", "creditCardFamilyName",
    "creditCardMiddleName", "creditCardType", "postalCode", "flightNumber", "shipmentTrackingNumber",
    "birthdate", "birthdateDay", "birthdateMonth", "birthdateYear",
  ]
  static let normalIDs: Set<String> = [
    "none", "name", "namePrefix", "givenName", "middleName", "familyName", "nameSuffix", "nickname",
    "jobTitle", "organizationName", "location", "fullStreetAddress", "streetAddressLine1",
    "streetAddressLine2", "addressCity", "addressState", "addressCityAndState", "sublocality",
    "countryName", "dateTime", "cellularEID", "cellularIMEI",
  ]
  static var platform: String {
#if canImport(UIKit)
    "native"
#else
    "core"
#endif
  }

  private let calibration: [String: String]
  private let calibrationJSON: Any
  private let resourceDirectory: URL

  convenience init(calibrationURL: URL, languageDataURL: URL) throws {
    let data = try Data(contentsOf: calibrationURL)
    try self.init(calibrationData: data, languageDataURL: languageDataURL)
  }

  init(calibrationData: Data, languageDataURL: URL) throws {
    struct Calibration: Decodable {
      struct Field: Decodable { let id: String; let rawValue: String }
      let fields: [Field]
    }
    let decoded = try JSONDecoder().decode(Calibration.self, from: calibrationData)
    var values: [String: String] = [:]
    for row in decoded.fields {
      guard values.updateValue(row.rawValue, forKey: row.id) == nil else {
        throw R38Error("Duplicate calibration ID: \(row.id)")
      }
    }
    for id in Self.privateIDs.union(Self.normalIDs).subtracting(["none"]) where values[id] == nil {
      throw R38Error("Missing measured native calibration value: \(id)")
    }
    calibration = values
    calibrationJSON = try JSONSerialization.jsonObject(with: calibrationData)
    resourceDirectory = languageDataURL
  }

  // Internal on purpose: both the separately compiled CLI and AppTests can call
  // the same entry point. Each case/language creates its own complete stack.
  func run(casesData: Data, label: String = "development") throws -> Data {
    let raw = try JSONSerialization.jsonObject(with: casesData)
    try Self.validateSchema(raw)
    let suite = try JSONDecoder().decode(R38Suite.self, from: casesData)
    guard suite.version == 1 else { throw R38Error("Expected schema version 1") }
    var results: [[String: Any]] = []
    var failures: [String] = []
    var seen: Set<String> = []
    for test in suite.cases {
      if !seen.insert(test.id).inserted { failures.append("Duplicate case ID: \(test.id)") }
      for language in Self.languages {
        guard let variant = test.languages[language] else {
          failures.append("\(test.id)/\(language): missing required language variant")
          results.append(["id": test.id, "intent": test.intent, "language": language,
                          "status": "setupFailed", "failures": ["Missing required language variant"], "steps": []])
          continue
        }
        do {
          let state = try State(variant: variant, language: language, runner: self)
          let initial = try state.snapshot()
          var steps: [[String: Any]] = []
          var localFailures: [String] = []
          for (index, step) in variant.steps.enumerated() {
            let before = try state.snapshot()
            var errors: [String] = []
            let action = state.perform(step, failures: &errors)
            let actual = try state.snapshot()
            errors += state.compare(step.expect, actual: actual, action: action)
            let prefix = "\(test.id)/\(language)/\(index): "
            localFailures += errors.map { prefix + $0 }
            let inactiveKeys = Self.platform == "core" ? ["automaticNative", "acceptedNative"] : ["automaticCore", "acceptedCore"]
            let expectedJSON = try r38JSON(step.expect) as! [String: Any]
            steps.append(["index": index, "op": step.op.rawValue, "before": before,
                          "actual": actual, "action": action,
                          "expected": expectedJSON,
                          "inactivePlatformExpectations": inactiveKeys.filter { expectedJSON[$0] != nil },
                          "failures": errors,
                          "passed": errors.isEmpty])
          }
          failures += localFailures
          results.append(["id": test.id, "intent": test.intent, "language": language,
                          "initial": initial, "steps": steps, "failures": localFailures,
                          "status": localFailures.isEmpty ? "passed" : "failed"])
        } catch {
          let failure = "\(test.id)/\(language): setup failed: \(error)"
          failures.append(failure)
          results.append(["id": test.id, "intent": test.intent, "language": language,
                          "status": "setupFailed", "failures": [failure], "steps": []])
        }
      }
    }
    let output: [String: Any] = [
      "version": 1, "runner": "R38-development-behavior", "label": label,
      "platform": Self.platform, "os": ProcessInfo.processInfo.operatingSystemVersionString,
      "evidence": "guard-level; no recorder/controller start or recovery acceptance",
      "controllerAcceptance": "unexercised", "calibration": calibrationJSON,
      "languageData": resourceDirectory.path, "input": raw, "results": results,
      "failureCount": failures.count, "failures": failures, "passed": failures.isEmpty,
    ]
    return try JSONSerialization.data(withJSONObject: output, options: [.prettyPrinted, .sortedKeys])
  }

  private func validate(_ field: R38Field) throws {
    switch field.route {
    case .native:
      guard Self.privateIDs.union(Self.normalIDs).contains(field.id) else {
        throw R38Error("Unsupported native field ID: \(field.id)")
      }
    case .legacy:
      guard Self.privateIDs.contains(field.id) else { throw R38Error("Unsupported legacy field ID: \(field.id)") }
    case .custom: break // Exact opaque string, including whitespace or empty.
    }
  }

  private func coreTraits(_ field: R38Field) throws -> InputTraits {
    try validate(field)
    var traits = InputTraits()
    switch field.keyboard {
    case .standard: traits.keyboard = .standard
    case .numberPad: traits.keyboard = .numberPad
    case .email: traits.keyboard = .email
    case .url: traits.keyboard = .url
    }
    traits.isSecure = field.secure
    traits.capitalization = .none
    traits.autocorrection = true
    traits.spellChecking = true
    traits.smartQuotes = true
    traits.smartDashes = true
    switch field.route {
    case .native:
      if field.id == "none" { traits.contentType = nil }
      else if let measured = calibration[field.id] { traits.contentType = measured }
      else { throw R38Error("Missing native calibration value: \(field.id)") }
    case .legacy, .custom: traits.contentType = field.id
    }
    return traits
  }

  @MainActor
  private final class State {
    let runner: R38BehaviorRunner
    let stack: LanguageStack
    let engine: KeyboardEngine
    var field: R38Field
    var captured: Suggestion?
    var insertionGuard: DictationInsertionGuard?
    var origin: [String: Any]?
    var invalidated = false
    var ticks: TimeInterval = 100
    var vocabulary: Set<String> = []
#if canImport(UIKit)
    let native: R38NativeDocument
    var document: TextDocument { native.host.document }
    var text: String { native.textView.text ?? "" }
    var selection: NSRange { native.textView.selectedRange }
#else
    let core: R38CoreDocument
    var document: TextDocument { core }
    var text: String { core.text }
    var selection: NSRange { core.selection }
#endif

    init(variant: R38Variant, language: String, runner: R38BehaviorRunner) throws {
      self.runner = runner
      field = variant.field
      try runner.validate(field)
      stack = try LanguageStack(dictionaryURL: runner.resourceDirectory.appendingPathComponent("\(language).pkdict"),
                                personalURL: nil, touchURL: nil)
      stack.setGeometry(.qwerty)
      var settings = KeyboardSettings()
      settings.language = KeyboardLanguage(rawValue: language)!
      settings.autoCapitalization = false
#if canImport(UIKit)
      native = R38NativeDocument()
      try native.apply(field)
      try native.replace(text: variant.initialText, selection: variant.selection, newIdentity: false)
      engine = KeyboardEngine(document: native.host.document, suggester: stack.suggester, glide: stack.glide, settings: settings)
      engine.traits = native.host.inputTraits
#else
      core = R38CoreDocument()
      try core.replace(text: variant.initialText, selection: variant.selection, newIdentity: false)
      engine = KeyboardEngine(document: core, suggester: stack.suggester, glide: stack.glide, settings: settings)
      engine.traits = try runner.coreTraits(field)
#endif
      engine.clock = { [weak self] in self?.ticks ?? 100 }
      engine.documentDidChange()
      track(variant.initialText)
      for step in variant.steps {
        track(step.text ?? "")
        track(step.expect.text)
        vocabulary.formUnion(step.expect.knownCounts?.keys.map { $0 } ?? [])
      }
    }

    func track(_ text: String) {
      let words = text.split(whereSeparator: { !$0.isLetter && !$0.isNumber && $0 != "'" && $0 != "’" }).map(String.init)
      vocabulary.formUnion(words)
      vocabulary.formUnion(words.map { $0.lowercased() })
    }

    func reconcile() {
#if canImport(UIKit)
      engine.traits = native.host.inputTraits
#endif
      engine.documentDidChange()
      engine.reconcileHostChanges()
    }

    func currentDestination() -> DictationDestination {
#if canImport(UIKit)
      DictationDestination(document: document, traits: native.host.inputTraits)
#else
      DictationDestination(document: document, traits: engine.traits)
#endif
    }

    func guardState() -> [String: Any] {
      let destination = currentDestination()
      let permission = insertionGuard?.allowsAutomaticInsertion(at: destination) ?? false
      let available = insertionGuard != nil && document.documentIdentifier != nil
      let reason: String
      if insertionGuard == nil { reason = "noOrigin" }
      else if document.documentIdentifier == nil { reason = "missingDocumentIdentifier" }
      else { reason = "available" }
      return ["hasOrigin": insertionGuard != nil, "origin": r38Nullable(origin),
              "invalidated": invalidated, "available": available, "availability": reason,
              "allowsAutomaticInsertion": permission]
    }

    func destinationState() -> [String: Any] {
      let destination = currentDestination()
      return ["isPrivate": destination.isPrivate, "hasSelection": destination.hasSelection,
              "allowsConfirmation": destination.allowsConfirmation,
              "documentIdentifier": r38Nullable(document.documentIdentifier?.uuidString),
              "before": r38Nullable(document.documentContextBeforeInput),
              "after": r38Nullable(document.documentContextAfterInput),
              "selectedText": r38Nullable(document.selectedText)]
    }

    func perform(_ step: R38Step, failures: inout [String]) -> [String: Any] {
      ticks += 2 // Deterministic taps, outside double-space/delayed-host windows.
      var action: [String: Any] = ["executed": false, "accepted": NSNull(),
                                    "acceptedCore": NSNull(), "acceptedNative": NSNull(),
                                    "evidence": "engine operation"]
      do {
        switch step.op {
        case .type:
          guard let value = step.text else { throw R38Error("type requires text") }
          for character in value {
            ticks += 2
            if character == " " { engine.insertSpace() }
            else {
              let key = String(character)
              let normalized = Array(TextNormalizer.key(key).utf8)
              let tap = normalized.count == 1 ? LetterGeometry.qwerty.center(normalized[0]) : nil
              engine.insertCharacter(key, tap: character.isLetter ? tap : nil)
            }
          }
        case .insert:
          guard let value = step.text else { throw R38Error("insert requires text") }
          engine.insertText(value, spacing: false)
        case .space: engine.insertSpace()
        case .delete: engine.deleteBackward()
        case .field:
          guard let value = step.field else { throw R38Error("field requires field") }
          try runner.validate(value)
#if canImport(UIKit)
          try native.apply(value)
#else
          engine.traits = try runner.coreTraits(value)
#endif
          field = value
          reconcile()
        case .host:
          guard let value = step.text, let range = step.selection, let newIdentity = step.newIdentity else {
            throw R38Error("host requires text, selection and newIdentity")
          }
#if canImport(UIKit)
          try native.replace(text: value, selection: range, newIdentity: newIdentity)
          if newIdentity { try native.apply(field) }
#else
          try core.replace(text: value, selection: range, newIdentity: newIdentity)
#endif
          reconcile()
        case .capture:
          captured = nil // A failed capture must not reuse an earlier choice.
          guard let value = step.text, let kind = step.kind else { throw R38Error("capture requires kind and text") }
          captured = engine.suggestions.slots.first { Array($0.text.utf16) == Array(value.utf16) && String(describing: $0.kind) == kind.rawValue }
          guard captured != nil else { throw R38Error("Missing offered choice: \(kind.rawValue) \(String(reflecting: value))") }
          action["captured"] = ["text": value, "kind": kind.rawValue]
        case .choose:
          guard let choice = captured else { throw R38Error("choose has no captured choice") }
          action["choice"] = ["text": choice.text, "kind": String(describing: choice.kind)]
          engine.select(choice) // Keep the original engine sourceContext token.
        case .guardOrigin:
          let destination = currentDestination()
          insertionGuard = DictationInsertionGuard(origin: destination)
          origin = destinationState()
          invalidated = false
        case .invalidateGuard:
          guard insertionGuard != nil else { throw R38Error("invalidateGuard has no captured guard origin") }
          insertionGuard?.invalidate()
          invalidated = true
        case .destination:
          action["destination"] = destinationState()
          action["guard"] = guardState()
        case .automatic, .confirm:
          guard let value = step.text else { throw R38Error("\(step.op.rawValue) requires text") }
          let expectedAcceptance = Self.activeAcceptance(step.expect)
          if expectedAcceptance == nil {
            failures.append("\(step.op.rawValue) requires predeclared accepted\(R38BehaviorRunner.platform == "core" ? "Core" : "Native") expectation")
          }
          let destination = currentDestination()
          let guardBefore = guardState()
          let available = step.op == .confirm || guardBefore["available"] as? Bool == true
          let allowed = step.op == .automatic
            ? insertionGuard?.allowsAutomaticInsertion(at: destination) ?? false
            : destination.allowsConfirmation
          action["evidence"] = "direct guard-level challenge; not controller acceptance"
          action["challengeFired"] = true
          action["destinationBefore"] = destinationState()
          action["guardBefore"] = guardBefore
          action["available"] = available
          action["outcome"] = available ? (allowed ? "accepted" : "denied") : "unavailable"
          action["guardInvoked"] = step.op == .confirm || insertionGuard != nil
          action["accepted"] = allowed
          action[R38BehaviorRunner.platform == "core" ? "acceptedCore" : "acceptedNative"] = allowed
          action["explicitExpectedRefusal"] = expectedAcceptance == false && !allowed && available
          if allowed { engine.insertTranscript(value) }
          action["insertTranscriptCalled"] = allowed
        }
        action["executed"] = true
      } catch {
        failures.append(String(describing: error))
        action["error"] = String(describing: error)
      }
      return action
    }

    static func activeAcceptance(_ expected: R38Expected) -> Bool? {
      R38BehaviorRunner.platform == "core" ? expected.acceptedCore : expected.acceptedNative
    }

    func snapshot() throws -> [String: Any] {
      let suggestions = engine.suggestions
      track(text)
      for suggestion in suggestions.slots { track(suggestion.text) }
      let traits = engine.traits, policy = engine.policy
      let plain = ["autocorrects": policy.autocorrects,
                   "appliesTextReplacements": policy.appliesTextReplacements,
                   "showsSuggestions": policy.showsSuggestions, "learns": policy.learns,
                   "smartQuotes": policy.smartQuotes, "smartSpacing": policy.smartSpacing,
                   "autoSpaces": policy.autoSpaces]
      var counts: [String: Int] = [:]
      var pairs: [String: [String: Int]] = [:]
      var histories: [String: Int] = [:]
      for word in vocabulary.sorted() {
        counts[word] = stack.personal.count(of: word)
        histories[word] = stack.personal.historyCount(word)
        pairs[word] = Dictionary(uniqueKeysWithValues: stack.personal.successors(of: word).map { ($0.word, $0.count) })
      }
      let guardInfo = guardState()
      let permission = guardInfo["allowsAutomaticInsertion"] as! Bool
      var result: [String: Any] = [
        "text": text, "utf16Length": text.utf16.count,
        "selection": [selection.location, selection.length], "field": try r38JSON(field),
        "private": traits.isPrivate, "plain": plain.values.allSatisfy { $0 }, "plainBits": plain,
        "traits": ["keyboard": String(describing: traits.keyboard), "capitalization": String(describing: traits.capitalization),
                   "returnKey": String(describing: traits.returnKey), "isSecure": traits.isSecure,
                   "contentType": r38Nullable(traits.contentType), "autocorrection": r38Nullable(traits.autocorrection),
                   "spellChecking": r38Nullable(traits.spellChecking), "smartQuotes": r38Nullable(traits.smartQuotes),
                   "smartDashes": r38Nullable(traits.smartDashes), "prefersDarkAppearance": r38Nullable(traits.prefersDarkAppearance),
                   "enablesReturnKeyAutomatically": traits.enablesReturnKeyAutomatically],
        "policy": plain.merging(["allowsGlide": policy.allowsGlide]) { _, new in new },
        "policyCapitalization": String(describing: policy.capitalization),
        "suggestions": suggestions.slots.map { ["text": $0.text, "kind": String(describing: $0.kind),
                                                 "isAutocorrection": $0.isAutocorrection] as [String: Any] },
        "autocorrection": r38Nullable(suggestions.autocorrection), "suggestionsEmpty": suggestions.slots.isEmpty,
        "knownCounts": counts, "pairCounts": pairs, "historyCounts": histories,
        "totalCount": stack.personal.totalCount, "touchWeight": stack.touch.weight, "touchPositive": stack.touch.weight > 0,
        "touchIsLearning": stack.touch.isLearning,
        "destination": destinationState(), "guard": guardInfo,
        "allowsConfirmation": currentDestination().allowsConfirmation,
        "automaticCore": NSNull(), "automaticNative": NSNull(),
        "capturedChoice": captured.map { ["text": $0.text, "kind": String(describing: $0.kind)] } as Any? ?? NSNull(),
      ]
      result[R38BehaviorRunner.platform == "core" ? "automaticCore" : "automaticNative"] = permission
#if canImport(UIKit)
      result["nativeTraits"] = ["contentType": r38Nullable(native.textView.textContentType?.rawValue),
                                "keyboardType": native.textView.keyboardType.rawValue,
                                "isSecureTextEntry": native.textView.isSecureTextEntry,

      return result
    }

    func compare(_ expected: R38Expected, actual: [String: Any], action: [String: Any]) -> [String] {
      var failures: [String] = []
      func check<T: Equatable>(_ key: String, _ expectation: T?, _ value: T?) {
        guard let expectation else { return }
        if value != expectation {
          failures.append("\(key): expected \(String(reflecting: expectation)), got \(String(reflecting: value))")
        }
      }
      // String == is canonically equivalent in Swift; the contract requires
      // exact complete-document text, so compare the actual UTF-16 units.
      check("text.UTF16", Array(expected.text.utf16), Array(text.utf16))
      check("selection", expected.selection, [selection.location, selection.length])
      check("private", expected.private, actual["private"] as? Bool)
      if let plain = expected.plain, let bits = actual["plainBits"] as? [String: Bool] {
        for key in bits.keys.sorted() { check("plain.\(key)", plain, bits[key]) }
      }
      check("suggestionsEmpty", expected.suggestionsEmpty, actual["suggestionsEmpty"] as? Bool)
      for (word, count) in expected.knownCounts ?? [:] {
        check("knownCounts[\(String(reflecting: word))]", count, stack.personal.count(of: word))
      }
      check("totalCount", expected.totalCount, stack.personal.totalCount)
      check("touchWeight", expected.touchWeight, stack.touch.weight)
      check("touchPositive", expected.touchPositive, stack.touch.weight > 0)
      check("allowsConfirmation", expected.allowsConfirmation, currentDestination().allowsConfirmation)
      let automaticKey = R38BehaviorRunner.platform == "core" ? "automaticCore" : "automaticNative"
      check(automaticKey, R38BehaviorRunner.platform == "core" ? expected.automaticCore : expected.automaticNative,
            actual[automaticKey] as? Bool)
      let acceptedKey = R38BehaviorRunner.platform == "core" ? "acceptedCore" : "acceptedNative"
      check(acceptedKey, Self.activeAcceptance(expected), action[acceptedKey] as? Bool)
      return failures
    }
  }

  // Reject unknown schema keys instead of quietly ignoring misspelled or future
  // expectations. Missing op arguments are runtime errors so later steps run.
  private static func validateSchema(_ object: Any) throws {
    func dictionary(_ value: Any, _ path: String, _ allowed: Set<String>) throws -> [String: Any] {
      guard let value = value as? [String: Any] else { throw R38Error("\(path): expected object") }
      let extra = Set(value.keys).subtracting(allowed)
      guard extra.isEmpty else { throw R38Error("\(path): unsupported keys \(extra.sorted())") }
      return value
    }
    func field(_ value: Any, _ path: String) throws {
      _ = try dictionary(value, path, ["id", "route", "keyboard", "secure"])
    }
    let root = try dictionary(object, "$", ["version", "cases"])
    guard let cases = root["cases"] as? [Any] else { throw R38Error("$.cases: expected array") }
    for (index, value) in cases.enumerated() {
      let path = "$.cases[\(index)]"
      let test = try dictionary(value, path, ["id", "intent", "languages"])
      let variants = try dictionary(test["languages"] as Any, path + ".languages", Set(languages))
      for (language, value) in variants {
        let path = path + ".languages." + language
        let variant = try dictionary(value, path, ["initialText", "selection", "field", "steps"])
        try field(variant["field"] as Any, path + ".field")
        guard let steps = variant["steps"] as? [Any] else { throw R38Error("\(path).steps: expected array") }
        for (index, value) in steps.enumerated() {
          let path = path + ".steps[\(index)]"
          let step = try dictionary(value, path, ["op", "text", "selection", "field", "newIdentity", "kind", "expect"])
          if let value = step["field"] { try field(value, path + ".field") }
          _ = try dictionary(step["expect"] as Any, path + ".expect", [
            "text", "selection", "private", "plain", "suggestionsEmpty", "knownCounts", "totalCount",
            "touchWeight", "touchPositive", "allowsConfirmation", "automaticCore", "automaticNative", "acceptedCore", "acceptedNative",
          ])
        }
      }
    }
  }
