import Testing
@testable import KeyboardCore

/// Independent R14 source-review finding, added after rejection. This is a
/// trigger/conservation regression test, not another held-out accuracy case.
struct BoundaryCorrectionReviewTests {
  @Test func swipeDoesNotPretendAnExplicitSpaceSeparatedANewWordFromItsSuffix() {
    let (engine, document) = Fixtures.engine("join meeting")
    engine.traits.capitalization = .none
    document.moveCursor(to: 5)
    engine.documentDidChange()
    engine.type("teh")
    var random = SeededGenerator(state: 123)
    let path = SyntheticSwipe.points(for: "hello", noise: 0, generator: &random)
    let decoded = engine.insertGlide(path)
    #expect(decoded.first?.text == "hello")
    #expect(document.text == "join teh hello meeting")
    engine.deleteBackward()
    #expect(document.text == "join teh meeting")
    engine.deleteBackward()
    #expect(document.text == "join tehmeeting")
  }
}
