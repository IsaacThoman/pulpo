import XCTest

@MainActor
final class MultilingualEmojiInteractionTests: XCTestCase {
  func testRealSearchKeysAndRenderedResultTapsAcrossConfiguredLanguages() {
    continueAfterFailure = false
    let app = XCUIApplication()
    app.launchArguments = ["-PKPreview", "-PKResetSettings", "-PKResetLearning"]
    app.launch()
    XCTAssertTrue(app.key("key-q").waitForExistence(timeout: 10))
    for title in ["English", "Español", "English + Español"] {
      app.key("keyboard-language").tap()
      app.buttons[title].firstMatch.tap()
      XCTAssertTrue(app.key("keyboard-language").label.contains(title))
      for (query, emoji) in [("vicuna", "🦙"), ("bicicleta", "🚲"), ("red corazon", "❤️")] {
        let before = app.previewText
        app.key("key-emoji").tap()
        XCTAssertTrue(app.key("emoji-search-open").waitForExistence(timeout: 8))
        app.key("emoji-search-open").tap()
        for letter in query { app.key(letter == " " ? "key-space" : "key-\(letter)").tap() }
        XCTAssertEqual(app.previewText, before, "Query keys must not edit the real host")
        let result = app.key("emoji-result-\(emoji)")
        XCTAssertTrue(result.waitForExistence(timeout: 4))
        XCTAssertTrue(result.isHittable, "Expected target must be tappable in the native strip")
        result.tap()
        XCTAssertEqual(app.previewText, before + emoji)
        app.key("emoji-search-back").tap()
        app.buttons["emoji-abc"].tap()
      }
    }
  }
}
