import Foundation
@main struct Main {static func main() throws {
 var checked=0,contextChecked=0,failures:[[String:Any]]=[],special:[[String:Any]]=[]
 for n:UInt32 in 0...0x10FFFF {
  guard let u=Unicode.Scalar(n) else{continue};checked += 1
  let x=String(u)
  // Letter/mark and legacy letter/nonbase contexts can enter the prefix path.
  guard Policy.base(u) || Policy.mark(u) || CharacterSet.letters.contains(u) || CharacterSet.nonBaseCharacters.contains(u) else{continue}
  let contexts=[x+"bar","a"+x+"bar","n"+x+"bar",x+"'bar","a"+x+"'bar","n"+x+"'bar"]
  for q in contexts {
   contextChecked += 1
   let k=Policy.legacy(q),t=Policy.tokens(q),p=Policy.project(k),a=t.map(Policy.project),r=a.filter{!Policy.ignored.contains($0)}
   if !t.isEmpty && (!r.allSatisfy{$0.isEmpty || p.contains($0)} || !a.contains{$0.isEmpty || p.hasPrefix($0)}) {
    if failures.count<100 {failures.append(["scalar":String(format:"U+%04X",n),"query":q,"legacy":k,"T":t,"Plegacy":p,"A":a])}
   }
  }
  if Policy.mark(u) && !Policy.project(x).isEmpty {special.append(["scalar":String(format:"U+%04X",n),"projection":Policy.project(x),"attachedVowel":Policy.project("a"+x),"attachedConsonant":Policy.project("n"+x)])}
 }
 let result:[String:Any]=["scalarCount":checked,"contextCount":contextChecked,"first100NecessaryConditionFailures":failures,"markToLetterProjections":special,"note":"Finite single-scalar context scan, not universal proof. Failure rows need actual API confirmation; tests necessary V/I conditions against projected synthetic legacy key."]
 try JSONSerialization.data(withJSONObject:result,options:[.prettyPrinted,.sortedKeys]).write(to:URL(fileURLWithPath:"scalar-scan.json"))
 print("scalars",checked,"contexts",contextChecked,"recorded failures",failures.count,"letter-projecting marks",special.count)
}}
