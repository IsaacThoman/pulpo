import Foundation
let reviewLocale = Locale(identifier:"en_US_POSIX")
let ignored: Set<String> = ["a","an","and","for","in","of","on","the","to","with"]
func legacyNormal(_ text:String) -> String {
 let folded=text.folding(options:[.diacriticInsensitive,.caseInsensitive],locale:reviewLocale)
 var converted=""
 for u in folded.unicodeScalars {
  if CharacterSet.alphanumerics.contains(u) { converted.unicodeScalars.append(u) }
  else if u != "'" && u != "\u{2019}" { converted.append(" ") }
 }
 return converted.split(whereSeparator:{$0.isWhitespace}).joined(separator:" ")
}
func exactFold(_ text:String)->String { text.folding(options:[.diacriticInsensitive,.caseInsensitive],locale:reviewLocale).split(whereSeparator:{$0.isWhitespace}).joined(separator:" ") }
func literalWords(_ text:String)->[String] { Set(text.split(separator:" ").map(String.init)).sorted() }
func referenceTokens(_ text:String)->[String] {
 var result:[String]=[], current:[Unicode.Scalar]=[], vowel=false
 func finish() { if !current.isEmpty { result.append(String(String.UnicodeScalarView(current)).precomposedStringWithCanonicalMapping) };current=[];vowel=false }
 for u in text.lowercased(with:reviewLocale).decomposedStringWithCanonicalMapping.unicodeScalars {
  let cat=u.properties.generalCategory
  if cat == .nonspacingMark || cat == .spacingMark || cat == .enclosingMark {
   if !current.isEmpty && !vowel { current.append(u) }
  } else if CharacterSet.letters.contains(u) && [.uppercaseLetter,.lowercaseLetter,.titlecaseLetter,.modifierLetter,.otherLetter].contains(cat) {
   current.append(u);vowel=["a","e","i","o","u"].contains(String(u))
  } else { finish() }
 }
 finish();return Set(result).sorted()
}
func policy(_ row:[String:Any], _ index:[[String:Any]])->[String:Any] {
 let q=row["q"] as! String,cap=min(row["cap"] as! Int,60)
 let cat=row["catalog"] as! [String:Any]
 let es=(cat["categories"] as! [[String:Any]]).flatMap{$0["emoji"] as! [[String:Any]]}
 let t=referenceTokens(q),w=literalWords(legacyNormal(q)),x=literalWords(exactFold(q))
 var keys=Array(repeating:[String](),count:es.count)
 for term in index { for m in term["matches"] as! [[String:Int]] { keys[m["id"]!].append(term["key"] as! String) } }
 var envelope:[Int]=[];var scores:[(Int,Int,Int,Int)]=[]
 for (i,e) in es.enumerated() {
  let texts=[e["n"] as! String]+(e["k"] as? [String] ?? [])
  let ordinary=Set(texts.flatMap{literalWords(legacyNormal($0))})
  let v=ordinary.union(keys[i]).union(keys[i].flatMap{literalWords($0)})
  let exact=literalWords(exactFold(e["n"] as! String)) == x
  let anchored=w.contains { a in keys[i].contains { $0.hasPrefix(a) } }
  let covered=w.filter{!ignored.contains($0)}.allSatisfy { a in v.contains { $0.hasPrefix(a) } }
  if !w.isEmpty && (exact || anchored && covered) { envelope.append(i) }
  let sp=referenceTokens(e["s"] as? String ?? "")
  var en=ordinary
  for text in texts { let n=legacyNormal(text);en.insert(n);en.insert(literalRawWords(n).joined()) }
  let pairs=t.map{($0,legacyNormal($0))}
  let eligible=pairs.allSatisfy { a,b in sp.contains{$0.hasPrefix(a)} || en.contains{$0.hasPrefix(b)} }
  let hasSpanish=t.contains { a in sp.contains{$0.hasPrefix(a)} }
  if eligible && hasSpanish {
   let completions=pairs.filter { a,b in !sp.contains(a) && !en.contains(b) }.count
   scores.append((sp==t ? 0:1,completions,sp.count,i))
  }
 }
 let eSet=Set(envelope)
 scores=scores.filter{!eSet.contains($0.3)}.sorted { a,b in
  if a.0 != b.0{return a.0<b.0};if a.1 != b.1{return a.1<b.1};if a.2 != b.2{return a.2<b.2};return a.3<b.3
 }
 let active=cap>0 && q.count<=80 && !w.isEmpty && !t.isEmpty && !(t.count==1 && t[0].utf8.count==1)
 let available=active ? max(0,cap-min(cap,envelope.count)):0
 return ["T":t,"W":w,"X":x,"E":envelope.map{es[$0]["e"]!},"rankedEligibleOutsideE":scores.map{es[$0.3]["e"]!},"suffix":scores.prefix(available).map{es[$0.3]["e"]!}]
}
func literalRawWords(_ s:String)->[String] {s.split(separator:" ").map(String.init)}
