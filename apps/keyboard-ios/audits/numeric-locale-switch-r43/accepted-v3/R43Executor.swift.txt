 final class Host: KeyboardHost, @preconcurrency TextDocument {
  let field = UITextField()
  let documentIdentifier: UUID? = UUID()
  var document: TextDocument { self }
  var inputTraits: InputTraits { InputTraits(field) }
  var needsInputModeSwitchKey = false
  var hasFullAccess: Bool { false }
  var isInsideApp: Bool { false }
  var advances = 0; var lists = 0; var typingLanguage = ""
  var documentContextBeforeInput: String? {
   guard let r = field.selectedTextRange else { return field.text ?? "" }
   return field.text(in: field.textRange(from: field.beginningOfDocument, to: r.start)!)
  }
  var documentContextAfterInput: String? {
   guard let r = field.selectedTextRange else { return "" }
   return field.text(in: field.textRange(from: r.end, to: field.endOfDocument)!)
  }
  var selectedText: String? { field.selectedTextRange.flatMap { field.text(in: $0) }.flatMap { $0.isEmpty ? nil : $0 } }
  var hasText: Bool { field.hasText }
  func insertText(_ text: String) { field.insertText(text) }
  func deleteBackward() { field.deleteBackward() }
  func advanceToNextInputMode() { advances += 1 }
  func handleInputModeList(from view: UIView, with event: UIEvent) { lists += 1 }
  func adjustTextPosition(byCharacterOffset offset: Int) {}
  func openApp(_ url: URL) {}
  func setTypingLanguage(_ language: String) { typingLanguage = language }
 }
 func keyboard(_ name: String) throws -> UIKeyboardType {
  switch name {
  case "decimal": .decimalPad
  case "number": .numberPad
  case "phone": .phonePad
  case "asciiNumber": .asciiCapableNumberPad
  case "standard": .default
  case "email": .emailAddress
  case "url": .URL
  default: throw NSError(domain: "unknown keyboard \(name)", code: 1)
  }
 }
 func run(_ row: R43Workflow) throws -> [String: Any] {
  let host = Host()
  host.field.keyboardType = try keyboard(row.keyboard)
  host.field.autocapitalizationType = .none
  host.field.autocorrectionType = .no
  host.field.smartQuotesType = .no
  host.field.smartDashesType = .no
  host.field.isSecureTextEntry = row.privateField
  host.field.clearsOnInsertion = false
  host.field.text = row.initialText
  host.field.selectedTextRange = host.field.textRange(from: host.field.endOfDocument, to: host.field.endOfDocument)
  host.needsInputModeSwitchKey = row.globe
  var settings = KeyboardSettings()
  settings.language = try XCTUnwrap(KeyboardLanguage(rawValue: row.language))
  settings.haptics = false; settings.sounds = false
  SettingsStore(defaults: .standard).save(settings)
  let c = KeyboardController(host: host, dictationBridge: nil)
  c.hostWillAppear()
  let root = c.rootView
  root.frame = CGRect(x: 0, y: 100, width: row.width, height: c.preferredHeight)
  c.layoutChanged(dark: false)
  root.frame.size.height = c.preferredHeight
  root.oneHanded = try XCTUnwrap(OneHandedMode(rawValue: row.oneHanded))
  root.layoutIfNeeded()
  let vc = UIViewController()
  let window = UIWindow(frame: CGRect(x: 0, y: 0, width: max(row.width, 320), height: 600))
  window.rootViewController = vc; window.makeKeyAndVisible()
  host.field.frame = CGRect(x: 8, y: 20, width: 280, height: 44)
  vc.view.addSubview(host.field); vc.view.addSubview(root)
  host.field.selectedTextRange = host.field.textRange(from:host.field.endOfDocument,to:host.field.endOfDocument)
  root.layoutIfNeeded(); root.keys.layoutIfNeeded()
  let learnedBefore = c.stack?.personal.totalCount
  var trace: [[String: Any]] = []; var errors: [String] = []
  if Locale(identifier: row.locale).decimalSeparator != Locale.current.decimalSeparator { errors.append("locale separator mismatch") }
  if Locale.current.region != Locale(identifier: row.locale).region { errors.append("locale region mismatch") }
  if c.stack == nil || c.stack?.lexicon.wordCount == 0 { errors.append("missing or empty language stack") }
  if c.engine.settings.language.rawValue != row.language { errors.append("language fallback") }
  let dirtyBefore = c.stack?.personal.hasUnsavedChanges
  let initialGeometry = checkGeometry(root.keys, controller: c, row:row)
  let initialKeys = root.keys.keyViews.filter { $0.key.action != .text("") }.map { $0.key.identifier }
  let initialFramesInside = root.keys.keyViews.allSatisfy { root.keys.bounds.contains($0.frame) }


  func tap(_ view: KeyView, globe: Bool = false) {
   let point = CGPoint(x: view.frame.midX, y: view.frame.midY)
   let touch = Contact(point)
   let event: UIEvent? = globe ? UIEvent() : nil
   root.keys.touchesBegan([touch], with: event)
   root.keys.touchesEnded([touch], with: event)
  }
  func stackIdentity() -> [String:String] {
   ["controller":String(describing:ObjectIdentifier(c)),"keysView":String(describing:ObjectIdentifier(root.keys)),"engine":String(describing:ObjectIdentifier(c.engine)),"dictionary":c.stack.map { String(describing:ObjectIdentifier($0.lexicon)) } ?? ""]
  }
  for (actionIndex, action) in row.actions.enumerated() {
   let countBefore = c.stack?.personal.totalCount ?? -1
   let dirtyActionBefore = c.stack?.personal.hasUnsavedChanges ?? false
   let before = host.field.text ?? ""
   let beforeCaret = host.field.selectedTextRange.map { host.field.offset(from: host.field.beginningOfDocument, to:$0.start) } ?? -1
   var error: String?
   switch action.kind {
   case "tap", "key", "delete", "globeTouch", "globeAX":
    let match: (KeyView) -> Bool = { view in
     switch action.kind {
     case "tap": view.key.action == .character(action.value ?? "")
     case "key": view.key.identifier == action.value && view.key.action != .nextKeyboard
     case "delete": view.key.action == .delete
     default: view.key.action == .nextKeyboard
     }
    }
    if let view = root.keys.keyViews.first(where: match) {
     if !root.keys.bounds.contains(view.frame) { error = "key outside bounds" }
     if action.kind == "globeAX" {
      if !view.accessibilityActivate() { error = "AX activation rejected" }
     } else { tap(view, globe: action.kind == "globeTouch") }
    } else { error = "missing action \(action.kind):\(action.value ?? "")" }
   case "refresh": break
   case "field":
    host.field.keyboardType = try keyboard(action.value ?? "")
   default: error = "unsupported action \(action.kind)"
   }
   c.documentDidChange(); root.layoutIfNeeded(); root.keys.layoutIfNeeded()
   if let error { errors.append(error) }
   var actionObject: [String:Any] = ["kind":action.kind]
   if let value=action.value { actionObject["value"]=value }
   trace.append(["index":actionIndex, "action":actionObject, "beforeText":before, "afterText":host.field.text ?? "", "beforeCaret":beforeCaret, "afterCaret":host.field.selectedTextRange.map { host.field.offset(from:host.field.beginningOfDocument,to:$0.start) } ?? -1, "advance":host.advances, "lists":host.lists, "documentDidChangeCount":1, "errors":error.map { [$0] } ?? [], "hostLanguage":host.typingLanguage, "loadedLanguage":c.engine.settings.language.rawValue, "stack":stackIdentity(), "learning":["policy":c.engine.policy.learns ? "allowed":"disabled", "beforeCount":countBefore, "afterCount":c.stack?.personal.totalCount ?? -1, "beforeDirty":dirtyActionBefore,"afterDirty":c.stack?.personal.hasUnsavedChanges ?? false]])
  }
  let geometry = checkGeometry(root.keys, controller: c, row:row)
  let frames: [[String: Any]] = root.keys.keyViews.map { v in
   ["id": v.key.identifier, "label": v.accessibilityLabel ?? "", "x": v.frame.minX, "y": v.frame.minY, "w": v.frame.width, "h": v.frame.height, "inside": root.keys.bounds.contains(v.frame), "centerRoutes": root.hitTest(v.convert(CGPoint(x: v.bounds.midX, y: v.bounds.midY), to: root), with: nil) === root.keys]
  }
  let image = UIGraphicsImageRenderer(bounds: vc.view.bounds).image { vc.view.layer.render(in: $0.cgContext) }
  let attachment = XCTAttachment(image: image); attachment.name = row.id; attachment.lifetime = .keepAlways; add(attachment)
  var result: [String: Any] = ["id":row.id, "errors":errors, "text":host.field.text ?? "", "advance":host.advances,"lists":host.lists,"locale":["currentIdentifier":Locale.current.identifier,"autoupdatingIdentifier":Locale.autoupdatingCurrent.identifier,"currentRegion":Locale.current.region?.identifier ?? "","autoupdatingRegion":Locale.autoupdatingCurrent.region?.identifier ?? "","currentDecimalSeparator":Locale.current.decimalSeparator ?? "<nil>","autoupdatingDecimalSeparator":Locale.autoupdatingCurrent.decimalSeparator ?? "<nil>"],"requestedLanguage":row.language,"hostLanguage":host.typingLanguage,"hostTypingLanguage":host.typingLanguage,"loadedLanguage":c.engine.settings.language.rawValue,"stack":stackIdentity(),"initialText":row.initialText,"initialAdvance":0,"initialLists":0,"initialKeys":initialKeys,"initialGeometry":initialGeometry,"finalGeometry":geometry,"actions":trace,"frames":frames,"learnedBefore":learnedBefore ?? -1,"learnedAfter":c.stack?.personal.totalCount ?? -1,"dirtyBefore":dirtyBefore ?? false,"dirtyAfter":c.stack?.personal.hasUnsavedChanges ?? false,"processArguments":ProcessInfo.processInfo.arguments,"argumentLocale":UserDefaults.standard.string(forKey:"AppleLocale") ?? "","preferredLanguages":Locale.preferredLanguages,"bundleLocalizations":Bundle.main.localizations,"stackAvailable":c.stack != nil,"initialFramesInside":initialFramesInside]

  c.hostDidDisappear(); result["learnedAfterDisappear"] = c.stack?.personal.totalCount ?? -1; result["dirtyAfterDisappear"] = c.stack?.personal.hasUnsavedChanges ?? false; window.isHidden = true
  return result
 }

  var keysViewAllowsGlide: Bool { false }; var keysViewShowsPopups: Bool { false }
  func keysView(_ view: KeysView, touchDown key: Key) { chosen = key.identifier }
  func keysView(_ view: KeysView, activate key: Key, at point: CGPoint) { chosen = key.identifier }
  func keysView(_ view: KeysView, insertAlternate text: String, for key: Key) {}

  func keysViewRepeatDelete(_ view: KeysView, wholeWord: Bool) {}

  func keysViewShiftUp(_ view: KeysView) {}

 }
 func checkGeometry(_ keys: KeysView, controller: KeyboardController, row: R43Workflow) -> [String: Any] {
  let recorder = Recorder(); keys.delegate = recorder
  defer { keys.delegate = controller }
  let rows = Dictionary(grouping: keys.keyViews, by: { $0.frame.minY }).sorted { $0.key < $1.key }
  var result: [[String: Any]] = []
  for (_, pair) in rows.enumerated() {
   let views = pair.value.sorted { $0.frame.minX < $1.frame.minX }
   for (_, v) in views.enumerated() where v.key.action != .text("") {
    let left = v.frame.minX
    let right = v.frame.maxX
    let top = max(0, v.frame.minY - 1)
    let bottom = min(keys.bounds.height, v.frame.maxY + 1)
    var routes: [Bool] = []
    var points = [CGPoint(x: v.frame.midX, y: v.frame.midY), CGPoint(x:left+0.5,y:top+0.5), CGPoint(x:right-0.5,y:top+0.5), CGPoint(x:left+0.5,y:bottom-0.5), CGPoint(x:right-0.5,y:bottom-0.5)]
    for y in stride(from: top+0.5, to: bottom, by: 2) {
     for x in stride(from: left+0.5, to: right, by: 2) { points.append(CGPoint(x:x,y:y)) }
    }
    var identityRoutes = true; var rootRoutes = true
    for p in points {
     identityRoutes = identityRoutes && keys.keyIndex(at:p).map { keys.keyViews[$0] === v } == true
     rootRoutes = rootRoutes && keys.superview?.hitTest(keys.convert(p, to: keys.superview), with: nil) === keys
    }
    for p in points.prefix(5) {
     recorder.chosen = nil; recorder.lists = 0
     let touch = Contact(p); let event = UIEvent()
     keys.touchesBegan([touch], with: event); keys.touchesEnded([touch], with: event)
     routes.append(v.key.action == .nextKeyboard ? recorder.lists == 2 : recorder.chosen == v.key.identifier)
    }
    func rect(_ r: CGRect) -> [Double] { [r.minX,r.minY,r.width,r.height].map(Double.init) }
    let target = CGRect(x:left,y:top,width:right-left,height:bottom-top)
    result.append(["id":v.key.identifier,"text":v.key.isCharacter ? v.key.label : "","cap":rect(v.frame),"target":rect(target),"centerRoutes":keys.superview?.hitTest(keys.convert(CGPoint(x:v.frame.midX,y:v.frame.midY),to:keys.superview),with:nil) === keys,"targetRoutes":identityRoutes,"identityRoutes":routes.allSatisfy { $0 },"rootRoutes":rootRoutes,"sampleCount":points.count,"sampleStep":2,"accessibilityLabel":v.accessibilityLabel ?? ""])

   }
  }
  return ["width":row.width,"height":keys.bounds.height,"grid":[0,0,keys.bounds.width,keys.bounds.height],"requestedOneHanded":row.oneHanded,"effectiveOneHanded":controller.rootView.canDock ? controller.rootView.oneHanded.rawValue : "off","effectiveWidth":keys.bounds.width,"keys":result]
 }
 func testExecute() throws {
  KeyboardResources.directory = URL(fileURLWithPath: "/Users/isaacthoman/.t3/worktrees/pulpo/t3-b262ad7a/apps/keyboard-ios/LanguageData")
  let path = "/tmp/r43-input.json"
  let rows = try JSONDecoder().decode([R43Workflow].self, from: Data(contentsOf: URL(fileURLWithPath: path)))
  XCTAssertFalse(rows.isEmpty)
  var results: [[String: Any]] = []
  for row in rows { results.append(try run(row)) }
  var dictionaries: [String:Any] = [:]
  for language in KeyboardLanguage.allCases {
   let url = KeyboardResources.directory!.appendingPathComponent(language.rawValue + ".pkdict")
   let bytes = try Data(contentsOf:url)
   dictionaries[language.rawValue] = ["available":true,"bytes":bytes.count,"sha256":SHA256.hash(data:bytes).map { String(format:"%02x",$0) }.joined()]
  }
  let data = try JSONSerialization.data(withJSONObject: ["schemaVersion":1,"dictionaries":dictionaries,"rows":results], options: [.prettyPrinted, .sortedKeys])
  try data.write(to: URL(fileURLWithPath: "/tmp/r43-output.json"))
  let a = XCTAttachment(data: data, uniformTypeIdentifier: "public.json"); a.name = "r43-results"; a.lifetime = .keepAlways; add(a)
  XCTAssertEqual(results.count, rows.count)
 }
