import KeyboardCore
import UIKit
import XCTest
@testable import KeyboardUI
@testable import PulpoKeyboard

@MainActor
final class R34NativeDocumentTests: XCTestCase {
  private func descendants(_ view: UIView) -> [UIView] { [view] + view.subviews.flatMap(descendants) }
  private func pump() { RunLoop.main.run(until: Date().addingTimeInterval(0.025)) }

  func testSelectedUnicodePrivateTraitsAndSearchOwnership() throws {
    let old = SettingsStore().load()
    defer { SettingsStore().save(old) }
    let scene = try XCTUnwrap(UIApplication.shared.connectedScenes.compactMap { $0 as? UIWindowScene }.first)
    let window = UIWindow(windowScene: scene)
    window.rootViewController = UIViewController()
    window.makeKeyAndVisible()
    defer { window.isHidden = true; window.rootViewController = nil }
    let text = UITextView(frame: CGRect(x: 12, y: 60, width: window.bounds.width - 24, height: 150))
    window.rootViewController!.view.addSubview(text)
    let host = PreviewHost(textView: text) { _ in }
    let keyboard = KeyboardController(host: host, dictationBridge: nil)
    keyboard.rootView.frame = CGRect(x: 0, y: 240, width: window.bounds.width, height: keyboard.preferredHeight)
    window.rootViewController!.view.addSubview(keyboard.rootView)
    keyboard.hostWillAppear()
    defer { keyboard.hostDidDisappear(); host.invalidate() }
    let deadline = Date().addingTimeInterval(60)
    while keyboard.engine.emojiSuggester == nil && Date() < deadline { pump() }
    XCTAssertNotNil(keyboard.engine.emojiSuggester)
    for language in KeyboardLanguage.allCases {
      keyboard.rootView.bar.onLanguage?(language)
      XCTAssertEqual(keyboard.engine.settings.language, language)
      for content in [nil, UITextContentType.password, .username, .oneTimeCode, .emailAddress, .URL] {
        text.isSecureTextEntry = content == .password
        text.textContentType = content
        text.keyboardType = content == .emailAddress ? .emailAddress : (content == .URL ? .URL : .default)
        let selected = "e\u{301}👩🏿‍🚀"
        let original = "Inicio 🧑🏽‍🦯 [" + selected + "] final 🇪🇸"
        let selection = (original as NSString).range(of: selected)
        text.text = original
        text.selectedRange = selection
        keyboard.documentDidChange()
        keyboard.keysView(keyboard.rootView.keys, activate: Key(action: .emoji, label: "emoji"), at: .zero)
        keyboard.rootView.layoutIfNeeded()
        keyboard.emojiPanelSearch(try XCTUnwrap(keyboard.rootView.emoji))
        for letter in "keyboard teclado" {
          let action: KeyAction = letter == " " ? .space : .character(String(letter))
          let key = try XCTUnwrap(keyboard.rootView.keys.keyViews.first { $0.key.action == action })
          XCTAssertTrue(key.accessibilityActivate())
          keyboard.rootView.layoutIfNeeded()
        }
        XCTAssertEqual(text.text, original)
        XCTAssertEqual(text.selectedRange, selection)
        XCTAssertEqual(keyboard.rootView.search.query, "keyboard teclado")
        let result = try XCTUnwrap(descendants(keyboard.rootView.search).compactMap { $0 as? EmojiResultControl }.first { $0.result.value == "⌨️" })
        XCTAssertNotNil(result.imageView.image)
        XCTAssertGreaterThan(result.bounds.width, 0)
        result.sendActions(for: .touchUpInside)
        XCTAssertEqual(text.text, original.replacingOccurrences(of: selected, with: "⌨️"))
        XCTAssertEqual((text.text as NSString).substring(from: text.selectedRange.location), "] final 🇪🇸")
        keyboard.rootView.search.onBack?()
        keyboard.keysView(keyboard.rootView.keys, activate: Key(action: .page(.letters), label: "ABC"), at: .zero)
        XCTAssertTrue(keyboard.rootView.search.isHidden)
      }
    }
  }

  func testColdIndexAndFirstSearch() throws {
    guard let catalogPath = ProcessInfo.processInfo.environment["PK_R34_CATALOG"],
          let output = ProcessInfo.processInfo.environment["PK_R34_OUTPUT"] else { throw XCTSkip("R34 instrumented paired invocation only") }
    let start = CACurrentMediaTime()
    let catalog = try EmojiCatalog.load(url: URL(fileURLWithPath: catalogPath))
    let decoded = CACurrentMediaTime()
    let index = EmojiSuggester(catalog: catalog)
    let built = CACurrentMediaTime()
    let values = index.search("teclado").map(\.value)
    let searched = CACurrentMediaTime()
    let data = try JSONSerialization.data(withJSONObject: ["pid": ProcessInfo.processInfo.processIdentifier, "decodeMS": (decoded-start)*1000, "constructionMS": (built-decoded)*1000, "firstSearchMS": (searched-built)*1000, "values": values, "native": "iOS27.0 simulator, hosted XCTest, Debug, first explicit index/search in fresh process"])
    let url = URL(fileURLWithPath: output + "-\(ProcessInfo.processInfo.processIdentifier).json")
    try data.write(to: url)
    let attachment = XCTAttachment(data: data, uniformTypeIdentifier: "public.json")
    attachment.name = "R34 native cold measurements"; attachment.lifetime = .keepAlways; add(attachment)
  }
}
