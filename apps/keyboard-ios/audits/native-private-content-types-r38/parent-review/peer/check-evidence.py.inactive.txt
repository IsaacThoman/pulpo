from pathlib import Path
from collections import Counter
import json,gzip,hashlib
ROOT=Path('/Users/isaacthoman/.t3/worktrees/pulpo/t3-b947196b')
P=ROOT/'apps/keyboard-ios/audits/native-private-content-types-r38'
OUT=Path('/tmp/pulpo-r38-parent-review')
def need(ok,msg):
 if not ok: raise ValueError(msg)
def raw(path):
 if path.is_file(): return path.read_bytes()
 q=Path(str(path)+'.gz')
 if q.is_file(): return gzip.decompress(q.read_bytes())
 raise ValueError('missing '+str(path))
def data(path):
 b=path.read_bytes();return json.loads(gzip.decompress(b) if path.suffix=='.gz' else b)
def sha(b): return hashlib.sha256(b).hexdigest()
def u16(s): return s.encode('utf-16-le','surrogatepass')
def eq(a,b):
 if type(a) in (int,float) and type(b) in (int,float): return a==b
 return type(a)==type(b) and a==b
bindings={}
for name,base in [('contract-freeze.json',P),('candidate-freeze.json',ROOT),('heldout-seal.json',P)]:
 x=data(P/name)
 for path,h in x['hashes'].items():
  need(sha(raw(base/path))==h,'hash '+path)
  bindings[name+':'+path]=h
seal=data(P/'heldout-seal.json')
need(sha((P/'candidate-freeze.json').read_bytes())==seal['candidateFreezeSHA256'],'freeze/seal')
payload=data(P/'blind/heldout-v2.json.gz')
expected={(c['id'],lang):(c,v) for c in payload['cases'] for lang,v in c['languages'].items()}
need(len(expected)==54,'54 variants')
need(all(set(c['languages'])=={'en_US','es','en_es'} for c in payload['cases']),'language Cartesian')
summary={'runtimeExecuted':False,'hashBindingsVerified':len(bindings),'inputSHA256':sha(gzip.decompress((P/'blind/heldout-v2.json.gz').read_bytes())),'platforms':{},'rawReportSHA256':{}}
reports={}
for platform in ['core','native']:
 reports[platform]={}
 for version in ['baseline','candidate']:
  name=f'heldout-{platform}-{version}.json.gz';r=data(P/name)
  summary['rawReportSHA256'][name]={'archive':sha((P/name).read_bytes()),'original':sha(gzip.decompress((P/name).read_bytes()))}
  need(r['input']==payload,'embedded input')
  need(r['platform']==platform,'platform')
  rows={(x['id'],x['language']):x for x in r['results']}
  need(len(rows)==len(r['results'])==54 and rows.keys()==expected.keys(),'unique complete rows')
  failureKinds=Counter();scores={};casePassed={};challenges=Counter();witness=[]
  for key,(case,variant) in expected.items():
   row=rows[key];need(row['intent']==case['intent'],'intent binding')
   need(len(row['steps'])==len(variant['steps']),'steps count')
   caseFail=[];prev=row['initial']
   for i,(step,source) in enumerate(zip(row['steps'],variant['steps'])):
    need(step['index']==i and step['op']==source['op'],'step identity')
    need(step['expected']==source['expect'],'frozen expectations')
    a=step['actual'];act=step['action'];e=source['expect'];fail=[]
    need(eq(act['executed'],True) and 'error' not in act,'execution failure')
    for k in ['text','selection','field','private','totalCount','touchWeight']:
     need(eq(step['before'][k],prev[k]),'broken state chain '+str((key,i,k)))
    need(eq(a['touchPositive'],a['touchWeight']>0),'touch flag')
    need(eq(a['suggestionsEmpty'],len(a['suggestions'])==0),'suggestion flag')
    need(a['utf16Length']==len(u16(a['text']))//2,'length')
    need(len(a['selection'])==2 and all(type(x)==int and x>=0 for x in a['selection']) and sum(a['selection'])<=a['utf16Length'],'selection bounds')
    for k,v in e.items():
     if k in ['automaticCore','acceptedCore'] and platform!='core': continue
     if k in ['automaticNative','acceptedNative'] and platform!='native': continue
     if k=='text':
      if u16(a[k])!=u16(v): fail.append('text.UTF16')
     elif k=='knownCounts':
      for word,n in v.items():
       if not eq(a[k][word],n): fail.append('knownCounts['+json.dumps(word,ensure_ascii=False)+']')
     elif k=='plain':
      need(set(a['plainBits'])=={'autocorrects','appliesTextReplacements','showsSuggestions','learns','smartQuotes','smartSpacing','autoSpaces'},'plain bits')
      for bit,b in a['plainBits'].items():
       if not eq(b,v): fail.append('plain.'+bit)
     else:
      val=act[k] if k.startswith('accepted') else a[k]
      if not eq(val,v): fail.append(k)
    need(type(step['passed'])==bool and step['passed']==(not fail),'transition score '+str((platform,version,key,i,step['passed'],fail,step['failures'])))
    reported=[f.split(': expected',1)[0] for f in step['failures']]
    need(Counter(reported)==Counter(fail),'failure keys '+str((key,i,reported,fail)))
    failureKinds.update(fail);caseFail.extend(fail);scores[(*key,i)]=not fail
    if act.get('challengeFired') is True:
     challenges['fired']+=1
     need(type(act['accepted'])==bool and type(act['available'])==bool,'challenge flags')
     if act['available']:
      challenges['accepted' if act['accepted'] else 'denied']+=1
     else:
      challenges['unavailable']+=1
      need(act['explicitExpectedRefusal'] is False,'unavailable credit')
    if fail or key[0] in ['r38-normal-learning-control','r38-secure-trait-transitions']:
     witness.append({'id':key[0],'language':key[1],'step':i,'op':source['op'],'failures':fail,'actual':{k:a[k] for k in ['text','selection','field','private','totalCount','touchWeight']},'expected':e})
    prev=a
   need(len(row['failures'])==len(caseFail),'row failure count')
   need(row['status']==('passed' if not caseFail else 'failed'),'case score')
   casePassed[key]=not caseFail
  need(r['failureCount']==len(r['failures'])==sum(failureKinds.values()),'report aggregate')
  need(eq(r['passed'],not failureKinds),'report score')
  reports[platform][version]={'rows':rows,'scores':scores,'casePassed':casePassed}
  summary['platforms'].setdefault(platform,{})[version]={'cases':len(rows),'passedCases':sum(casePassed.values()),'transitions':len(scores),'passedTransitions':sum(scores.values()),'failureKinds':dict(failureKinds),'challenges':dict(challenges)}
  (OUT/f'{platform}-{version}-witnesses.json').write_text(json.dumps(witness,ensure_ascii=False,indent=2)+'\n')
 b,c=reports[platform]['baseline'],reports[platform]['candidate']
 comparison=Counter();loss=[];newFailures=[]
 for k,passed in b['scores'].items():
  cp=c['scores'][k];comparison['commonPass' if passed and cp else 'gain' if cp else 'loss' if passed else 'commonFailure']+=1
  if passed and not cp: loss.append(k)
  bs=b['rows'][k[:2]]['steps'][k[2]];cs=c['rows'][k[:2]]['steps'][k[2]]
  bf={f.split(': expected',1)[0] for f in bs['failures']};cf={f.split(': expected',1)[0] for f in cs['failures']}
  if cf-bf:newFailures.append({'transition':k,'newFailureKinds':sorted(cf-bf)})
 summary['platforms'][platform]['paired']={'transitions':dict(comparison),'losses':loss,'newFailureKinds':newFailures}
need(all(reports['native']['baseline']['rows'][('r38-secure-trait-transitions',l)]['steps'][i]['actual']==reports['native']['candidate']['rows'][('r38-secure-trait-transitions',l)]['steps'][i]['actual'] for l in ['en_US','es','en_es'] for i in range(len(reports['native']['baseline']['rows'][('r38-secure-trait-transitions',l)]['steps']))),'secure case changed actual state')
summary['nativeSecureCaseActualStatesIdenticalBetweenVersions']=True
(OUT/'evidence-check.json').write_text(json.dumps(summary,indent=2,ensure_ascii=False)+'\n')
print(json.dumps(summary,ensure_ascii=False,indent=2))
